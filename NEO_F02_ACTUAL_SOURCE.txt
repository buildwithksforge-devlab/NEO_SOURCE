/** True when the folio state permits a revert without accounting side effects. */
export function revertAllowed(input: {
  hasStay: boolean;
  checkedOut: boolean;
  liveFolioLines: number;
  reason: string;
}): { allowed: boolean; reason?: string } {
  if (!input.reason.trim()) return { allowed: false, reason: "reason_required" };
  if (!input.hasStay) return { allowed: false, reason: "not_checked_in" };
  if (input.checkedOut) return { allowed: false, reason: "already_checked_out" };
  if (input.liveFolioLines > 0) return { allowed: false, reason: "folio_has_activity" };
  return { allowed: true };
}

/** How a policy reads to the desk, in words rather than in configuration. */
export function describePolicy(policy: Pick<DepositPolicy, "basis" | "value" | "minimum_amount">) {
  const base =
    policy.basis === "percentage"
      ? `${policy.value}% of the stay value`
      : policy.basis === "nights"
        ? `${policy.value} night(s) at the nightly rate`
        : `A flat ${policy.value}`;
  return policy.minimum_amount > 0 ? `${base}, minimum ${policy.minimum_amount}` : base;
}

export function useDepositPolicies() {
  const { organizationId } = useAuth();
  return useQuery({
    queryKey: ["pms", "deposit-policies", organizationId],
    enabled: Boolean(organizationId),
    queryFn: async (): Promise<DepositPolicy[]> => {
      const { data, error } = await db
        .from("deposit_policies")
        .select("*")
        .order("priority", { ascending: true });
      if (error) throw new Error(error.message);
      // eslint-disable-next-line @typescript-eslint/no-explicit-any
      return ((data ?? []) as any[]).map((r) => ({
        id: String(r.id),
        branch_id: r.branch_id ?? null,
        rate_plan_id: r.rate_plan_id ?? null,

        name: String(r.name ?? ""),
        basis: (r.basis ?? "percentage") as DepositBasis,
        value: Number(r.value ?? 0),
        minimum_amount: Number(r.minimum_amount ?? 0),
        reason: String(r.reason ?? ""),
        priority: Number(r.priority ?? 100),
        active: Boolean(r.active),
      }));
    },
  });
}

export function useSaveDepositPolicy() {
  const { organizationId } = useAuth();
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async (input: Partial<DepositPolicy> & { name: string }) => {
      const payload = {
        organization_id: organizationId,
        branch_id: input.branch_id ?? null,
        rate_plan_id: input.rate_plan_id ?? null,

        name: input.name,
        basis: input.basis ?? "percentage",
        value: Number(input.value ?? 0),
        minimum_amount: Number(input.minimum_amount ?? 0),
        reason: input.reason ?? "Standard pre-arrival deposit",
        priority: Number(input.priority ?? 100),
        active: input.active ?? true,
      };
      const q = input.id
        ? db.from("deposit_policies").update(payload).eq("id", input.id)
        : db.from("deposit_policies").insert(payload);
      const { error } = await q;
      if (error) throw new Error(error.message);
      return true;
    },
    onSuccess: () => {
      toast.success("Deposit policy saved");
      void qc.invalidateQueries({ queryKey: ["pms"] });
    },
    onError: (e: Error) => toast.error(e.message),
  });
}

/**
 * Re-runs the Finance recognition for a deposit. The routine is repeat-safe:
 * a deposit already carrying a journal returns that journal rather than a
 * second entry.
 */
export function usePostDeposit() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async (depositId: string) => {
      const { data, error } = await guardedRpc(db, "neo_deposit_post", { _deposit: depositId });
      if (error) throw new Error(error.message);
      const result = (data ?? {}) as { ok?: boolean; reason?: string; detail?: string };
      if (result.ok === false) {
        throw new Error(result.detail ?? result.reason ?? "The deposit could not be posted.");
      }
      return result;
    },
    onSuccess: (r) => {
      toast.success(
        r.reason === "already_posted"
          ? "This deposit is already recognised in Finance"
          : "Deposit posted to Finance",
      );
      void qc.invalidateQueries({ queryKey: ["finance"] });
    },
    onError: (e: Error) => toast.error(e.message),
  });
}

/**
 * Pre-arrival deposits.
 *
 * A deposit is a liability held against a reservation, not revenue, and it is
 * the gate the arrivals desk must pass before a stay may open. Nothing here
 * computes the position: `neo_reservation_deposit_position` is the single
 * source of truth, and the same rule is enforced again by the database trigger
 * on stays, so a refresh, a reopen or a direct call can never bypass it.
 */
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { toast } from "sonner";
import { supabase } from "@/integrations/supabase/client";
import { useAuth } from "@/lib/auth";
import { stableIdempotencyKey } from "@/lib/transaction-scope";


/* eslint-disable @typescript-eslint/no-explicit-any */
const rpc = (fn: string, args: Record<string, unknown>) =>
  (supabase as any).rpc(fn, args) as Promise<{ data: any; error: { message: string } | null }>;

export type DepositPosition = {
  ok: boolean;
  reason?: string;
  reservation_id?: string;
  currency?: string;
  required?: number;
  held?: number;
  applied?: number;
  paid?: number;
  outstanding?: number;
  waiver?: string | null;
  override_id?: string | null;
  may_check_in?: boolean;
  required_source?: "manual" | "policy" | "none" | string | null;
  policy_id?: string | null;
  policy_name?: string | null;
  policy_reason?: string | null;
  policy_amount?: number | null;
};

/** Why check-in is or is not permitted, in the words the desk should read. */
export function depositGate(position: DepositPosition | undefined | null): {
  blocked: boolean;
  message: string;
} {
  if (!position || !position.ok) {
    return { blocked: false, message: "Deposit position unavailable." };
  }
  if (position.waiver === "corporate-credit") {
    return { blocked: false, message: "Charged to a corporate credit account â€” no deposit due." };
  }
  if (position.override_id) {
    return {
      blocked: false,
      message: "An authorised override is on file for this arrival. It is used once.",
    };
  }
  const outstanding = Number(position.outstanding ?? 0);
  if (outstanding > 0.009) {
    return {
      blocked: true,
      message: `Deposit outstanding: ${outstanding.toFixed(2)} of ${Number(position.required ?? 0).toFixed(2)} required. Record the deposit or obtain an authorised override.`,
    };
  }
  return { blocked: false, message: "Deposit settled." };
}

export function useDepositPosition(reservationId: string | null | undefined, enabled = true) {
  const { organizationId } = useAuth();
  return useQuery({
    queryKey: ["pms", "deposit-position", organizationId, reservationId],
    enabled: enabled && Boolean(organizationId) && Boolean(reservationId),
    queryFn: async (): Promise<DepositPosition> => {
      const { data, error } = await rpc("neo_reservation_deposit_position", {
        _reservation: reservationId,
      });
      if (error) throw new Error(error.message);
      return (data ?? { ok: false }) as DepositPosition;
    },
  });
}

function useInvalidateDeposits() {
  const qc = useQueryClient();
  return () => {
    void qc.invalidateQueries({ queryKey: ["pms"] });
    void qc.invalidateQueries({ queryKey: ["finance"] });
  };
}

/**
 * Records a received deposit. The idempotency key is minted per submission, so
 * a double-click, a retry or a refreshed request returns the first deposit
 * instead of taking the guest's money twice.
 */
export function useRecordDeposit() {
  const invalidate = useInvalidateDeposits();
  return useMutation({
    mutationFn: async (input: {
      reservationId: string;
      amount: number;
      method: string;
      reference?: string;
      notes?: string;
      idempotencyKey: string;
    }) => {
      const { data, error } = await rpc("neo_reservation_deposit_record", {
        _reservation: input.reservationId,
        _amount: input.amount,
        _method: input.method,
        _reference: input.reference ?? null,
        _idempotency_key: input.idempotencyKey,
        _notes: input.notes ?? null,
      });
      if (error) throw new Error(error.message);
       const result = (data ?? {}) as {
         deposit_id?: string;
         idempotent?: boolean;
         position?: DepositPosition;
         finance?: { ok?: boolean; reason?: string; detail?: string };
       };
       if (!result.deposit_id) throw new Error("The deposit was not persisted.");
       return result;
    },
    onSuccess: (r) => {
      if (r.finance?.ok) {
        toast.success(r.idempotent ? "Deposit already recorded and posted" : "Deposit recorded and posted to Finance");
      } else {
        toast.warning(`Deposit recorded â€” Finance ${r.finance?.reason ?? "posting is pending"}`);
      }
      invalidate();
    },
    onError: (e: Error) => toast.error(e.message),
  });
}

/** Authorised, single-use permission to check in with a deposit still outstanding. */
export function useDepositOverride() {
  const invalidate = useInvalidateDeposits();
  return useMutation({
    mutationFn: async (input: { reservationId: string; reason: string; authority: string }) => {
      const { data, error } = await rpc("neo_checkin_deposit_override", {
        _reservation: input.reservationId,
        _reason: input.reason,
        _authority: input.authority,
      });
      if (error) throw new Error(error.message);
      return (data ?? {}) as Record<string, unknown>;
    },
    onSuccess: () => {
      toast.success("Override authorised for this arrival");
      invalidate();
    },
    onError: (e: Error) => toast.error(e.message),
  });
}

/**
 * One posting intent per source event and property.
 *
 * A key minted at random per submission made every retry â€” and every second
 * operator confirming the same money â€” a *new* deposit, which is how one
 * reservation came to sit twice in the Finance posting queue. The key is now
 * derived from the facts of the intent itself (property, reservation, amount,
 * method, reference, business date), so repeating the same intent returns the
 * existing deposit while a genuinely different one still records.
 */
export const depositIdempotencyKey = (input: {
  reservationId: string;
  propertyId: string | null;
  amount: number;
  method: string;
  reference?: string;
  businessDate?: string;
}) =>
  stableIdempotencyKey("deposit", "record", [
    input.propertyId,
    input.reservationId,
    input.amount.toFixed(2),
    input.method,
    input.reference ?? "",
    input.businessDate ?? new Date().toISOString().slice(0, 10),
  ]);

/**
 * Pre-arrival deposit container.
 *
 * Shows the liability held against a reservation â€” required, received,
 * outstanding â€” and lets an authorised user record a deposit or request a

  PAYMENT_METHOD_CATALOGUE.filter((m) => m.value === v),

  const { activeBranchId } = useBranchScope();
  const position = useDepositPosition(reservationId);
  const record = useRecordDeposit();
  const override = useDepositOverride();

  const [method, setMethod] = useState("cash");
  const [reference, setReference] = useState("");
  const [reason, setReason] = useState("");

  const p = position.data;
  const gate = depositGate(p);
  const mayRecord = can("pms", "reservations.edit") || can("finance", "collections.collect_payment");
  const mayOverride = can("front-office", "folios.approve");

  return (
    <SectionCard
      title="Deposit"
      description="Held against this reservation before arrival. A deposit is a liability, never revenue."
    >
      {position.isLoading ? (
        <p className="text-sm text-muted-foreground">Reading the deposit positionâ€¦</p>
      ) : position.error ? (
        <p className="text-sm text-destructive">{(position.error as Error).message}</p>

          <dl className="grid gap-3 text-sm sm:grid-cols-4">

          </dl>

          {p.policy_name ? (
            <p className="text-xs text-muted-foreground">
              Required by policy <span className="font-medium">{p.policy_name}</span> â€”{" "}
              {p.policy_reason ?? "property deposit rule"} ({money(p.policy_amount ?? 0)}).
            </p>
          ) : p.required_source === "manual" && (p.required ?? 0) > 0 ? (
            <p className="text-xs text-muted-foreground">
              Deposit entered by hand on this reservation; no property policy applied.
            </p>
          ) : null}

          <div
            className={`rounded-lg border p-3 text-sm ${
              gate.blocked
                ? "border-warning/40 bg-warning/10"
                : "border-border/70 bg-muted/30 text-muted-foreground"

          >
            {gate.blocked ? null : <ShieldCheck className="mr-1 inline size-4" />}
            {gate.message}
            {p.waiver ? (
              <Badge variant="outline" className="ml-2">
                {p.waiver}
              </Badge>
            ) : null}

          {mayRecord ? (

              <div className="grid gap-1.5">
                <Label htmlFor="dep-amount">Amount</Label>
                <Input
                  id="dep-amount"
                  type="number"
                  min={0}
                  value={amount}
                  onChange={(e) => setAmount(e.target.value)}

              <div className="grid gap-1.5">
                <PaymentMethodPicker methods={METHODS} value={method} onChange={setMethod} />
              </div>
              <div className="grid gap-1.5">
                <Label htmlFor="dep-ref">Reference</Label>

                  id="dep-ref"
                  value={reference}
                  onChange={(e) => setReference(e.target.value)}

                      amount: Number(amount),

                      idempotencyKey: depositIdempotencyKey({
                        reservationId,

              </p>
            </div>
          ) : null}
        </div>
      )}
    </SectionCard>
  );
}

function Figure({ label, value }: { label: string; value: string }) {

  const [customerName, setCustomerName] = useState("");
  const [newGuestName, setNewGuestName] = useState("");
  const [newGuestEmail, setNewGuestEmail] = useState("");
  const [newGuestPhone, setNewGuestPhone] = useState("");
  const [isVip, setIsVip] = useState(false);
  /** Email stays optional; it is only rejected when something was typed and it
   *  is not a valid address. */
  const emailInvalid =
    newGuestEmail.trim().length > 0 && !/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(newGuestEmail.trim());

  const [roomTypeId, setRoomTypeId] = useState<string>("");
  const [roomId, setRoomId] = useState<string>(handoff.roomId ?? "");
  const [ratePlanId, setRatePlanId] = useState<string>("");
  const [mealPlanId, setMealPlanId] = useState<string>("");
  const [overrideOn, setOverrideOn] = useState(false);
  const [overrideRate, setOverrideRate] = useState(0);
  const [overrideReason, setOverrideReason] = useState("");
  const [specialRequests, setSpecialRequests] = useState("");

  const [guarantee, setGuarantee] = useState<string>("none");
  const [depositRequired, setDepositRequired] = useState(0);
  const [status, setStatus] = useState("tentative");

  const nights = nightsBetween(arrival, departure);
  const roomTypes = useRoomTypes();
  const ratePlans = useRatePlans();
  const mealPlans = useMealPlans();
  const customers = useCustomers(customerSearch);
  const canOverrideRate = can("front-office", "rates", "manage");
  const quote = useRateQuote({
    ratePlanId: ratePlanId || null,
    arrival,
    departure,
    adults,
    children,
    enabled: step >= 2 && nights > 0,
  });
  const availability = useAvailability({
    branch: activeBranchId ?? profile?.branch_id ?? null,
    arrival,
    departure,
    roomType: roomTypeId || null,
    adults,
    children,
    enabled: step >= 2 && nights > 0,
  });
  // A room handed off from Availability resolves to its own room type so the
  // applicable rate plans are shown without weakening the availability check.
  const rooms = useRooms();
  const handoffRoomType = roomTypeForRoom(
    (rooms.data ?? []) as Array<{ id: string; room_type_id?: string | null }>,
    roomId,
  );
  useEffect(() => {
    if (roomTypeId || !handoffRoomType) return;
    setRoomTypeId(handoffRoomType);
  }, [handoffRoomType, roomTypeId]);

  /** Changing the room type clears every selection that no longer applies. */
  const changeRoomType = useCallback(
    (next: string) => {
      setRoomTypeId(next);
      setRatePlanId("");
      setMealPlanId("");
      setOverrideOn(false);
      setOverrideRate(0);
      setOverrideReason("");
      if (roomId && handoffRoomType !== next) setRoomId("");
    },
    [roomId, handoffRoomType],
  );




  const derivedRate = quote.data?.ok ? Number(quote.data.nightly_rate ?? 0) : 0;
  const nightlyRate = overrideOn && canOverrideRate ? overrideRate : derivedRate;
  const taxAmount = overrideOn && canOverrideRate ? 0 : Number(quote.data?.tax_amount ?? 0);
  const serviceCharge = overrideOn && canOverrideRate ? 0 : Number(quote.data?.service_charge ?? 0);
  const total = useMemo(
    () => Math.round((nightlyRate * nights + taxAmount + serviceCharge) * 100) / 100,
    [nightlyRate, nights, taxAmount, serviceCharge],
  );

  const create = useMutation({
    mutationFn: async () => {
      if (!organizationId) throw new Error("No organisation context");
      if (nights < 1) throw new Error("Departure must be after arrival");
      if (!ratePlanId) throw new Error("Select a rate plan so the nightly rate can be derived");
      if (nightlyRate <= 0)
        throw new Error("The selected rate plan produced no price for these dates");
      if (overrideOn && (!canOverrideRate || overrideReason.trim().length < 4))
        throw new Error("An authorised reason is required for a rate override");
      const branchId = activeBranchId ?? profile?.branch_id;
      if (!branchId)
        throw new Error(
          "Select a property in the property selector before creating a reservation.",
        );

      let guestId = customerId;
      if (!guestId) {
        if (!newGuestName.trim()) throw new Error("A guest name is required");
        // Resolve against existing guest profiles first so repeat guests are never duplicated.
        const { data: resolved, error: customerError } = await guardedRpc(supabase, 
          "neo_customer_resolve",
          {
            _org: organizationId,
            _branch: branchId,
            _full_name: newGuestName.trim(),
            ...(newGuestPhone.trim() ? { _phone: newGuestPhone.trim() } : {}),
            ...(newGuestEmail.trim() ? { _email: newGuestEmail.trim() } : {}),
          },
        );
        if (customerError) throw customerError;
        guestId = resolved as string;
        await supabase
          .from("customers")
          .update({ customer_type: "individual-guest", is_vip: isVip, source: bookingSource })
          .eq("id", guestId);
      }

      const { data, error } = await guardedRpc(supabase, "neo_reservation_create", {
        _payload: {
          branch_id: branchId,
          customer_id: guestId as string,
          booking_source: bookingSource,
          reservation_type: reservationType,
          arrival_date: arrival,
          departure_date: departure,
          nights,
          adults,
          children,
          infants,
          room_type_id: roomTypeId || null,
          room_id: roomId || null,
          rate_plan_id: ratePlanId || null,
          meal_plan_id: mealPlanId || null,
          nightly_rate: nightlyRate,
          tax_amount: taxAmount,
          total_amount: total,
          rate_source: overrideOn && canOverrideRate ? "override" : "rate-plan",
          rate_override_reason: overrideOn && canOverrideRate ? overrideReason.trim() : null,
          rate_quoted_amount: derivedRate,
          deposit_required: depositRequired,
          balance: total,
          guarantee_method: guarantee,
          special_requests: specialRequests,
          is_vip: isVip,
          status,
        },
        _idempotency_key: crypto.randomUUID(),
      });
      if (error) throw error;
      const result = data as { ok?: boolean; id?: string; confirmation_number?: string } | null;
      if (!result?.ok || !result.id || !result.confirmation_number) {
        throw new Error("The reservation was not persisted.");
      }
      return { id: result.id, confirmation_number: result.confirmation_number };
    },
    onSuccess: (data) => {
      toast.success(`Reservation ${data.confirmation_number} created`);
      // Refresh every surface that lists the new booking before we navigate.
      for (const key of ["reservations", "pms", "arrivals", "availability", "facts"]) {
        void queryClient.invalidateQueries({ queryKey: [key] });
      }
      void navigate({ to: "/pms/reservations/$id", params: { id: data.id } });
    },
    onError: (e: Error) => toast.error(e.message),
  });

  const canAdvance = [
    nights > 0,
    (Boolean(customerId) || newGuestName.trim().length > 1) && !emailInvalid,
    Boolean(roomTypeId) &&
      Boolean(ratePlanId) &&
      nightlyRate > 0 &&
      (!overrideOn || overrideReason.trim().length > 3),
    true,
    true,
  ][step];

  return (
    <div>
      <PageHeader
        eyebrow="Front Office"
        title="New reservation"
        description="Availability is checked against real rooms, existing stays, blocks and holds before the booking is written."
      />

      <ol className="mb-6 flex flex-wrap gap-2">
        {STEPS.map((s, i) => (
          <li key={s}>
            <button
              type="button"
              onClick={() => setStep(i)}
              className={`flex items-center gap-2 rounded-full border px-4 py-1.5 text-sm transition-colors ${
                i === step
                  ? "border-primary/40 bg-chip-selected font-semibold text-chip-selected-foreground shadow-[var(--elevation-1)]"
                  : i < step
                    ? "border-success/40 bg-success/10 text-success"
                    : "border-border text-muted-foreground"
              }`}
            >
              {i < step ? <Check className="size-3.5" /> : <span className="text-xs">{i + 1}</span>}
              {s}
            </button>
          </li>
        ))}
      </ol>

      <div className="grid gap-6 lg:grid-cols-[1fr_320px]">
        <SectionCard title={STEPS[step] ?? ""} description="">
          {step === 0 ? (
            <div className="grid gap-4 sm:grid-cols-2">
              <div className="space-y-2 sm:col-span-2">
                <Label>Stay dates</Label>
                <NeoDateRangePicker
                  from={arrival}
                  to={departure}
                  onChange={({ from, to }) => {
                    setArrival(from);
                    setDeparture(to);
                  }}
                  fromLabel="Arrival"
                  toLabel="Departure"
                  showNights
                />
              </div>
              <div className="space-y-2">
                <Label>Adults</Label>
                <Input
                  type="number"
                  min={1}
                  value={adults}
                  onChange={(e) => setAdults(Number(e.target.value))}
                />
              </div>
              <div className="space-y-2">
                <Label>Children</Label>
                <Input
                  type="number"
                  min={0}
                  value={children}
                  onChange={(e) => setChildren(Number(e.target.value))}
                />
              </div>
              <div className="space-y-2">
                <Label>Infants</Label>
                <Input
                  type="number"
                  min={0}
                  value={infants}
                  onChange={(e) => setInfants(Number(e.target.value))}
                />
              </div>
              <div className="space-y-2">
                <Label>Reservation type</Label>
                <Select value={reservationType} onValueChange={setReservationType}>
                  <SelectTrigger>
                    <SelectValue />
                  </SelectTrigger>
                  <SelectContent>
                    {RESERVATION_TYPES.map((t) => (
                      <SelectItem key={t} value={t}>
                        {titleCase(t)}
                      </SelectItem>
                    ))}
                  </SelectContent>
                </Select>
              </div>
              <div className="space-y-2">
                <Label>Booking source</Label>
                <Select value={bookingSource} onValueChange={setBookingSource}>
                  <SelectTrigger>
                    <SelectValue />
                  </SelectTrigger>
                  <SelectContent>
                    {BOOKING_SOURCES.map((s) => (
                      <SelectItem key={s} value={s}>
                        {titleCase(s)}
                      </SelectItem>
                    ))}
                  </SelectContent>
                </Select>
              </div>
              {nights < 1 ? (
                <p className="sm:col-span-2 text-sm text-destructive">
                  Departure must be at least one night after arrival.
                </p>
              ) : null}
            </div>
          ) : null}

          {step === 1 ? (
            <div className="space-y-5">
              <div className="space-y-2">
                <Label htmlFor="guest-search">Search existing guests</Label>
                <Input
                  id="guest-search"
                  name="guest-search"
                  value={customerSearch}
                  onChange={(e) => setCustomerSearch(e.target.value)}
                  placeholder="Name, email, phone or customer number"
                />
                {customerSearch.trim().length >= 2 ? (
                  <div className="max-h-52 overflow-y-auto rounded-lg border border-border/70">
                    {(customers.data ?? []).length === 0 ? (
                      <p className="p-3 text-sm text-muted-foreground">
                        No matching guest â€” capture the details below to create a new profile.
                      </p>
                    ) : (
                      (customers.data ?? []).map((c) => (
                        <button
                          key={c.id}
                          type="button"
                          onClick={() => {
                            setCustomerId(c.id);
                            setCustomerName(c.full_name);
                            setIsVip(Boolean(c.is_vip));
                          }}
                          className={`flex w-full items-center justify-between px-3 py-2 text-left text-sm hover:bg-muted ${
                            customerId === c.id ? "bg-primary-soft" : ""
                          }`}
                        >
                          <span>
                            <span className="font-medium">{c.full_name}</span>
                            <span className="ml-2 text-xs text-muted-foreground">
                              {c.email || c.phone || c.customer_number}
                            </span>
                          </span>
                          {customerId === c.id ? <Check className="size-4 text-primary" /> : null}
                        </button>
                      ))
                    )}
                  </div>
                ) : null}
              </div>

              {customerId ? (
                <div className="flex items-center justify-between rounded-lg border border-border/70 p-3">
                  <p className="text-sm">
                    Booking for <span className="font-semibold">{customerName}</span>
                  </p>
                  <Button variant="ghost" size="sm" onClick={() => setCustomerId(null)}>
                    Clear
                  </Button>
                </div>
              ) : (
                <div className="grid gap-4 sm:grid-cols-3">
                  <div className="space-y-2 sm:col-span-3">
                    <Label htmlFor="new-guest-name">New guest full name</Label>
                    <Input
                      id="new-guest-name"
                      name="new-guest-name"
                      value={newGuestName}
                      onChange={(e) => setNewGuestName(e.target.value)}
                    />
                  </div>
                  <div className="space-y-2">
                    <Label htmlFor="new-guest-email">Email (optional)</Label>
                    {/* A plain text field with an explicit pattern: the browser
                        email widget and password managers must never rewrite or
                        clear what the agent typed. Validation is ours. */}
                    <Input
                      id="new-guest-email"
                      name="new-guest-email"
                      type="text"
                      inputMode="email"
                      autoComplete="off"
                      spellCheck={false}
                      value={newGuestEmail}
                      onChange={(e) => setNewGuestEmail(e.target.value)}
                      aria-invalid={emailInvalid || undefined}
                      aria-describedby="new-guest-email-hint"
                    />
                    <p
                      id="new-guest-email-hint"
                      className={`text-xs ${emailInvalid ? "text-destructive" : "text-muted-foreground"}`}
                    >
                      {emailInvalid
                        ? "Enter a valid email address, or leave it blank."
                        : "Optional â€” used for confirmations and the folio statement."}
                    </p>
                  </div>
                  <div className="space-y-2">
                    <Label htmlFor="new-guest-phone">Phone</Label>
                    <Input
                      id="new-guest-phone"
                      name="new-guest-phone"
                      value={newGuestPhone}
                      onChange={(e) => setNewGuestPhone(e.target.value)}
                    />
                  </div>
                </div>
              )}

              <div className="flex items-center justify-between rounded-lg border border-border/70 p-3">
                <div>
                  <p className="text-sm font-medium">VIP handling</p>
                  <p className="text-xs text-muted-foreground">
                    Flags the arrival for management attention.
                  </p>
                </div>
                <Switch checked={isVip} onCheckedChange={setIsVip} />
              </div>
            </div>
          ) : null}

          {step === 2 ? (
            <div className="space-y-5">
              <div className="grid gap-4 sm:grid-cols-2">
                <div className="space-y-2">
                  <Label>Room type</Label>
                  <Select value={roomTypeId} onValueChange={changeRoomType}>
                    <SelectTrigger>
                      <SelectValue
                        placeholder={
                          (roomTypes.data ?? []).length
                            ? "Select a room type"
                            : "No room types configured"
                        }
                      />
                    </SelectTrigger>
                    <SelectContent>
                      {(roomTypes.data ?? []).map((rt) => (
                        <SelectItem key={rt.id} value={rt.id}>
                          {rt.name} Â· sleeps {rt.max_occupancy}
                        </SelectItem>
                      ))}
                    </SelectContent>
                  </Select>
                </div>
                <div className="space-y-2">
                  <Label id="rate-plan-label">Rate plan</Label>
                  {ratePlans.isLoading ? (
                    <LoadingState variant="rows" rows={2} />
                  ) : ratePlans.error ? (
                    <ErrorState
                      inline
                      error={ratePlans.error}
                      title="Rate plans could not be loaded"
                      onRetry={() => void ratePlans.refetch()}
                    />
                  ) : (
                    <RatePlanPicker
                      plans={(ratePlans.data ?? []) as never}
                      roomTypeId={roomTypeId || null}
                      value={ratePlanId}
                      onSelect={(planId) => {
                        setRatePlanId(planId);
                        setOverrideOn(false);
                        const plan = selectableRatePlans(
                          (ratePlans.data ?? []) as never,
                          roomTypeId || null,
                        ).find((p) => p.id === planId);
                        setMealPlanId(plan?.meal_plan_id ?? "");
                      }}
                    />
                  )}
                </div>



                <div className="space-y-2">
                  <Label>Meal plan</Label>
                  <Select value={mealPlanId} onValueChange={setMealPlanId}>
                    <SelectTrigger>
                      <SelectValue
                        placeholder={
                          (mealPlans.data ?? []).length
                            ? "Select a meal plan"
                            : "No meal plans configured"
                        }
                      />
                    </SelectTrigger>
                    <SelectContent>
                      {(mealPlans.data ?? []).map((mp) => (
                        <SelectItem key={mp.id} value={mp.id}>
                          {mp.code} â€” {mp.name}
                        </SelectItem>
                      ))}
                    </SelectContent>
                  </Select>
                </div>
                <div className="space-y-2">
                  <Label>Nightly rate</Label>
                  {!ratePlanId ? (
                    <p className="text-sm text-muted-foreground">
                      Select a rate plan â€” the nightly rate is derived, never typed.
                    </p>
                  ) : quote.isLoading ? (
                    <LoadingState variant="rows" rows={1} />
                  ) : quote.error ? (
                    <ErrorState
                      inline
                      error={quote.error}
                      title="Rate could not be calculated"
                      onRetry={() => void quote.refetch()}
                    />
                  ) : quote.data && !quote.data.ok ? (
                    <p className="text-sm text-destructive">{quote.data.reason}</p>
                  ) : (
                    <div className="rounded-lg border border-border p-3 text-sm">
                      <div className="flex justify-between font-medium">
                        <span>{money(derivedRate, quote.data?.currency)}</span>
                        <span className="text-muted-foreground">per night</span>
                      </div>
                      <p className="mt-1 text-xs text-muted-foreground">
                        {nights} night(s) Â· service {money(serviceCharge)} Â· tax {money(taxAmount)}{" "}
                        Â· total {money(total)}
                      </p>
                      {(quote.data?.restrictions ?? []).map((r) => (
                        <p key={r} className="mt-1 text-xs text-on-tint-warning">
                          {r}
                        </p>
                      ))}
                    </div>
                  )}
                  {canOverrideRate && ratePlanId ? (
                    <div className="space-y-2 rounded-lg border border-dashed border-border p-3">
                      <div className="flex items-center justify-between">
                        <Label className="text-xs">Authorised rate override</Label>
                        <Switch
                          checked={overrideOn}
                          onCheckedChange={(v) => {
                            setOverrideOn(v);
                            if (v) setOverrideRate(derivedRate);
                          }}
                        />
                      </div>
                      {overrideOn ? (
                        <>
                          <Input
                            type="number"
                            min={0}
                            value={overrideRate}
                            onChange={(e) => setOverrideRate(Number(e.target.value))}
                          />
                          <Input
                            placeholder="Reason for the override (required, audited)"
                            value={overrideReason}
                            onChange={(e) => setOverrideReason(e.target.value)}
                          />
                          <p className="text-xs text-muted-foreground">
                            Plan price {money(derivedRate)} will be preserved on the reservation
                            record.
                          </p>
                        </>
                      ) : null}
                    </div>
                  ) : null}
                </div>
              </div>

              <div className="space-y-2">
                <Label>Assign a specific room (optional)</Label>
                {availability.isLoading ? (
                  <LoadingState variant="rows" rows={1} />
                ) : (availability.data ?? []).length === 0 ? (
                  <EmptyState
                    inline
                    title="No room is free for these dates"
                    description="The booking can still be taken and assigned later, or added to the waitlist."
                  />
                ) : (
                  <Select value={roomId} onValueChange={setRoomId}>
                    <SelectTrigger>
                      <SelectValue
                        placeholder={`${(availability.data ?? []).length} room(s) available`}
                      />
                    </SelectTrigger>
                    <SelectContent>
                      {(availability.data ?? []).map((r) => (
                        <SelectItem key={r.room_id} value={r.room_id}>
                          {r.room_number} Â· {r.room_type_name}
                        </SelectItem>
                      ))}
                    </SelectContent>
                  </Select>
                )}
              </div>

              <div className="space-y-2">
                <Label>Special requests</Label>
                <Textarea
                  value={specialRequests}
                  onChange={(e) => setSpecialRequests(e.target.value)}
                  rows={3}
                />
              </div>
            </div>
          ) : null}

          {step === 3 ? (
            <div className="grid gap-4 sm:grid-cols-2">
              <div className="space-y-2">
                <Label>Guarantee method</Label>
                <Select value={guarantee} onValueChange={setGuarantee}>
                  <SelectTrigger>
                    <SelectValue />
                  </SelectTrigger>
                  <SelectContent>
                    {GUARANTEE_METHODS.map((g) => (
                      <SelectItem key={g} value={g}>
                        {titleCase(g)}
                      </SelectItem>
                    ))}
                  </SelectContent>
                </Select>
              </div>
              <div className="space-y-2">
                <Label>Deposit required</Label>
                <Input
                  type="number"
                  min={0}
                  value={depositRequired}
                  onChange={(e) => setDepositRequired(Number(e.target.value))}
                />
              </div>
              <div className="space-y-2 sm:col-span-2">
                <Label>Reservation status</Label>
                <Select value={status} onValueChange={setStatus}>
                  <SelectTrigger>
                    <SelectValue />
                  </SelectTrigger>
                  <SelectContent>
                    {["enquiry", "tentative", "pending-deposit", "confirmed", "guaranteed"].map(
                      (s) => (
                        <SelectItem key={s} value={s}>
                          {titleCase(s)}
                        </SelectItem>
                      ),
                    )}
                  </SelectContent>
                </Select>
                <p className="text-xs text-muted-foreground">
                  Deposits are recorded against the folio after check-in, or as a pre-arrival
                  payment from the folio screen.
                </p>
              </div>
            </div>
          ) : null}

          {step === 4 ? (
            <dl className="grid gap-3 text-sm sm:grid-cols-2">
              <Detail label="Guest" value={customerId ? customerName : newGuestName || "â€”"} />
              <Detail label="Stay" value={`${arrival} â†’ ${departure} (${nights} night(s))`} />
              <Detail
                label="Party"
                value={`${adults} adult(s), ${children} child(ren), ${infants} infant(s)`}
              />
              <Detail
                label="Room type"
                value={(roomTypes.data ?? []).find((r) => r.id === roomTypeId)?.name ?? "â€”"}
              />
              <Detail
                label="Room"
                value={
                  (availability.data ?? []).find((r) => r.room_id === roomId)?.room_number ??
                  "Assign later"
                }
              />
              <Detail label="Rate" value={`${money(nightlyRate)} per night`} />
              <Detail label="Total" value={money(total)} />
              <Detail label="Deposit" value={money(depositRequired)} />
              <Detail label="Status" value={titleCase(status)} />
              <Detail label="Source" value={titleCase(bookingSource)} />
            </dl>
          ) : null}

          <div className="mt-6 flex items-center justify-between">
            <Button variant="ghost" disabled={step === 0} onClick={() => setStep((s) => s - 1)}>
              <ChevronLeft className="mr-1 size-4" /> Back
            </Button>
            {step < STEPS.length - 1 ? (
              <Button disabled={!canAdvance} onClick={() => setStep((s) => s + 1)}>
                Continue <ChevronRight className="ml-1 size-4" />
              </Button>
            ) : (
              <Button
                disabled={
                  create.isPending ||
                  !ratePlanId ||
                  nightlyRate <= 0 ||
                  (overrideOn && overrideReason.trim().length < 4)
                }
                onClick={() => create.mutate()}
              >
                {create.isPending ? "Creatingâ€¦" : "Create reservation"}
              </Button>
            )}
          </div>
        </SectionCard>

        <SectionCard title="Booking summary" description="Updates as you complete each step.">
          <dl className="space-y-3 text-sm">
            <Detail label="Nights" value={String(nights)} />
            <Detail label="Nightly rate" value={money(nightlyRate)} />
            <Detail label="Accommodation total" value={money(total)} />
            <Detail label="Deposit required" value={money(depositRequired)} />
            <Detail
              label="Balance on arrival"
              value={money(Math.max(total - depositRequired, 0))}
            />
          </dl>
          {isVip ? <Badge className="mt-4">VIP arrival</Badge> : null}
        </SectionCard>
      </div>
    </div>
  );
}

function Detail({ label, value }: { label: string; value: string }) {
  return (
    <div className="flex items-center justify-between gap-4 border-b border-border/50 pb-2 last:border-0">
      <dt className="text-muted-foreground">{label}</dt>
      <dd className="text-right font-medium text-foreground">{value}</dd>
    </div>
  );

CREATE OR REPLACE FUNCTION public.pms_check_in(_reservation uuid, _keys integer DEFAULT 1, _notes text DEFAULT ''::text)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE _org uuid := public.current_org_id(); _res public.reservations; _stay uuid; _folio uuid;
  _room public.rooms; _key text; _folio_no text; _tax_rate numeric; _svc_rate numeric;
  _treatment text; _tax numeric; _svc numeric; _line uuid;
BEGIN
  IF NOT public.has_permission('pms','checkin.perform') THEN RAISE EXCEPTION 'Not permitted to check in guests'; END IF;
  SELECT * INTO _res FROM public.reservations WHERE id=_reservation AND organization_id=_org FOR UPDATE;
  IF _res.id IS NULL THEN RAISE EXCEPTION 'Reservation not found'; END IF;
  _key := 'checkin:' || _reservation::text;

  SELECT id INTO _stay FROM public.stays WHERE reservation_id=_reservation;
  IF _stay IS NOT NULL THEN
    SELECT id INTO _folio FROM public.folios WHERE stay_id=_stay AND folio_type='guest';
    RETURN jsonb_build_object('stay_id',_stay,'folio_id',_folio,'idempotent',true);
  END IF;

  IF _res.status NOT IN ('confirmed','guaranteed','tentative','pending-deposit') THEN
    RAISE EXCEPTION 'Reservation status % is not eligible for check-in', _res.status;
  END IF;
  IF _res.room_id IS NULL THEN RAISE EXCEPTION 'Assign a room before check-in'; END IF;
  SELECT * INTO _room FROM public.rooms WHERE id=_res.room_id FOR UPDATE;
  IF _room.room_condition IN ('out-of-order','out-of-service','maintenance','blocked') THEN
    RAISE EXCEPTION 'Room % is not available (%).', _room.room_number, _room.room_condition;
  END IF;
  IF _room.occupancy_status = 'occupied' THEN
    RAISE EXCEPTION 'Room % is already occupied', _room.room_number;
  END IF;

  INSERT INTO public.stays (organization_id, branch_id, reservation_id, customer_id, room_id,
    checked_in_by, expected_departure)
  VALUES (_org, _res.branch_id, _reservation, _res.customer_id, _res.room_id, auth.uid(), _res.departure_date)
  RETURNING id INTO _stay;

  _folio_no := public.next_document_number('invoice');
  INSERT INTO public.folios (organization_id, branch_id, folio_number, reservation_id, stay_id,
    customer_id, corporate_account_id, group_booking_id, created_by)
  VALUES (_org, _res.branch_id, _folio_no, _reservation, _stay, _res.customer_id,
          _res.corporate_account_id, _res.group_booking_id, auth.uid())
  RETURNING id INTO _folio;

  UPDATE public.reservations SET status='checked-in', updated_at=now() WHERE id=_reservation;
  UPDATE public.rooms SET occupancy_status='occupied', room_condition='occupied-clean',
    housekeeping_status='occupied', updated_at=now() WHERE id=_res.room_id;
  UPDATE public.room_assignments SET stay_id=_stay, updated_at=now()
    WHERE reservation_id=_reservation AND status='active';

  -- The arrival night is charged immediately so the folio carries a real
  -- balance from check-in; the nightly posting run skips it via the
  -- one-night-per-reservation unique index.
  --
  -- Tax and service follow exactly the same rules as public.pms_quote_rate:
  -- an inclusive rate plan adds nothing on top, an exclusive plan adds the
  -- service charge first and taxes the sum. Without this the folio line
  -- disagreed with the price the guest was quoted and confirmed.
  SELECT COALESCE(rp.tax_rate,0), COALESCE(rp.service_charge_rate,0),
         COALESCE(rp.tax_treatment,'exclusive')
    INTO _tax_rate, _svc_rate, _treatment
  FROM public.rate_plans rp WHERE rp.id = _res.rate_plan_id;
  _tax_rate := COALESCE(_tax_rate, 0);
  _svc_rate := COALESCE(_svc_rate, 0);
  _treatment := COALESCE(_treatment, 'exclusive');

  IF COALESCE(_res.nightly_rate,0) > 0 AND _res.arrival_date < _res.departure_date THEN
    IF _treatment = 'exclusive' THEN
      _svc := round(_res.nightly_rate * _svc_rate / 100.0, 2);
      _tax := round((_res.nightly_rate + _svc) * _tax_rate / 100.0, 2);
    ELSE
      _svc := 0;
      _tax := 0;
    END IF;
    BEGIN
      INSERT INTO public.folio_lines (organization_id, branch_id, folio_id, reservation_id, line_type,
        category, source_module, description, operational_date, quantity, unit_price,
        gross_amount, tax_amount, service_charge_amount, net_amount, posted_by)
      VALUES (_org, _res.branch_id, _folio, _reservation, 'charge', 'accommodation', 'pms',
        'Accommodation â€” room ' || _room.room_number || ' Â· ' || _res.arrival_date::text,
        _res.arrival_date, 1, _res.nightly_rate, _res.nightly_rate, _tax, _svc,
        _res.nightly_rate + _tax + _svc, auth.uid())
      RETURNING id INTO _line;
      UPDATE public.folios SET total_charges = total_charges + _res.nightly_rate,
        total_taxes = total_taxes + _tax,
        balance = balance + _res.nightly_rate + _tax + _svc, updated_at=now()
      WHERE id=_folio;
      PERFORM public.pkg_allocate_folio_line(_line, 'check-in');
    EXCEPTION WHEN unique_violation THEN
      NULL;
    END;
  END IF;

  INSERT INTO public.room_status_history (organization_id, branch_id, room_id, field, previous_value, new_value, reason, changed_by)
  VALUES (_org, _res.branch_id, _res.room_id, 'occupancy_status', _room.occupancy_status, 'occupied', 'Guest checked in', auth.uid());

  INSERT INTO public.notifications (organization_id, branch_id, category, title, body, link)
  VALUES (_org, _res.branch_id, 'operations', 'Guest checked in',
    'Room ' || _room.room_number || ' â€” folio ' || _folio_no || ' is now active.', '/pms/in-house');

  INSERT INTO public.customer_timeline_events (organization_id, branch_id, customer_id, event_type,
    description, source_module, source_record_id, staff_user_id)
  SELECT _org, _res.branch_id, _res.customer_id, 'stay',
    'Checked in Â· room ' || _room.room_number || ' Â· ' || _res.arrival_date::text || ' â†’ ' || _res.departure_date::text,
    'front-office', _reservation::text, auth.uid()
  WHERE _res.customer_id IS NOT NULL;

  INSERT INTO public.pms_workflow_events (organization_id, branch_id, event_type, reservation_id, stay_id, folio_id, idempotency_key, created_by)
  VALUES (_org, _res.branch_id, 'guest.checked-in', _reservation, _stay, _folio, _key, auth.uid())
  ON CONFLICT DO NOTHING;

  INSERT INTO public.audit_logs (organization_id, user_id, user_name, action, module, record_id, new_value, reason)
  VALUES (_org, auth.uid(), '', 'pms.checkin', 'front-office', _reservation::text,
    jsonb_build_object('stay_id',_stay,'folio_id',_folio,'room',_room.room_number,'keys',_keys), _notes);

  RETURN jsonb_build_object('stay_id',_stay,'folio_id',_folio,'folio_number',_folio_no,'idempotent',false);
END; $function$;
