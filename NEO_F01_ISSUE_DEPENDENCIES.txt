  335: CREATE OR REPLACE FUNCTION public.inventory_apply_movement(_item uuid, _location uuid, _quantity numeric, _unit text, _movement_type text, _unit_cost numeric DEFAULT NULL::numeric, _reference_type text DEFAULT NULL::text, _reference_id uuid DEFAULT NULL::uuid, _reason text DEFAULT NULL::text, _idempotency_key text DEFAULT NULL::text, _batch_number text DEFAULT NULL::text, _expiry date DEFAULT NULL::date)
  336: RETURNS uuid
  337: LANGUAGE plpgsql SECURITY DEFINER SET search_path TO 'public'
  338: AS $function$
  339: DECLARE
  340:   v_org uuid := current_org_id();
  341:   v_item record;
  342:   v_qty_base numeric;
  343:   v_cost numeric;
  344:   v_level record;
  345:   v_batch uuid;
  346:   v_movement uuid;
  347:   v_existing uuid;
  348:   v_available numeric;
  349:   v_store text;
  350: BEGIN
  351:   IF v_org IS NULL THEN RAISE EXCEPTION 'No active organization'; END IF;
  352:   IF NOT public.neo_system_posting()
  353:      AND NOT (has_permission('inventory','stock.manage') OR has_permission('procurement','receiving.manage')) THEN
  354:     RAISE EXCEPTION 'Not permitted to move stock';
  355:   END IF;
  356: 
  357:   IF _idempotency_key IS NOT NULL THEN
  358:     SELECT id INTO v_existing FROM stock_movements WHERE organization_id = v_org AND idempotency_key = _idempotency_key;
  359:     IF v_existing IS NOT NULL THEN RETURN v_existing; END IF;
  360:   END IF;
  361: 
  362:   SELECT * INTO v_item FROM stock_items WHERE id = _item AND organization_id = v_org;
  363:   IF v_item IS NULL THEN RAISE EXCEPTION 'Unknown stock item'; END IF;
  364: 
  365:   v_qty_base := inventory_convert(_quantity, COALESCE(_unit, v_item.base_unit), v_item.base_unit);
  366:   v_cost := COALESCE(_unit_cost, v_item.average_cost);
  367: 
  368:   INSERT INTO stock_levels (organization_id, stock_item_id, location_id, quantity_on_hand, last_movement_at)
  369:   VALUES (v_org, _item, _location, 0, now())
  370:   ON CONFLICT (stock_item_id, location_id) DO NOTHING;
  371: 
  372:   SELECT * INTO v_level FROM stock_levels WHERE stock_item_id = _item AND location_id = _location FOR UPDATE;
  373: 
  374:   -- Taking out more than is there creates impossible inventory. A count,
  375:   -- adjustment or opening balance must still be able to state reality.
  376:   IF v_qty_base < 0
  377:      AND COALESCE(v_level.quantity_on_hand, 0) + v_qty_base < 0
  378:      AND lower(COALESCE(_movement_type,'')) NOT IN
  379:          ('count','stock_take','stock-take','adjustment','opening','opening_balance','variance','recount','correction')
  380:      AND NOT public.neo_inventory_allows_negative(v_org, v_item.branch_id) THEN
  381:     SELECT sl.name INTO v_store FROM stock_locations sl WHERE sl.id = _location;
  382:     v_available := round(COALESCE(v_level.quantity_on_hand, 0), 4);
  383:     RAISE EXCEPTION 'Not enough %: % % available in %, % % requested',
  384:       v_item.name,
  385:       trim(to_char(GREATEST(v_available,0), 'FM999999990.###')), v_item.base_unit,
  386:       COALESCE(v_store, 'this store'),
  387:       trim(to_char(abs(v_qty_base), 'FM999999990.###')), v_item.base_unit
  388:       USING ERRCODE = 'check_violation';
  389:   END IF;
  390: 
  391:   IF v_qty_base > 0 AND (v_item.track_batches OR _batch_number IS NOT NULL OR _expiry IS NOT NULL) THEN
  392:     INSERT INTO stock_batches (organization_id, stock_item_id, location_id, batch_number, expiry_date, quantity, unit_cost)
  393:     VALUES (v_org, _item, _location, _batch_number, _expiry, v_qty_base, v_cost)
  394:     RETURNING id INTO v_batch;
  395:   ELSIF v_qty_base < 0 THEN
  396:     SELECT id INTO v_batch FROM stock_batches
  397:       WHERE stock_item_id = _item AND location_id = _location AND quantity > 0 AND status = 'active'
  398:       ORDER BY expiry_date NULLS LAST, received_at LIMIT 1;
  399:     IF v_batch IS NOT NULL THEN
  400:       UPDATE stock_batches SET quantity = GREATEST(0, quantity + v_qty_base) WHERE id = v_batch;
  401:     END IF;
  402:   END IF;
  403: 
  404:   UPDATE stock_levels
  405:      SET quantity_on_hand = quantity_on_hand + v_qty_base,
  406:          quantity_damaged = quantity_damaged + CASE WHEN _movement_type = 'damage' THEN abs(v_qty_base) ELSE 0 END,
  407:          quantity_expired = quantity_expired + CASE WHEN _movement_type = 'expiry' THEN abs(v_qty_base) ELSE 0 END,
  408:          last_movement_at = now()
  409:    WHERE id = v_level.id;
  410: 
  411:   IF v_qty_base > 0 AND _unit_cost IS NOT NULL THEN
  412:     UPDATE stock_items
  413:        SET last_cost = v_cost,
  414:            average_cost = CASE
  415:              WHEN COALESCE(v_level.quantity_on_hand, 0) + v_qty_base <= 0 THEN v_cost
  416:              ELSE ((GREATEST(v_level.quantity_on_hand, 0) * average_cost) + (v_qty_base * v_cost))
  417:                   / (GREATEST(v_level.quantity_on_hand, 0) + v_qty_base)
  418:            END
  419:      WHERE id = _item;
  420:   END IF;
  421: 
  422:   INSERT INTO stock_movements (
  423:     organization_id, branch_id, stock_item_id, location_id, batch_id, movement_type,
  424:     quantity, unit, unit_cost, value, balance_after, reference_type, reference_id,
  425:     reason, idempotency_key, performed_by
  426:   ) VALUES (
  427:     v_org, v_item.branch_id, _item, _location, v_batch, _movement_type,
  428:     v_qty_base, v_item.base_unit, v_cost, v_qty_base * v_cost,
  429:     (SELECT quantity_on_hand FROM stock_levels WHERE id = v_level.id),
  430:     _reference_type, _reference_id, _reason, _idempotency_key, auth.uid()
  431:   ) RETURNING id INTO v_movement;
  432: 
  433:   RETURN v_movement;
  434: END;
