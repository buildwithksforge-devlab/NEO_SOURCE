 390: BEGIN
 391:     IF auth.uid() IS NULL OR v_org IS NULL THEN
 392:         RAISE EXCEPTION 'Authentication and organization required'
 393:             USING ERRCODE = '42501';
 394:     END IF;
 395:

 397:         RAISE EXCEPTION 'Independent review authorization required'
 398:             USING ERRCODE = '42501';
 399:     END IF;
 400: 
 401:     IF v_decision NOT IN ('approved', 'rejected') THEN
 402:         RAISE EXCEPTION 'Decision must be approved or rejected'
 403:             USING ERRCODE = '22023';
 404:     END IF;
 405: 
 406:     IF v_notes IS NULL THEN
 407:         RAISE EXCEPTION 'A review reason is required'
 408:             USING ERRCODE = '22023';
 409:     END IF;
 410: 
 411:     SELECT c.*
 412:     INTO v_count
 413:     FROM public.neo_blind_receiving_counts c
 414:     WHERE c.id = _count_id
 415:       AND c.organization_id = v_org
 416:     FOR UPDATE;
 417: 
 418:     IF NOT FOUND THEN
 419:         RAISE EXCEPTION 'Blind receiving count not found'
 420:             USING ERRCODE = '22023';
 421:     END IF;
 422: 
 423:     IF NOT public.neo_branch_in_scope(v_count.branch_id) THEN
 424:         RAISE EXCEPTION 'Count is outside authorized branch'
 425:             USING ERRCODE = '42501';
 426:     END IF;
 427: 
 428:     IF v_count.submitted_by = auth.uid() THEN
 429:         RAISE EXCEPTION 'You cannot review your own receiving count'
 430:             USING ERRCODE = '42501';
 431:     END IF;
 432: 
 433:     IF v_count.status NOT IN ('submitted', 'under_review') THEN
 434:         RAISE EXCEPTION 'This count has already been decided'
 435:             USING ERRCODE = '22023';
 436:     END IF;
 437: 
 438:     SELECT
 439:         count(*),
 440:         count(*) FILTER (
 441:             WHERE (cl.quantity_delivered - cl.quantity_rejected) >
 442:                 GREATEST(
 443:                     COALESCE(pl.quantity, 0)
 444:                     - COALESCE(pl.accepted_quantity, 0)
 445:                     - COALESCE(pl.cancelled_quantity, 0),
 446:                     0
 447:                 )
 448:         )
 449:     INTO v_lines, v_excess
 450:     FROM public.neo_blind_receiving_count_lines cl
 451:     JOIN public.purchase_order_lines pl
 452:       ON pl.id = cl.purchase_order_line_id
 453:      AND pl.order_id = v_count.purchase_order_id
 454:     WHERE cl.count_id = _count_id;

 456:     IF v_lines = 0 THEN
 457:         RAISE EXCEPTION 'Cannot review a count without lines'
 458:             USING ERRCODE = '22023';
 459:     END IF;
 460: 
 461:     IF v_decision = 'approved' AND v_excess > 0 THEN
 462:         RAISE EXCEPTION
 463:             'Excess accepted quantities require separate exception authorization; approval blocked'
 464:             USING ERRCODE = '22023';
 465:     END IF;
 466: 
 467:     UPDATE public.neo_blind_receiving_counts
 468:     SET
 469:         status = v_decision,
 470:         reviewed_by = auth.uid(),
 471:         reviewed_at = now(),
 472:         review_notes = v_notes
 473:     WHERE id = _count_id;
 474: 
 475:     IF v_decision = 'approved' THEN
 476:         v_posting := public.neo_post_approved_blind_receiving(_count_id);

 480:             RAISE EXCEPTION
 481:                 'Approved count did not produce a verified GRN and stock posting'
 482:                 USING ERRCODE = '22023';
 483:         END IF;
 484: 
 485:         RETURN jsonb_build_object(
 486:             'ok', true,
 487:             'count_id', _count_id,
 488:             'status', 'posted',
 489:             'reviewed_by', auth.uid(),
 490:             'grn_id', v_posting->>'grn_id',
 491:             'stock_posted', true,
 492:             'posting_result', v_posting
 493:         );
 494:     END IF;
 495: 
 496:     RETURN jsonb_build_object(
 497:         'ok', true,
 498:         'count_id', _count_id,
 499:         'status', 'rejected',
 500:         'reviewed_by', auth.uid(),
