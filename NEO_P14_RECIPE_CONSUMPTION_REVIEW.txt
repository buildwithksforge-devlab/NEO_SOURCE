 817:         WHEN v_kind = 'waste' THEN 'waste:'

 840:       v_cost

 844:

 846:       v_waste_lines := v_waste_lines + 1;

 851:       v_branch,
 852:       v_comp.item_id,

 940: -- ============================================================
 941: -- Batch-aware void semantics
 942: --
 943: -- direct_stock:
 944: --   may return to physical stock when voided
 945: --
 946: -- recipe:
 947: --   before preparation -> reverse canonical recipe consumption
 948: --   after preparation  -> retain consumption and record waste
 949: --
 950: -- batch:
 951: --   before preparation -> reverse canonical recipe consumption
 952: --                         AND restore exact operational batch quantity
 953: --   after preparation  -> retain recipe consumption
 954: --                         AND retain batch usage as waste
 955: -- ============================================================
 956: 
 957: CREATE OR REPLACE FUNCTION public.neo_pos_item_void_effect()
 958: RETURNS trigger
 959: LANGUAGE plpgsql
 960: SECURITY DEFINER
 961: SET search_path TO 'public'
 962: AS $$
 963: DECLARE
 964:   v_event record;
 965:   v_mv record;
 966:   v_batch_audit record;
 967:   v_alloc record;
 968: 
 969:   v_prepared boolean;
 970:   v_branch uuid;
 971:   v_waste_no text;
 972:   v_cost numeric := 0;
 973:   v_mode text;
 974: BEGIN
 975:   IF NEW.status NOT IN ('voided','cancelled')
 976:      OR OLD.status = NEW.status THEN
 977:     RETURN NEW;
 978:   END IF;
 979: 
 980:   SELECT branch_id
 981:     INTO v_branch
 982:     FROM public.pos_orders
 983:    WHERE id = NEW.order_id;
 984: 
 985:   SELECT consumption_mode
 986:     INTO v_mode
 987:     FROM public.menu_items
 988:    WHERE id = NEW.menu_item_id;
 989: 
 990:   IF v_mode IS NULL OR v_mode = 'none' THEN
 991:     SELECT CASE
 992:              WHEN r.recipe_type = 'direct_sale'
 993:                THEN 'direct_stock'
 994:              ELSE 'recipe'
 995:            END
 996:       INTO v_mode
 997:       FROM public.recipes r
 998:      WHERE r.organization_id = NEW.organization_id
 999:        AND r.menu_item_id = NEW.menu_item_id
1000:        AND r.status = 'active'
1001:      ORDER BY
1002:        (r.recipe_type = 'menu_item') DESC,
1003:        r.version DESC
1004:      LIMIT 1;
1005:   END IF;
1006: 
1007:   /*
1008:    * Direct packaged stock is never treated as kitchen-prepared.
1009:    *
1010:    * Recipe and batch items become prepared once they entered
1011:    * the operational preparation/service lifecycle.
1012:    */
1013:   v_prepared :=
1014:     COALESCE(v_mode, '') <> 'direct_stock'
1015:     AND (
1016:       NEW.sent_at IS NOT NULL
1017:       OR NEW.ready_at IS NOT NULL
1018:       OR NEW.served_at IS NOT NULL
1019:     );
1020: 
1021:   SELECT *
1022:     INTO v_event
1023:     FROM public.inventory_posting_events
1024:    WHERE order_item_id = NEW.id
1025:      AND status = 'posted'
1026:    LIMIT 1;
1027: 
1028:   IF v_event IS NULL THEN
1029:     INSERT INTO public.audit_logs (
1030:       organization_id,
1031:       branch_id,
1032:       station_id,
1033:       user_name,
1034:       action,
1035:       module,
1036:       record_id,
1037:       new_value,
1038:       reason
1039:     )
1040:     VALUES (
1041:       NEW.organization_id,
1042:       v_branch,
1043:       NEW.station_id,
