  255: RETURNS accounting_periods
  256: LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path TO 'public'
  257: AS $$
  258: DECLARE p public.accounting_periods;
  259: BEGIN
  260:   SELECT * INTO p
  261:     FROM public.accounting_periods
  262:    WHERE organization_id = _org AND _date BETWEEN starts_on AND ends_on
  263:    ORDER BY
  264:      CASE status WHEN 'open' THEN 0 WHEN 'closed' THEN 1 WHEN 'locked' THEN 2 ELSE 3 END,
  265:      created_at DESC
  266:    LIMIT 1;

  268:   -- A month whose dates have already arrived is open, not future. Closing and
  269:   -- locking stay deliberate acts and are never undone here.

  271:     p.status := 'open';
  272:   END IF;
  273:   RETURN p;
  274: END $$;

   37: CREATE OR REPLACE FUNCTION public.pms_check_out(_reservation uuid, _force boolean DEFAULT false, _notes text DEFAULT ''::text)
   38:  RETURNS jsonb
   39:  LANGUAGE plpgsql
   40:  SECURITY DEFINER
   41:  SET search_path TO 'public'
   42: AS $function$
   43: DECLARE _stay record; _f record; _charges numeric; _paid numeric; _line_paid numeric; _outstanding numeric;
   44: BEGIN
   45:   SELECT * INTO _stay FROM public.stays WHERE reservation_id=_reservation FOR UPDATE;
   46:   IF _stay IS NULL THEN RAISE EXCEPTION 'No active stay for this reservation'; END IF;
   47:   IF _stay.status = 'checked-out' THEN
   48:     RETURN jsonb_build_object('stay_id',_stay.id,'idempotent',true);
   49:   END IF;
   50: 
   51:   SELECT * INTO _f FROM public.folios WHERE stay_id=_stay.id AND folio_type='guest' FOR UPDATE;
   52: 
   53:   -- Charges exclude payment/deposit lines; those are settlement, not consumption.
   54:   SELECT COALESCE(SUM(l.net_amount), 0) INTO _charges
   55:     FROM public.folio_lines l
   56:     JOIN public.folios f ON f.id = l.folio_id
   57:    WHERE f.stay_id = _stay.id
   58:      AND COALESCE(l.status,'posted') NOT IN ('void','voided','reversed')
   59:      AND COALESCE(l.line_type,'charge') NOT IN ('payment','deposit','refund');
   60: 
   61:   SELECT COALESCE(SUM(a.amount), 0) INTO _paid
   62:     FROM public.payment_allocations a
   63:     JOIN public.folios f ON f.id = a.folio_id
   64:     JOIN public.payments p ON p.id = a.payment_id
   65:    WHERE f.stay_id = _stay.id
   66:      AND p.status = 'confirmed';
   67: 
   68:   -- Deposits applied at check-in are recorded as folio payment lines.
   69:   SELECT COALESCE(SUM(l.net_amount), 0) INTO _line_paid
   70:     FROM public.folio_lines l
   71:     JOIN public.folios f ON f.id = l.folio_id
   72:    WHERE f.stay_id = _stay.id
   73:      AND COALESCE(l.status,'posted') NOT IN ('void','voided','reversed')
   74:      AND COALESCE(l.line_type,'charge') IN ('payment','deposit');
   75: 
   76:   _outstanding := _charges - _paid - _line_paid;
   77: 
   78:   IF round(_outstanding, 2) > 0 THEN
   79:     IF NOT _force THEN

/**

 *
 * A deposit is a liability held against a reservation, not revenue, and it is
 * the gate the arrivals desk must pass before a stay may open. Nothing here
 * computes the position: `neo_reservation_deposit_position` is the single
 * source of truth, and the same rule is enforced again by the database trigger
 * on stays, so a refresh, a reopen or a direct call can never bypass it.
 */

  (supabase as any).rpc(fn, args) as Promise<{ data: any; error: { message: string } | null }>;

  ok: boolean;
  reason?: string;

};

/** Why check-in is or is not permitted, in the words the desk should read. */
export function depositGate(position: DepositPosition | undefined | null): {
  blocked: boolean;
  message: string;
} {
  if (!position || !position.ok) {
    return {
      blocked: true,
      message:
        position?.reason ??
        "Deposit eligibility could not be verified. Refresh the reservation before checking in.",
    };
  }
  if (position.waiver === "corporate-credit") {
    return { blocked: false, message: "Charged to a corporate credit account — no deposit due." };
  }
  if (position.override_id) {
    return {
      blocked: false,
      message: "An authorised override is on file for this arrival. It is used once.",
    };
  }
  const outstanding = Number(position.outstanding ?? 0);
  if (outstanding > 0.009) {
    return {
      blocked: true,
      message: `Deposit outstanding: ${outstanding.toFixed(2)} of ${Number(position.required ?? 0).toFixed(2)} required. Record the deposit or obtain an authorised override.`,
    };
  }
  return { blocked: false, message: "Deposit settled." };
}

    },
  });
}

/**

 * a double-click, a retry or a refreshed request returns the first deposit
 * instead of taking the guest's money twice.
 */
export function useRecordDeposit() {
  const invalidate = useInvalidateDeposits();
  return useMutation({
    mutationFn: async (input: {
      reservationId: string;
      amount: number;
      method: string;
      reference?: string;
      notes?: string;
      idempotencyKey: string;
    }) => {
      const { data, error } = await rpc("neo_reservation_deposit_record", {
        _reservation: input.reservationId,
        _amount: input.amount,
        _method: input.method,
        _reference: input.reference ?? null,
        _idempotency_key: input.idempotencyKey,
        _notes: input.notes ?? null,
      });
      if (error) throw new Error(error.message);
       const result = (data ?? {}) as {
         deposit_id?: string;
         idempotent?: boolean;
         position?: DepositPosition;
         finance?: { ok?: boolean; reason?: string; detail?: string };
       };
       if (!result.deposit_id) throw new Error("The deposit was not persisted.");
       return result;

    onSuccess: (r) => {
      if (r.finance?.ok) {
        toast.success(r.idempotent ? "Deposit already recorded and posted" : "Deposit recorded and posted to Finance");

/**
 * One posting intent per source event and property.
 *
 * A key minted at random per submission made every retry — and every second
 * operator confirming the same money — a *new* deposit, which is how one
 * reservation came to sit twice in the Finance posting queue. The key is now
 * derived from the facts of the intent itself (property, reservation, amount,
 * method, reference, business date), so repeating the same intent returns the
 * existing deposit while a genuinely different one still records.
 */
export const depositIdempotencyKey = (input: {
  reservationId: string;
  propertyId: string | null;
  amount: number;
  method: string;
  reference?: string;
  businessDate?: string;
}) =>
  stableIdempotencyKey("deposit", "record", [
    input.propertyId,
    input.reservationId,
    input.amount.toFixed(2),
    input.method,
    input.reference ?? "",
    input.businessDate ?? new Date().toISOString().slice(0, 10),
  ]);
