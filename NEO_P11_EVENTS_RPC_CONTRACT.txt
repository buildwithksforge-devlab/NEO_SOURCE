   5: CREATE OR REPLACE FUNCTION public.neo_event_stock_need_to_requisition(
   6:     _need uuid
   7: )
   8: RETURNS jsonb
   9: LANGUAGE plpgsql
  10: SECURITY DEFINER
  11: SET search_path = public
  12: AS $function$
  13: DECLARE
  14:     v_org uuid := public.current_org_id();
  15:     v_need public.department_stock_needs;
  16:     v_req uuid;
  17:     v_number text;
  18: BEGIN
  19: 
  20:     IF auth.uid() IS NULL
  21:        OR v_org IS NULL
  22:        OR public.is_active_staff() IS NOT TRUE THEN
  23:         RAISE EXCEPTION 'An active staff session is required';
  24:     END IF;
  25: 
  26:     -- Only authorised Events managers may raise this requisition.
  27:     IF public.has_permission('events', 'stock_needs', 'approve')
  28:        IS NOT TRUE THEN
  29:         RAISE EXCEPTION
  30:           'Only an authorised Events Manager can requisition stock';
  31:     END IF;
  32: 
  33:     -- Lock the originating need to prevent concurrent duplication.
  34:     SELECT *
  35:       INTO v_need
  36:       FROM public.department_stock_needs
  37:      WHERE id = _need
  38:        AND organization_id = v_org
  39:        AND module = 'events'
  40:      FOR UPDATE;
  41: 
  42:     IF NOT FOUND THEN
  43:         RAISE EXCEPTION 'Event stock need not found';
  44:     END IF;
  45: 
  46:     IF v_need.branch_id IS NULL
  47:        OR NOT (v_need.branch_id = ANY(public.current_branch_ids())) THEN
  48:         RAISE EXCEPTION 'The stock need is outside your property access';
  49:     END IF;
  50: 
  51:     -- E04: Every new Events requisition must belong to a real,
  52:     -- active booking in the same organization and property.
  53:     IF v_need.event_booking_id IS NULL THEN
  54:         RAISE EXCEPTION
  55:           'Select an event booking before requisitioning supplies';
  56:     END IF;
  57: 
  58:     IF NOT EXISTS (
  59:         SELECT 1
  60:         FROM public.event_bookings b
  61:         WHERE b.id = v_need.event_booking_id
  62:           AND b.organization_id = v_org
  63:           AND b.branch_id = v_need.branch_id
  64:           AND lower(coalesce(b.status, '')) NOT IN (
  65:               'cancelled',
  66:               'closed',
  67:               'completed',
  68:               'no-show'
  69:           )
  70:     ) THEN
  71:         RAISE EXCEPTION
  72:           'The originating event is invalid, closed, or belongs to another property';
  73:     END IF;
  74: 
  75:     -- Return a previously linked requisition without creating another.
  76:     IF v_need.requisition_id IS NOT NULL THEN
  77: 
  78:         SELECT id, request_number
  79:           INTO v_req, v_number
  80:           FROM public.internal_requisitions
  81:          WHERE id = v_need.requisition_id
  82:            AND organization_id = v_org
  83:            AND branch_id = v_need.branch_id
  84:            AND department_id IS NOT DISTINCT FROM v_need.department_id
  85:            AND event_booking_id = v_need.event_booking_id;
  86: 
  87:         IF v_req IS NULL THEN
  88:             RAISE EXCEPTION
  89:               'The stock need references an invalid requisition';
  90:         END IF;
  91: 
  92:         RETURN jsonb_build_object(
  93:             'ok', true,
  94:             'already_exists', true,
  95:             'requisition_id', v_req,
  96:             'request_number', v_number,
  97:             'status', 'requisitioned'
  98:         );
  99:     END IF;
 100: 
 101:     IF v_need.status NOT IN ('pending', 'acknowledged') THEN
 102:         RAISE EXCEPTION
 103:           'Only pending or acknowledged stock needs can be requisitioned';
 104:     END IF;
 105: 
 106:     IF coalesce(v_need.quantity, 0) <= 0 THEN
 107:         RAISE EXCEPTION 'Requested quantity must be greater than zero';
 108:     END IF;
 109: 
 110:     v_number :=
 111:         'REQ-' ||
 112:         to_char(now(), 'YYMMDD') ||
 113:         '-' ||
 114:         upper(substr(replace(gen_random_uuid()::text, '-', ''), 1, 5));
 115: 
 116:     -- New requisitions require supervisor approval.
 117:     -- The creator is never recorded as their own approver.
 118:     INSERT INTO public.internal_requisitions (
 119:         organization_id,
 120:         branch_id,
 121:         request_number,
 122:         department_id,
 123:         event_booking_id,
 124:         fulfilment_model,
 125:         requested_by,
 126:         request_date,
 127:         priority,
 128:         reason,
 129:         status
 130:     )
 131:     VALUES (
 132:         v_org,
 133:         v_need.branch_id,
 134:         v_number,
 135:         v_need.department_id,
 136:         v_need.event_booking_id,
 137:         'stores-issue',
 138:         auth.uid(),
 139:         current_date,
 140:         coalesce(v_need.priority, 'normal'),
 141:         'Events stock need: ' || v_need.item ||
 142:           ' (Need ' || v_need.id::text || ')',
 143:         'pending-supervisor'
 144:     )
 145:     RETURNING id INTO v_req;

 148:         organization_id,
