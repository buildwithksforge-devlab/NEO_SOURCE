   2: RETURNS uuid[]
   3: LANGUAGE sql
   4: STABLE
   5: SECURITY DEFINER
   6: SET search_path = public
   7: AS $$
   8:   WITH me AS (
   9:     SELECT public.current_org_id()          AS org,
  10:            public.current_access_scope()    AS scope,
  11:            public.current_branch_ids()      AS branches,
  12:            public.current_department_ids()  AS depts,
  13:            public.current_station_ids()     AS stations
  14:   )
  15:   SELECT COALESCE(array_agg(DISTINCT s.id), '{}'::uuid[])
  16:   FROM public.operational_stations s, me
  17:   WHERE me.org IS NOT NULL
  18:     AND s.organization_id = me.org
  19:     -- Property boundary stays hard for everyone below organisation reach.
  20:     AND (
  21:       public.scope_rank(me.scope) >= 5
  22:       OR s.branch_id IS NULL
  23:       OR s.branch_id = ANY (me.branches)
  24:     )
  25:     AND (
  26:       public.scope_rank(me.scope) >= 5
  27:       -- Broad production oversight is a configured capability, never a role name.
  28:       OR public.has_permission('kitchen', 'all_stations', 'view')
  29:       -- Explicitly assigned production station(s).
  30:       OR s.id = ANY (me.stations)
  31:       -- Departmental oversight is for supervisory reach only. Frontline staff
  32:       -- (team/self scope) see the station(s) they are assigned to and nothing
  33:       -- else, so a barista never sees Main Bar work and a bar attendant never
  34:       -- sees Coffee Station work just because both sit under Food & Beverage.
  35:       OR (
  36:         public.scope_rank(me.scope) >= 3
  37:         AND s.department_id IS NOT NULL
  38:         AND s.department_id = ANY (me.depts)
  39:       )
  40:     );
  41: $$;

   1: /**
   2:  * Kitchen production visibility contract.
   3:  *
   4:  * Mirrors `public.neo_kitchen_visible_station_ids()` and the

   7:  *
   8:  * Rule (in order):
   9:  *  1. Tenant boundary: the ticket must belong to the viewer's organisation.
  10:  *  2. Property boundary: below organisation/technical reach, the station's
  11:  *     property must be one of the viewer's properties.
  12:  *  3. Reach:
  13:  *     - organisation / technical scope  â†’ every production station,
  14:  *     - `kitchen.all_stations.view`     â†’ every production station at the property,
  15:  *     - explicitly assigned station(s)  â†’ those stations,
  16:  *     - department scope or above       â†’ stations of the viewer's department(s),
  17:  *     - frontline (team/self) scope     â†’ assigned stations only.
  18:  *  4. The viewer must additionally hold `kitchen.queue.view`.
  19:  */

  21: export const SCOPE_RANK: Record<string, number> = {
  22:   organisation: 5,
  23:   technical: 5,
  24:   branch: 4,
  25:   department: 3,
  26:   team: 2,
  27:   self: 1,
  28: };

  31:   return SCOPE_RANK[scope ?? "self"] ?? 1;

  53: export function visibleStationIds(
  54:   viewer: VisibilityViewer,
  55:   stations: VisibilityStation[],
  56: ): string[] {
  57:   const universal = scopeRank(viewer.scope) >= 5;
  58:   return stations
  59:     .filter((s) => s.organizationId === viewer.organizationId)
  60:     .filter((s) => universal || s.branchId == null || viewer.branchIds.includes(s.branchId))
  61:     .filter(
  62:       (s) =>
  63:         universal ||
  64:         viewer.allStations ||
  65:         viewer.stationIds.includes(s.id) ||
  66:         // Departmental reach is supervisory only: frontline staff (team/self)
  67:         // see the station(s) they are assigned to, never every station that
  68:         // happens to share their department.
  69:         (scopeRank(viewer.scope) >= 3 &&
  70:           s.departmentId != null &&
  71:           viewer.departmentIds.includes(s.departmentId)),
  72:     )
  73:     .map((s) => s.id);
  74: }
