function isNewSupabaseApiKey(value: string): boolean {
  return value.startsWith('sb_publishable_') || value.startsWith('sb_secret_');
}

function createSupabaseFetch(supabaseKey: string): typeof fetch {
  return (input, init) => {
    const headers = new Headers(
      typeof Request !== 'undefined' && input instanceof Request ? input.headers : undefined,
    );

    if (init?.headers) {
      new Headers(init.headers).forEach((value, key) => headers.set(key, value));
    }

    // New Supabase API keys are opaque strings, not bearer JWTs.
    if (isNewSupabaseApiKey(supabaseKey) && headers.get('Authorization') === `Bearer ${supabaseKey}`) {
      headers.delete('Authorization');
    }

    headers.set('apikey', supabaseKey);
    return fetch(input, { ...init, headers });
  };
}


function createSupabaseClient() {
  // Use import.meta.env for client-side (Vite build-time replacement)
  // Fall back to process.env for SSR (server-side rendering)
  const SUPABASE_URL = import.meta.env['VITE_SUPABASE_URL'] || process.env['SUPABASE_URL'];
  const SUPABASE_PUBLISHABLE_KEY = import.meta.env['VITE_SUPABASE_PUBLISHABLE_KEY'] || process.env['SUPABASE_PUBLISHABLE_KEY'];

  if (!SUPABASE_URL || !SUPABASE_PUBLISHABLE_KEY) {
    const missing = [
      ...(!SUPABASE_URL ? ['SUPABASE_URL'] : []),
      ...(!SUPABASE_PUBLISHABLE_KEY ? ['SUPABASE_PUBLISHABLE_KEY'] : []),
    ];
    const message = `Missing Supabase environment variable(s): ${missing.join(', ')}. Connect Supabase in Lovable Cloud.`;
    console.error(`[Supabase] ${message}`);
    throw new Error(message);
  }

  return createClient<Database>(SUPABASE_URL, SUPABASE_PUBLISHABLE_KEY, {
    global: {
      fetch: createSupabaseFetch(SUPABASE_PUBLISHABLE_KEY),
    },
    auth: {
      storage: brokeredPreviewStorage(),
      persistSession: true,
      autoRefreshToken: true,
    }
  });
}

let _supabase: ReturnType<typeof createSupabaseClient> | undefined;

// Import the supabase client like this:
// import { supabase } from "@/integrations/supabase/client";

  get(_, prop, receiver) {
    if (!_supabase) _supabase = createSupabaseClient();
    return Reflect.get(_supabase, prop, receiver);
  },
});

export const requireSupabaseAuth = createMiddleware({ type: 'function' }).server(
  async ({ next }) => {
    
    const SUPABASE_URL = process.env['SUPABASE_URL'];
    const SUPABASE_PUBLISHABLE_KEY = process.env['SUPABASE_PUBLISHABLE_KEY'];

    if (!SUPABASE_URL || !SUPABASE_PUBLISHABLE_KEY) {
      const missing = [
        ...(!SUPABASE_URL ? ['SUPABASE_URL'] : []),
        ...(!SUPABASE_PUBLISHABLE_KEY ? ['SUPABASE_PUBLISHABLE_KEY'] : []),
      ];
      const message = `Missing Supabase environment variable(s): ${missing.join(', ')}. Connect Supabase in Lovable Cloud.`;
      console.error(`[Supabase] ${message}`);
      throw new Error(message);
    }
    
    const request = getRequest();

    if (!request?.headers) {
      throw new Error('Unauthorized: No request headers available');
    }

    const authHeader = request.headers.get('authorization');

    if (!authHeader) {
      throw new Error('Unauthorized: No authorization header provided');
    }

    if (!authHeader.startsWith('Bearer ')) {
      throw new Error('Unauthorized: Only Bearer tokens are supported');
    }

    const token = authHeader.replace('Bearer ', '');
    if (!token) {
      throw new Error('Unauthorized: No token provided');
    }

    if (token.split('.').length !== 3) {
      throw new Error('Unauthorized: Invalid token');
    }

    const supabase = createClient<Database>(
      SUPABASE_URL!,
      SUPABASE_PUBLISHABLE_KEY!,
      {
        global: {
          fetch: createSupabaseFetch(SUPABASE_PUBLISHABLE_KEY!),
          headers: {
            Authorization: `Bearer ${token}`,
          },
        },
        auth: {
          storage: undefined,
          persistSession: false,
          autoRefreshToken: false,
        },
      }
    );

    const { data, error } = await supabase.auth.getClaims(token);
    if (error || !data?.claims) {
      throw new Error('Unauthorized: Invalid token');
    }

    if (!data.claims.sub) {
      throw new Error('Unauthorized: No user ID found in token');
    }

    return next({
      context: {
        supabase,
        userId: data.claims.sub,
        claims: data.claims,
      },
    });
  },
);

import { createServerFn } from "@tanstack/react-start";
import type { SupabaseClient } from "@supabase/supabase-js";
import { requireSupabaseAuth } from "@/integrations/supabase/auth-middleware";
import type { Database } from "@/integrations/supabase/types";
import { buildStaffInviteMetadata } from "@/lib/auth-email-metadata";
import {
  isDuplicateAccountError,
  normaliseStaffAccountInput,
  validateStaffAccountInput,
} from "@/lib/staff-account-input";

type CapabilityClient = Pick<SupabaseClient<Database>, "rpc">;

/**
 * Authority for staff administration comes from effective capabilities, never
 * from a role name or the organisation-admin flag. `is_org_admin` remains a
 * fallback so an Owner is never locked out of their own workspace.
 */
async function requireCapability(
  supabase: CapabilityClient,
  feature: string,
  action: string,
) {
  const { data: allowed } = await supabase.rpc("has_permission", {
    _module: "administration",
    _feature: feature,
    _action: action,
  });
  if (allowed === true) return;
  const { data: isAdmin } = await supabase.rpc("is_org_admin");
  if (isAdmin !== true) throw new Error("Forbidden");
}

/**
 * Canonical role ↔ department ↔ property check. The interface filters the
 * dropdowns, but a typed request must never seat a role in a department it does
 * not belong to, so the same catalogue relationships are re-evaluated in the
 * database by `neo_validate_role_assignment` before anything is written.
 */
async function requireValidAssignment(
  supabase: CapabilityClient,
  branchId: string | null,
  departmentId: string | null,
  roleIds: string[],
) {
  const { error } = await supabase.rpc("neo_validate_role_assignment" as never, {
    _branch: branchId,
    _department: departmentId,
    _role_ids: roleIds ?? [],
  } as never);
  if (error) throw new Error(error.message);
}


type CreateInput = {
  email: string;
  password: string;
  full_name: string;
  staff_id: string;
  phone: string;
  job_title: string;
  branch_id: string | null;
  department_id: string | null;
  station_id?: string | null;
  employment_status: string;
  role_ids: string[];
  primary_role_id?: string | null;
  /** Organisational reporting line, chosen from the canonical candidates. */
  reporting_manager_profile_id?: string | null;
};

/**
 * Writes the organisational reporting line for one staff account.
 *
 * Eligibility is always re-resolved through `neo_reporting_candidates`, which
 * walks the canonical role hierarchy (including its unstaffed-role fallback),
 * so a typed request cannot seat a manager the interface would have refused.
 * A database trigger additionally refuses self-reporting, cycles, inactive
 * managers and cross-organisation managers.
 */
async function writeReportingLine(args: {
  supabase: CapabilityClient;
  supabaseAdmin: SupabaseClient<Database>;
  organizationId: string;
  userId: string;
  branchId: string | null;
  departmentId: string | null;
  stationId: string | null;
  roleIds: string[];
  managerId: string | null;
}) {
  const managerId = args.managerId || null;
  if (managerId) {
    if (managerId === args.userId) throw new Error("An employee cannot report to themselves");
    const { data: candidates, error } = await args.supabase.rpc(
      "neo_reporting_candidates" as never,
      {
        _user: args.userId,
        _branch: args.branchId,
        _department: args.departmentId,
        _station: args.stationId,
        _role_ids: args.roleIds,
      } as never,
    );
    if (error) throw new Error(error.message);
    const eligible = ((candidates ?? []) as { profile_id: string }[]).some(
      (c) => c.profile_id === managerId,
    );
    if (!eligible) {
      throw new Error(
        "That manager is not eligible for this employee's roles, department or property",
      );
    }
  }

  let reviewRequired = false;
  let reviewReason = "";
  if (!managerId) {
    const { data: meta } = await args.supabase.rpc("neo_role_reporting_meta" as never, {
      _role_ids: args.roleIds,
    } as never);
    const required = ((meta ?? []) as { manager_required: boolean }[])[0]?.manager_required;
    if (required) {
      reviewRequired = true;
      reviewReason = "No reporting manager recorded — confirm the reporting line.";
    }
  }

  const { error: managerError } = await args.supabaseAdmin
    .from("profiles")
    .update({
      reporting_manager_id: managerId,
      reporting_effective_from: managerId ? new Date().toISOString().slice(0, 10) : null,
      assignment_review_required: reviewRequired,
      assignment_review_reason: reviewReason,
    })
    .eq("id", args.userId)
    .eq("organization_id", args.organizationId);
  if (managerError) throw new Error(managerError.message);
  return managerId;
}


/**
 * Universal worker provisioning.
 *
 * The authentication account is the only thing created here; everything that
 * makes a worker functional — profile, property, department, work area, roles,
 * primary role, notification preferences and the audit record — is written by
 * the single backend routine `neo_provision_worker`, which validates scope,
 * refuses to grant restricted authority incidentally and records a resumable
 * provisioning run. Retrying with the same account produces no duplicates.
 */
export const createStaffUser = createServerFn({ method: "POST" })
  .middleware([requireSupabaseAuth])
  .inputValidator((input: CreateInput) => {
    const normalised = normaliseStaffAccountInput(input);
    const problem = validateStaffAccountInput(normalised);
    if (problem) throw new Error(problem);
    return normalised;
  })
  .handler(async ({ data, context }) => {
    const { supabase, userId } = context;

    await requireCapability(supabase, "users", "create");
    await requireValidAssignment(supabase, data.branch_id, data.department_id, data.role_ids);


    const { data: me } = await supabase
      .from("profiles")
      .select("organization_id, full_name")
      .eq("id", userId)
      .maybeSingle();
    if (!me?.organization_id) throw new Error("No organization");

    // Personalisation for the branded confirmation e-mail. Every lookup is
    // read through the caller's own session, so it can only ever resolve names
    // inside their tenant, and only display names reach the payload.
    const [{ data: org }, { data: property }, { data: role }] = await Promise.all([
      supabase.from("organizations").select("name").eq("id", me.organization_id).maybeSingle(),
      data.branch_id
        ? supabase.from("branches").select("name").eq("id", data.branch_id).maybeSingle()
        : Promise.resolve({ data: null as { name: string } | null }),
      data.primary_role_id ?? data.role_ids[0]
        ? supabase
            .from("roles")
            .select("name")
            .eq("id", (data.primary_role_id ?? data.role_ids[0]) as string)
            .maybeSingle()
        : Promise.resolve({ data: null as { name: string } | null }),
    ]);

    const inviteMetadata = buildStaffInviteMetadata({
      full_name: data.full_name,
      organisation_name: org?.name ?? null,
      property_name: property?.name ?? null,
      role_title: role?.name ?? null,
      job_title: data.job_title,
      invited_by_name: me.full_name ?? null,
    });

    const { supabaseAdmin } = await import("@/integrations/supabase/client.server");

    // Reuse an existing authentication account rather than failing a retry.
    let newUserId: string | null = null;
    const created = await supabaseAdmin.auth.admin.createUser({
      email: data.email,
      password: data.password,
      email_confirm: true,
      user_metadata: inviteMetadata,
    });
    if (created.error) {
      const { data: list } = await supabaseAdmin.auth.admin.listUsers();
      const existing = list?.users?.find((u) => (u.email ?? "").toLowerCase() === data.email);
      if (!existing) throw new Error(created.error.message);

      // Resuming an interrupted provisioning run for this workspace is safe.
      // Adopting an account that already belongs to somebody — in this or any
      // other workspace — is not, so that case is reported as a duplicate.
      const { data: existingProfile } = await supabaseAdmin
        .from("profiles")
        .select("id, organization_id")
        .eq("id", existing.id)
        .maybeSingle();
      if (existingProfile && existingProfile.organization_id !== me.organization_id) {
        throw new Error("That email address already belongs to another account.");
      }
      if (!existingProfile && !isDuplicateAccountError(created.error.message)) {
        throw new Error(created.error.message);
      }
      newUserId = existing.id;
    } else {
      newUserId = created.data.user?.id ?? null;
    }
    if (!newUserId) throw new Error("Could not create the account");

    const { data: result, error } = await supabase.rpc("neo_provision_worker", {
      _payload: {
        user_id: newUserId,
        email: data.email,
        full_name: data.full_name,
        staff_id: data.staff_id,
        phone: data.phone,
        job_title: data.job_title,
        branch_id: data.branch_id,
        department_id: data.department_id,
        station_id: data.station_id ?? null,
        role_ids: data.role_ids,
        primary_role_id: data.primary_role_id ?? data.role_ids[0] ?? null,
        employment_status: data.employment_status,
        idempotency_key: `worker:${data.email.toLowerCase()}`,
      } as never,
    });
    if (error) throw new Error(error.message);

    // Organisational reporting line, recorded with the account rather than as
    // a follow-up task. Leaving it empty flags assignment review instead.
    const managerId = await writeReportingLine({
      supabase,
      supabaseAdmin,
      organizationId: me.organization_id,
      userId: newUserId,
      branchId: data.branch_id,
      departmentId: data.department_id,
      stationId: data.station_id ?? null,
      roleIds: data.role_ids,
      managerId: data.reporting_manager_profile_id ?? null,
    });
    if (managerId) {
      await supabaseAdmin.from("audit_logs").insert({
        organization_id: me.organization_id,
        user_id: userId,
        user_name: me.full_name ?? "",
        action: "user.reporting_line_set",
        module: "administration",
        record_id: newUserId,
        previous_value: { reporting_manager_id: null } as never,
        new_value: {
          reporting_manager_id: managerId,
          branch_id: data.branch_id,
          department_id: data.department_id,
          role_ids: data.role_ids,
        } as never,
      });
    }

    return { id: newUserId, provisioning: (result ?? null) as Record<string, string> | null };

  });


type AssignmentInput = {
  user_id: string;
  job_title: string;
  employment_status: string;
  branch_id: string | null;
  department_id: string | null;
  station_id: string | null;
  role_ids: string[];
  reporting_manager_profile_id?: string | null;
};


/**
 * Single authoritative write path for an employee's assignment.
 *
 * User Management previously edited nothing but the account status, so a role,
 * property, department or station change made in the interface never reached
 * `user_roles`, `user_branches`, `user_departments` or `user_stations` — which
 * is exactly why Role Testing and an actual login disagreed with the directory.
 * Everything is written together here, always org-scoped and audited.
 */
export const updateStaffAssignment = createServerFn({ method: "POST" })
  .middleware([requireSupabaseAuth])
  .inputValidator((input: AssignmentInput) => input)
  .handler(async ({ data, context }) => {
    const { supabase, userId } = context;

    await requireCapability(supabase, "users", "edit");
    await requireValidAssignment(supabase, data.branch_id, data.department_id, data.role_ids);


    const { data: me } = await supabase
      .from("profiles")
      .select("organization_id, full_name")
      .eq("id", userId)
      .maybeSingle();
    const organizationId = me?.organization_id;
    if (!organizationId) throw new Error("No organization");

    // Target must live in the caller's organisation — never cross-tenant.
    const { data: target } = await supabase
      .from("profiles")
      .select("id, organization_id, branch_id, department_id, job_title, employment_status")
      .eq("id", data.user_id)
      .maybeSingle();
    if (!target || target.organization_id !== organizationId) {
      throw new Error("User not found in your organization");
    }

    /**
     * Everything below is validated *before* anything is written, and the
     * write itself happens inside one database routine. A station change can
     * therefore never leave the account with its role deleted because a later
     * step failed — either the whole assignment lands or none of it does.
     */
    let previousManagerId: string | null = null;
    let newManagerId: string | null = null;
    let reviewRequired = false;
    let reviewReason = "";
    const setManager = data.reporting_manager_profile_id !== undefined;

    if (setManager) {
      const { data: currentLine } = await supabase
        .from("profiles")
        .select("reporting_manager_id")
        .eq("id", data.user_id)
        .maybeSingle();
      previousManagerId =
        (currentLine as { reporting_manager_id: string | null } | null)?.reporting_manager_id ??
        null;
      newManagerId = data.reporting_manager_profile_id || null;

      if (newManagerId) {
        if (newManagerId === data.user_id) {
          throw new Error("An employee cannot report to themselves");
        }
        const { data: candidates, error: candidateError } = await supabase.rpc(
          "neo_reporting_candidates" as never,
          {
            _user: data.user_id,
            _branch: data.branch_id,
            _department: data.department_id,
            _station: data.station_id,
            _role_ids: data.role_ids,
          } as never,
        );
        if (candidateError) throw new Error(candidateError.message);
        const eligible = ((candidates ?? []) as { profile_id: string }[]).some(
          (c) => c.profile_id === newManagerId,
        );
        if (!eligible) {
          throw new Error(
            "That manager is not eligible for this employee's roles, department or property",
          );
        }
      } else {
        const { data: meta } = await supabase.rpc("neo_role_reporting_meta" as never, {
          _role_ids: data.role_ids,
        } as never);
        const required = ((meta ?? []) as { manager_required: boolean }[])[0]?.manager_required;
        if (required) {
          reviewRequired = true;
          reviewReason = "No reporting manager recorded — confirm the reporting line.";
        }
      }
    }

    const { data: saved, error: saveError } = await supabase.rpc(
      "neo_staff_assignment_save" as never,
      {
        _user: data.user_id,
        _job_title: data.job_title,
        _employment_status: data.employment_status,
        _branch: data.branch_id || null,
        _department: data.department_id || null,
        _role_ids: data.role_ids,
        _station: data.station_id || null,
        _reporting_manager: newManagerId,
        _set_reporting_manager: setManager,
      } as never,
    );
    if (saveError) throw new Error(saveError.message);

    const { supabaseAdmin } = await import("@/integrations/supabase/client.server");

    if (setManager) {
      await supabaseAdmin
        .from("profiles")
        .update({
          reporting_effective_from: newManagerId ? new Date().toISOString().slice(0, 10) : null,
          assignment_review_required: reviewRequired,
          assignment_review_reason: reviewReason,
        })
        .eq("id", data.user_id)
        .eq("organization_id", organizationId);
    }

    const snapshot = (saved ?? {}) as { before?: unknown; after?: unknown };

    await supabaseAdmin.from("audit_logs").insert({
      organization_id: organizationId,
      user_id: userId,
      user_name: me?.full_name ?? "",
      action: "user.assignment_updated",
      module: "administration",
      record_id: data.user_id,
      previous_value: (snapshot.before ?? {
        branch_id: target.branch_id,
        department_id: target.department_id,
        job_title: target.job_title,
        employment_status: target.employment_status,
        reporting_manager_id: previousManagerId,
      }) as never,
      new_value: (snapshot.after ?? {
        branch_id: data.branch_id,
        department_id: data.department_id,
        station_id: data.station_id,
        job_title: data.job_title,
        employment_status: data.employment_status,
        role_ids: data.role_ids,
        reporting_manager_id: newManagerId,
      }) as never,
    });

    return { ok: true, assignment: snapshot.after ?? null };
  });

/**
 * Individual permission grant or denial, optionally time-limited.
 *
 * An individual grant never widens what the workspace itself allows: it can
 * only name a capability that already exists in the catalogue, and the
 * effective-access resolver still applies scope and denials afterwards.
 */
export const setUserPermissionOverride = createServerFn({ method: "POST" })
  .middleware([requireSupabaseAuth])
  .inputValidator(
    (input: {
      user_id: string;
      module: string;
      feature: string;
      action: string;
      effect: "grant" | "deny" | "clear";
      valid_until?: string | null;
      reason?: string | null;
    }) => input,
  )
  .handler(async ({ data, context }) => {
    const { supabase, userId } = context;
    await requireCapability(supabase, "users", "override_permission");

    const { data: target } = await supabase
      .from("profiles")
      .select("organization_id")
      .eq("id", data.user_id)
      .maybeSingle();
    if (!target?.organization_id) throw new Error("User not found in your organization");

    const { data: permission } = await supabase
      .from("permissions")
      .select("id")
      .eq("module", data.module)
      .eq("feature", data.feature)
      .eq("action", data.action)
      .maybeSingle();
    if (!permission?.id) throw new Error("Unknown capability");

    const { supabaseAdmin } = await import("@/integrations/supabase/client.server");

    if (data.effect === "clear") {
      await supabaseAdmin
        .from("user_permissions")
        .delete()
        .eq("user_id", data.user_id)
        .eq("permission_id", permission.id);
    } else {
      const { error } = await supabaseAdmin.from("user_permissions").upsert(
        {
          organization_id: target.organization_id,
          user_id: data.user_id,
          permission_id: permission.id,
          effect: data.effect,
          granted_by: userId,
          valid_from: new Date().toISOString(),
          valid_until: data.valid_until ?? null,
          reason: data.reason ?? null,
        },
        { onConflict: "user_id,permission_id" },
      );
      if (error) throw new Error(error.message);
    }

    await supabaseAdmin.from("audit_logs").insert({
      organization_id: target.organization_id,
      user_id: userId,
      action: `user.permission_${data.effect}`,
      module: "administration",
      record_id: data.user_id,
      new_value: {
        module: data.module,
        feature: data.feature,
        action: data.action,
        effect: data.effect,
        valid_until: data.valid_until ?? null,
        reason: data.reason ?? null,
      } as never,
    });
    return { ok: true };
  });

/**
 * Every mutation below is server-authorised through the same capability gate and
 * writes an audit record with actor, target, previous/new state and, where the
 * action carries consequence, the reason given by the administrator. No handler
 * ever reads or returns an existing password, hash or token — a password can
 * only be replaced, never revealed.
 */
async function loadTarget(
  supabase: CapabilityClient & { from: SupabaseClient<Database>["from"] },
  userId: string,
) {
  const { data: target } = await supabase
    .from("profiles")
    .select("organization_id, account_status, must_change_password, email, full_name")
    .eq("id", userId)
    .maybeSingle();
  if (!target?.organization_id) throw new Error("User not found in your organization");
  return { ...target, organization_id: target.organization_id as string };
}

      new_value: { factors: 0 } as never,
      reason: data.reason ?? null,
    });
    return { ok: true, removed: factors.length };
  });

/**
 * Sign-in and security metadata for the account panel. Only non-secret
 * attributes are returned — never a password hash, recovery code or token.
 */

  .middleware([requireSupabaseAuth])
  .inputValidator((input: { user_id: string }) => input)
  .handler(async ({ data, context }) => {

      updated_at: u?.updated_at ?? null,
      mfa_factors: factors.length,
      mfa_verified_factors: verified.length,
      mfa_state: (factors.length > 0 ? "enrolled" : "not_enrolled") as
        | "enrolled"
        | "not_enrolled",
      mfa_factor_types: [
        ...new Set(factors.map((f) => (f as { factor_type?: string }).factor_type ?? "unknown")),
      ],
      providers: (u?.identities ?? []).map((i) => i.provider),

/**
 * NEØ AI provider resolution — server only.
 *
 * A workspace may either use the built-in Lovable AI gateway or point NEØ at
 * its own OpenAI-compatible provider. The secret key never leaves the server:
 * it is read with the service-role client inside a handler and used only to
 * make the outbound request.
 */
export type AiProviderConfig = {
  provider: string;
  model: string;
  baseUrl: string;
  apiKey: string | null;
  enabled: boolean;
  source: "workspace" | "platform" | "none";
};

const LOVABLE_BASE = "https://ai.gateway.lovable.dev/v1";
const DEFAULT_BASE: Record<string, string> = {
  lovable: LOVABLE_BASE,
  openai: "https://api.openai.com/v1",
};

export const DEFAULT_MODEL = "google/gemini-3-flash-preview";

/** Resolves the provider for an organisation, falling back to the platform gateway. */
export async function resolveAiProvider(organizationId: string | null): Promise<AiProviderConfig> {
  const platformKey = process.env["LOVABLE_API_KEY"] ?? null;
  const fallback: AiProviderConfig = {
    provider: "lovable",
    model: DEFAULT_MODEL,
    baseUrl: LOVABLE_BASE,
    apiKey: platformKey,
    enabled: true,
    source: platformKey ? "platform" : "none",
  };

  if (!organizationId) return fallback;

  try {
    const { supabaseAdmin } = await import("@/integrations/supabase/client.server");
    const { data } = await (supabaseAdmin as unknown as {
      from: (t: string) => any; // eslint-disable-line @typescript-eslint/no-explicit-any
    })
      .from("ai_provider_settings")
      .select("provider, model, base_url, api_key, enabled")
      .eq("organization_id", organizationId)
      .maybeSingle();

    if (!data) return fallback;
    const row = data as Record<string, unknown>;
    const provider = String(row["provider"] ?? "lovable");
    const key = (row["api_key"] as string | null) || null;
    const baseUrl =
      (row["base_url"] as string | null) || DEFAULT_BASE[provider] || LOVABLE_BASE;
    const enabled = row["enabled"] !== false;

    if (provider === "lovable" && !key) {
      return { ...fallback, model: String(row["model"] ?? DEFAULT_MODEL), enabled };
    }
    return {
      provider,
      model: String(row["model"] ?? DEFAULT_MODEL),
      baseUrl,
      apiKey: key,
      enabled,
      source: key ? "workspace" : platformKey ? "platform" : "none",
    };
  } catch {
    return fallback;
  }
}

export type ProviderCallResult = {
  ok: boolean;
  text: string | null;
  status: "ok" | "not_configured" | "disabled" | "quota" | "unauthorised" | "unavailable";
  detail?: string;
  httpStatus?: number;
};

/** Single outbound call used by both the narrative path and the connectivity test. */
export async function callProvider(
  config: AiProviderConfig,
  messages: { role: string; content: string }[],
  maxTokens = 320,
): Promise<ProviderCallResult> {
  if (!config.enabled) {
    return { ok: false, text: null, status: "disabled", detail: "AI is switched off for this workspace." };
  }
  if (!config.apiKey) {
    return { ok: false, text: null, status: "not_configured", detail: "No AI provider key is configured." };
  }

  try {
    const res = await fetch(`${config.baseUrl.replace(/\/$/, "")}/chat/completions`, {
      method: "POST",
      headers: { "Content-Type": "application/json", Authorization: `Bearer ${config.apiKey}` },
      body: JSON.stringify({ model: config.model, temperature: 0.2, max_tokens: maxTokens, messages }),
    });

    if (res.ok) {
      const json = (await res.json()) as { choices?: { message?: { content?: string } }[] };
      const text = json?.choices?.[0]?.message?.content ?? null;
      return { ok: Boolean(text), text, status: text ? "ok" : "unavailable", httpStatus: res.status };
    }

    const body = await res.text();
    const quota = /insufficient_quota|credit_balance|no credits|quota/i.test(body);
    if (res.status === 402 || quota) {
      return {
        ok: false,
        text: null,
        status: "quota",
        httpStatus: res.status,
        detail: "The AI provider reports no remaining credits or quota for this key.",
      };
    }
    if (res.status === 429) {
      return { ok: false, text: null, status: "quota", httpStatus: 429, detail: "AI usage limit reached — try again shortly." };
    }
    if (res.status === 401 || res.status === 403) {
      return { ok: false, text: null, status: "unauthorised", httpStatus: res.status, detail: "The AI provider rejected this key." };
    }
    if (res.status === 400) {
      return { ok: false, text: null, status: "unavailable", httpStatus: 400, detail: "The provider rejected the request — check the model name." };
    }
    return { ok: false, text: null, status: "unavailable", httpStatus: res.status, detail: "AI temporarily unavailable." };
  } catch {
    return { ok: false, text: null, status: "unavailable", detail: "The AI provider could not be reached." };
  }
}

/**
 * Phase 16 — AI Chat Assistant entry point.
 *
 * Flow: authenticate → resolve tenant → resolve an ALLOWLISTED intent →
 * check the module permission that intent requires → run the permission-aware
 * `bi_kpis` function as the user (RLS applies) → minimise the result →
 * optionally add a labelled AI narrative → log the interaction.
 *
 * The model never sees SQL, schema, identifiers or personal data, and cannot
 * modify anything: there is no write path in this function.
 */
import { createServerFn } from "@tanstack/react-start";
import { requireSupabaseAuth } from "@/integrations/supabase/auth-middleware";
import { REFUSAL_MESSAGE, metricValue, resolveIntent } from "@/lib/analytics";
import { generateNarrative } from "@/lib/ai-assistant.server";

export const askAssistant = createServerFn({ method: "POST" })
  .middleware([requireSupabaseAuth])
  .inputValidator((input: { question: string; branchId?: string | null; from?: string; to?: string }) => {
    const question = String(input?.question ?? "").trim();
    if (!question) throw new Error("Ask a question first.");
    if (question.length > 500) throw new Error("Please shorten the question.");
    return {
      question,
      branchId: input.branchId ?? null,
      from: input.from ?? null,
      to: input.to ?? null,
    };
  })
  .handler(async ({ data, context }) => {
    const client = context.supabase as unknown as {
      rpc: (fn: string, args?: Record<string, unknown>) => Promise<{ data: unknown; error: { message: string } | null }>;
      from: (t: string) => any; // eslint-disable-line @typescript-eslint/no-explicit-any
    };

    const profile = await client
      .from("profiles")
      .select("organization_id")
      .eq("id", context.userId)
      .maybeSingle();
    const organizationId = profile?.data?.organization_id as string | undefined;
    if (!organizationId) throw new Error("No organisation is linked to this account.");

    const logInteraction = async (fields: Record<string, unknown>) => {
      await client.from("ai_query_logs").insert({
        organization_id: organizationId,
        user_id: context.userId,
        question: data.question,
        branch_scope: data.branchId,
        data_period_start: data.from,
        data_period_end: data.to,
        ...fields,
      });
    };

    // 1. Can this user use the assistant at all?
    const allowedAi = await client.rpc("has_permission", { _module: "ai", _action: "create" });
    const isAdmin = await client.rpc("is_org_admin");
    if (allowedAi.data !== true && isAdmin.data !== true) {
      await logInteraction({ refused: true, refusal_reason: "no_ai_permission", result_classification: "refusal" });
      return {
        refused: true as const,
        answer: "You do not have permission to use the AI assistant. Ask an administrator for the AI assistant permission.",
        classification: "refusal",
      };
    }

    // 2. Allowlisted intent only — no free-form querying is possible.
    const intent = resolveIntent(data.question);
    if (!intent) {
      await logInteraction({ refused: true, refusal_reason: "no_matching_intent", result_classification: "refusal" });
      return { refused: true as const, answer: REFUSAL_MESSAGE, classification: "refusal" };
    }

    // 3. The module permission this intent requires.
    const allowedModule = await client.rpc("has_permission", {
      _module: intent.permission.module,
      _action: intent.permission.action,
    });
    if (allowedModule.data !== true && isAdmin.data !== true) {
      await logInteraction({
        resolved_intent: intent.id,
        refused: true,
        refusal_reason: "module_permission_denied",
        result_classification: "refusal",
      });
      return {
        refused: true as const,
        answer: `That question needs ${intent.permission.module} access, which your role does not have. I cannot answer it.`,
        classification: "refusal",
      };
    }

    // 4. Scope enforcement — the assistant may only read the properties this
    //    user is actually posted to. A wider request is narrowed, never widened.
    const scopeRes = await client.rpc("current_access_scope");
    const scope = String(scopeRes.data ?? "personal");
    const branchRes = await client.rpc("current_branch_ids");
    const myBranches = Array.isArray(branchRes.data) ? (branchRes.data as string[]) : [];
    const orgWide = isAdmin.data === true || scope === "organization" || scope === "all";

    let branchScope = data.branchId;
    if (!orgWide) {
      if (myBranches.length === 0) {
        await logInteraction({
          resolved_intent: intent.id,
          refused: true,
          refusal_reason: "no_scope_assignment",
          result_classification: "refusal",
        });
        return {
          refused: true as const,
          answer:
            "Your account is not posted to a property yet, so there is no data in your scope to report on.",
          classification: "refusal",
        };
      }
      if (branchScope && !myBranches.includes(branchScope)) {
        await logInteraction({
          resolved_intent: intent.id,
          refused: true,
          refusal_reason: "branch_outside_scope",
          result_classification: "refusal",
        });
        return {
          refused: true as const,
          answer: "That property is outside your assigned scope, so I cannot report on it.",
          classification: "refusal",
        };
      }
      if (!branchScope) branchScope = myBranches[0] ?? null;
    }

    const period = {
      from: data.from ?? new Date(Date.now() - 29 * 86_400_000).toISOString().slice(0, 10),
      to: data.to ?? new Date().toISOString().slice(0, 10),
    };
    const kpis = await client.rpc("bi_kpis", {
      _branch: branchScope,
      _from: period.from,
      _to: period.to,
    });

    if (kpis.error) {
      await logInteraction({
        resolved_intent: intent.id,
        refused: true,
        refusal_reason: "analytics_unavailable",
        result_classification: "error",
      });
      throw new Error("Analytics data could not be read for your scope.");
    }

    // 5. Minimise: only the metric paths this intent declares.
    const payload = (kpis.data ?? {}) as Record<string, unknown>;
    const metrics = intent.metricPaths.map((path) => {
      const value = metricValue(payload as never, path);
      return {
        label: path.split(".").pop()!.replace(/_/g, " "),
        path,
        value: value === null ? "no recorded value" : String(value),
      };
    });

    // 6. Optional labelled narrative. Facts stand alone if AI is unavailable.
    const narrative = await generateNarrative({
      question: data.question,
      intentId: intent.id,
      period,
      metrics: metrics.map((m) => ({ label: m.label, value: m.value })),
    }, organizationId);

    await logInteraction({
      resolved_intent: intent.id,
      approved_query: `bi_kpis → ${intent.metricPaths.join(", ")}`,
      sources_used: intent.sources,
      result_classification: intent.classification,
      refused: false,
    });

    await client.from("ai_usage_records").insert({
      organization_id: organizationId,
      user_id: context.userId,
      feature: "assistant",
      model: narrative.model,
      request_count: narrative.status === "ok" ? 1 : 0,
      error: narrative.status === "ok" ? null : (narrative.detail ?? narrative.status),
      limit_status: narrative.status === "ok" ? "ok" : narrative.status,
    });

    return {
      refused: false as const,
      intentId: intent.id,
      intentQuestion: intent.question,
      classification: intent.classification,
      period,
      branchScoped: Boolean(branchScope),
      scope,


      metrics,
      sources: intent.sources,
      generatedAt: (payload["generated_at"] as string | undefined) ?? new Date().toISOString(),
      narrative: narrative.text,
      aiStatus: narrative.status,
      aiDetail: narrative.detail ?? null,
      aiProvider: narrative.provider,
      aiModel: narrative.model,
    };
  });

/**
 * Outbound delivery adapters (server only).
 *
 * Until now the communication queue looked up a provider and stopped: nothing
 * ever left the building. These adapters perform the real provider call and
 * return a truthful result — a provider reference on success, a classified
 * failure otherwise. Nothing is ever reported as sent without a provider
 * response, and a failure the provider will never accept (a bad address, a
 * rejected sender) is marked permanent so the queue does not retry it forever.
 */

export type SendOutcome =
  | { ok: true; reference: string | null; detail: string }
  | { ok: false; permanent: boolean; code: string; detail: string };

export type OutboundMessage = {
  channel: string;
  recipient: string;
  subject: string | null;
  body: string;
};

/** Per-organisation provider settings held on `provider_connections.configuration`. */
export type ProviderConfiguration = Record<string, unknown>;

/**
 * The workspace's own provider credentials, loaded server-side from
 * `provider_credentials`. Empty when the workspace has not connected its own
 * account, in which case a platform-wide key is used if one exists.
 */
export type ProviderSecrets = Record<string, string>;

function text(config: ProviderConfiguration, ...keys: string[]): string | null {
  for (const key of keys) {
    const value = config[key];
    if (typeof value === "string" && value.trim()) return value.trim();
  }
  return null;
}

function looksLikeEmail(value: string): boolean {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value.trim());
}

/** Normalises a Kenyan-style mobile number to E.164 without inventing a country. */
export function normaliseMsisdn(raw: string, defaultCountryCode: string | null): string | null {
  const trimmed = (raw ?? "").trim();
  if (!trimmed) return null;
  if (trimmed.startsWith("+")) {
    const digits = trimmed.slice(1).replace(/\D/g, "");
    return digits.length >= 9 ? `+${digits}` : null;
  }
  const digits = trimmed.replace(/\D/g, "");
  if (digits.length < 9) return null;
  if (digits.startsWith("00")) return `+${digits.slice(2)}`;
  if (!defaultCountryCode) return null;
  const cc = defaultCountryCode.replace(/\D/g, "");
  if (!cc) return null;
  return `+${cc}${digits.replace(/^0+/, "")}`;
}

/** Email through Resend. Requires RESEND_API_KEY and a configured sender. */
export async function sendEmail(
  message: OutboundMessage,
  config: ProviderConfiguration,
  secrets: ProviderSecrets = {},
): Promise<SendOutcome> {
  const apiKey = secrets["api_key"] || process.env["RESEND_API_KEY"];
  if (!apiKey) {
    return {
      ok: false,
      permanent: false,
      code: "provider_credentials_missing",
      detail: "No email provider key is available to this workspace.",
    };
  }
  const fromEmail = text(config, "from_email", "sender_email", "from");
  if (!fromEmail || !looksLikeEmail(fromEmail)) {
    return {
      ok: false,
      permanent: false,
      code: "sender_not_configured",
      detail:
        "No verified sending address is set for this workspace. Set the sender address on the email provider before messages can go out.",
    };
  }
  if (!looksLikeEmail(message.recipient)) {
    return {
      ok: false,
      permanent: true,
      code: "invalid_recipient",
      detail: `"${message.recipient}" is not a usable email address.`,
    };
  }
  const fromName = text(config, "from_name", "sender_name");
  const from = fromName ? `${fromName} <${fromEmail}>` : fromEmail;
  const replyTo = text(config, "reply_to", "reply_to_email");

  let response: Response;
  try {
    response = await fetch("https://api.resend.com/emails", {
      method: "POST",
      headers: { Authorization: `Bearer ${apiKey}`, "Content-Type": "application/json" },
      body: JSON.stringify({
        from,
        to: [message.recipient],
        subject: message.subject ?? "(no subject)",
        html: message.body,
        ...(replyTo ? { reply_to: replyTo } : {}),
      }),
    });
  } catch (error) {
    return {
      ok: false,
      permanent: false,
      code: "provider_unreachable",
      detail: error instanceof Error ? error.message : "The email provider could not be reached.",
    };
  }

  const payload = (await response.json().catch(() => null)) as
    | { id?: string; message?: string; name?: string }
    | null;
  if (!response.ok) {
    return {
      ok: false,
      // 4xx other than rate limiting will not succeed on a retry.
      permanent: response.status >= 400 && response.status < 500 && response.status !== 429,
      code: `email_${response.status}`,
      detail: payload?.message ?? `The email provider refused the message (${response.status}).`,
    };
  }
  return { ok: true, reference: payload?.id ?? null, detail: "Accepted by the email provider." };
}

/**
 * SMS through Africa's Talking. Uses the workspace's own account when a
 * manager has saved one, and otherwise falls back to a platform key.
 */
export async function sendSms(
  message: OutboundMessage,
  config: ProviderConfiguration,
  secrets: ProviderSecrets = {},
): Promise<SendOutcome> {
  const apiKey = secrets["api_key"] || process.env["AT_API_KEY"];
  const username = secrets["username"] || process.env["AT_USERNAME"];
  if (!apiKey || !username) {
    return {
      ok: false,
      permanent: false,
      code: "provider_credentials_missing",
      detail: "No SMS provider credentials are available to this workspace.",
    };
  }
  const to = normaliseMsisdn(message.recipient, text(config, "default_country_code") ?? null);
  if (!to) {
    return {
      ok: false,
      permanent: true,
      code: "invalid_recipient",
      detail: `"${message.recipient}" is not a usable mobile number in international format.`,
    };
  }
  const senderId = text(config, "sender_id", "from", "short_code");

  const host =
    username.toLowerCase() === "sandbox"
      ? "https://api.sandbox.africastalking.com"
      : "https://api.africastalking.com";

  const form = new URLSearchParams({ username, to, message: message.body });
  if (senderId) form.set("from", senderId);

  let response: Response;
  try {
    response = await fetch(`${host}/version1/messaging`, {
      method: "POST",
      headers: {
        apiKey,
        Accept: "application/json",
        "Content-Type": "application/x-www-form-urlencoded",
      },
      body: form.toString(),
    });
  } catch (error) {
    return {
      ok: false,
      permanent: false,
      code: "provider_unreachable",
      detail: error instanceof Error ? error.message : "The SMS provider could not be reached.",
    };
  }

  const payload = (await response.json().catch(() => null)) as {
    SMSMessageData?: { Message?: string; Recipients?: { status?: string; statusCode?: number; messageId?: string }[] };
  } | null;

  if (!response.ok) {
    return {
      ok: false,
      permanent: response.status >= 400 && response.status < 500 && response.status !== 429,
      code: `sms_${response.status}`,
      detail: payload?.SMSMessageData?.Message ?? `The SMS provider refused the message (${response.status}).`,
    };
  }

  const recipient = payload?.SMSMessageData?.Recipients?.[0];
  const status = recipient?.status ?? "";
  // Africa's Talking answers 200 even when it rejects an individual recipient.
  if (!recipient || !/^success$/i.test(status)) {
    const code = recipient?.statusCode ?? 0;
    return {
      ok: false,
      // 4xx-family recipient codes (invalid number, blacklisted) will not change.
      permanent: code >= 400 && code < 500,
      code: `sms_recipient_${code || "unknown"}`,
      detail: status || payload?.SMSMessageData?.Message || "The SMS provider did not accept the recipient.",
    };
  }
  return { ok: true, reference: recipient.messageId ?? null, detail: `Accepted by the SMS provider (${status}).` };
}

/**
 * WhatsApp through the Meta Cloud API. Requires WHATSAPP_TOKEN and
 * WHATSAPP_PHONE_ID. When the workspace has configured an approved template
 * name the message is sent as that template (the only way to open a new
 * conversation); otherwise it is sent as a plain text reply, which Meta only
 * accepts inside an open 24-hour customer window.
 */
export async function sendWhatsApp(
  message: OutboundMessage,
  config: ProviderConfiguration,
  secrets: ProviderSecrets = {},
): Promise<SendOutcome> {
  const token = secrets["token"] || process.env["WHATSAPP_TOKEN"];
  const phoneId = secrets["phone_id"] || process.env["WHATSAPP_PHONE_ID"];
  if (!token || !phoneId) {
    return {
      ok: false,
      permanent: false,
      code: "provider_credentials_missing",
      detail: "No WhatsApp credentials are available to this workspace.",
    };
  }
  const to = normaliseMsisdn(message.recipient, text(config, "default_country_code") ?? null);
  if (!to) {
    return {
      ok: false,
      permanent: true,
      code: "invalid_recipient",
      detail: `"${message.recipient}" is not a usable mobile number in international format.`,
    };
  }

  const templateName = text(config, "template_name", "pin_template", "default_template");
  const language = text(config, "template_language", "language") ?? "en";
  const version = text(config, "api_version") ?? "v21.0";

  const body = templateName
    ? {
        messaging_product: "whatsapp",
        to: to.replace(/^\+/, ""),
        type: "template",
        template: {
          name: templateName,
          language: { code: language },
          components: [{ type: "body", parameters: [{ type: "text", text: message.body }] }],
        },
      }
    : {
        messaging_product: "whatsapp",
        to: to.replace(/^\+/, ""),
        type: "text",
        text: { preview_url: false, body: message.body },
      };

  let response: Response;
  try {
    response = await fetch(`https://graph.facebook.com/${version}/${phoneId}/messages`, {
      method: "POST",
      headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" },
      body: JSON.stringify(body),
    });
  } catch (error) {
    return {
      ok: false,
      permanent: false,
      code: "provider_unreachable",
      detail: error instanceof Error ? error.message : "WhatsApp could not be reached.",
    };
  }

  const payload = (await response.json().catch(() => null)) as {
    messages?: { id?: string }[];
    error?: { message?: string; code?: number };
  } | null;

  if (!response.ok || payload?.error) {
    const code = payload?.error?.code ?? response.status;
    // 131047 is "no open conversation": a template is required, and retrying
    // the same plain message will never succeed.
    const permanent =
      code === 131047 || (response.status >= 400 && response.status < 500 && response.status !== 429);
    return {
      ok: false,
      permanent,
      code: `whatsapp_${code}`,
      detail:
        code === 131047 && !templateName
          ? "WhatsApp refused the message because the customer has no open conversation. Set an approved template name on the WhatsApp provider so messages like this can start one."
          : (payload?.error?.message ?? `WhatsApp refused the message (${response.status}).`),
    };
  }

  return {
    ok: true,
    reference: payload?.messages?.[0]?.id ?? null,
    detail: "Accepted by WhatsApp.",
  };
}

/** Dispatches by channel. An unsupported channel stays queued, never "sent". */
export async function dispatchMessage(
  channel: string,
  message: OutboundMessage,
  config: ProviderConfiguration,
  secrets: ProviderSecrets = {},
): Promise<SendOutcome> {
  if (channel === "email") return sendEmail(message, config, secrets);
  if (channel === "sms") return sendSms(message, config, secrets);
  if (channel === "whatsapp") return sendWhatsApp(message, config, secrets);
  return {
    ok: false,
    permanent: false,
    code: "channel_not_supported",
    detail: `Sending over ${channel} is not wired up yet; the message stays queued.`,
  };
}

/**
 * Configured communication templates → runtime sending path.
 *
 * Administration → Communication lets a hotel brand its guest messaging per
 * channel. Those edits are saved on `organization_settings.communication_templates`.
 * The sending engine, meanwhile, looked only at the `message_templates` table —
 * so a hotel could customise every template and still have the queue fall back
 * to the wording written into the workflow action. This module is the bridge:
 * it turns a configured template into the subject and body the queue sends.
 *
 * Nothing here invents wording. When the hotel has not enabled a template for
 * that key and channel, it returns null and the caller keeps its existing
 * behaviour.
 */

export type ConfiguredTemplate = {
  key: string;
  channel: string;
  subject: string;
  greeting: string;
  body: string;
  signature: string;
  enabled: boolean;
};

export type ComposedMessage = { subject: string | null; body: string };

/** Reads the stored JSON defensively — the column is hotel-editable. */
export function parseConfiguredTemplates(raw: unknown): ConfiguredTemplate[] {
  if (!Array.isArray(raw)) return [];
  const out: ConfiguredTemplate[] = [];
  for (const entry of raw) {
    if (!entry || typeof entry !== "object") continue;
    const row = entry as Record<string, unknown>;
    const key = typeof row["key"] === "string" ? row["key"] : "";
    const channel = typeof row["channel"] === "string" ? row["channel"] : "";
    const body = typeof row["body"] === "string" ? row["body"] : "";
    if (!key || !channel || !body.trim()) continue;
    out.push({
      key,
      channel,
      subject: typeof row["subject"] === "string" ? row["subject"] : "",
      greeting: typeof row["greeting"] === "string" ? row["greeting"] : "",
      body,
      signature: typeof row["signature"] === "string" ? row["signature"] : "",
      enabled: row["enabled"] !== false,
    });
  }
  return out;
}

export function findConfiguredTemplate(
  templates: ConfiguredTemplate[],
  key: string,
  channel: string,
): ConfiguredTemplate | null {
  if (!key || !channel) return null;
  return templates.find((t) => t.key === key && t.channel === channel && t.enabled) ?? null;
}

/**
 * Greeting, body and sign-off become the message the recipient reads. A
 * channel without subjects (SMS, WhatsApp) keeps a null subject rather than an
 * empty one, so the queue does not show a blank subject line.
 */
export function composeConfiguredMessage(
  template: ConfiguredTemplate,
  render: (text: string) => string,
): ComposedMessage {
  const parts = [template.greeting, template.body, template.signature]
    .map((part) => (part ?? "").trim())
    .filter(Boolean)
    .map(render);
  const subject = template.subject.trim() ? render(template.subject.trim()) : null;
  return { subject, body: parts.join("\n\n") };
}

const HOSTS = {
  sandbox: "https://sandbox.safaricom.co.ke",
  production: "https://api.safaricom.co.ke",
} as const;

export function darajaHost(environment: string): string {
  return environment === "production" ? HOSTS.production : HOSTS.sandbox;
}

/** Daraja expects 2547XXXXXXXX — normalise anything a cashier might type. */
export function normaliseMsisdn(raw: string): string | null {
  const digits = String(raw ?? "").replace(/[^0-9]/g, "");
  if (!digits) return null;
  if (digits.startsWith("254") && digits.length === 12) return digits;
  if (digits.startsWith("0") && digits.length === 10) return `254${digits.slice(1)}`;
  if (digits.length === 9) return `254${digits}`;
  if (digits.startsWith("254")) return digits;
  return null;
}

export function darajaTimestamp(now = new Date()): string {
  const p = (n: number) => String(n).padStart(2, "0");
  return (
    String(now.getUTCFullYear()) +
    p(now.getUTCMonth() + 1) +
    p(now.getUTCDate()) +
    p(now.getUTCHours()) +
    p(now.getUTCMinutes()) +
    p(now.getUTCSeconds())
  );
}

export function stkPassword(shortCode: string, passkey: string, timestamp: string): string {
  return btoa(`${shortCode}${passkey}${timestamp}`);
}

/** Loads tenant credentials with the trusted client; secrets stay server-side. */

export async function darajaToken(creds: MpesaCredentials): Promise<DarajaResult<string>> {
  const auth = btoa(`${creds.consumerKey}:${creds.consumerSecret}`);
  try {
    const res = await fetch(`${darajaHost(creds.environment)}/oauth/v1/generate?grant_type=client_credentials`, {
      headers: { Authorization: `Basic ${auth}` },
    });
    const text = await res.text();
    if (!res.ok) return { ok: false, status: `http-${res.status}`, detail: text.slice(0, 400) };
    const parsed = JSON.parse(text) as { access_token?: string };
    if (!parsed.access_token) return { ok: false, status: "no-token", detail: text.slice(0, 400) };
    return { ok: true, data: parsed.access_token };
  } catch (error) {
    return { ok: false, status: "network", detail: (error as Error).message };
  }
}

export type StkPushResult = {
  MerchantRequestID?: string;
  CheckoutRequestID?: string;
  ResponseCode?: string;
  ResponseDescription?: string;
  CustomerMessage?: string;
  errorCode?: string;
  errorMessage?: string;
};

export async function stkPush(
  creds: MpesaCredentials,
  input: { msisdn: string; amount: number; accountReference: string; description: string; callbackUrl: string },
): Promise<DarajaResult<StkPushResult & { request: Record<string, unknown> }>> {
  const tokenResult = await darajaToken(creds);
  if (!tokenResult.ok) return tokenResult;

  const timestamp = darajaTimestamp();
  const body = {
    BusinessShortCode: creds.shortCode,
    Password: stkPassword(creds.shortCode, creds.passkey, timestamp),
    Timestamp: timestamp,
    TransactionType: creds.transactionType,
    Amount: Math.max(1, Math.round(input.amount)),
    PartyA: input.msisdn,
    PartyB: creds.partyB,
    PhoneNumber: input.msisdn,
    CallBackURL: input.callbackUrl,
    AccountReference: input.accountReference.slice(0, 12) || "NEO",
    TransactionDesc: input.description.slice(0, 60) || "Payment",
  };

  try {
    const res = await fetch(`${darajaHost(creds.environment)}/mpesa/stkpush/v1/processrequest`, {
      method: "POST",
      headers: { Authorization: `Bearer ${tokenResult.data}`, "Content-Type": "application/json" },
      body: JSON.stringify(body),
    });
    const text = await res.text();
    let parsed: StkPushResult = {};
    try {
      parsed = JSON.parse(text) as StkPushResult;
    } catch {
      return { ok: false, status: `http-${res.status}`, detail: text.slice(0, 400) };
    }
    if (!res.ok || parsed.ResponseCode !== "0") {
      return {
        ok: false,
        status: `http-${res.status}`,
        detail: parsed.errorMessage ?? parsed.ResponseDescription ?? text.slice(0, 400),
      };
    }
    const safeRequest = { ...body, Password: "[redacted]" } as Record<string, unknown>;
    return { ok: true, data: { ...parsed, request: safeRequest } };
  } catch (error) {
    return { ok: false, status: "network", detail: (error as Error).message };
  }
}

    }

/** Extracts the fields NEØ stores from an STK callback body. */
export function readStkCallback(payload: unknown): {
  checkoutRequestId: string | null;
  merchantRequestId: string | null;
  resultCode: number | null;
  resultDesc: string;
  receipt: string | null;
  transactionDate: string | null;
  amount: number | null;
} {
  const root = (payload ?? {}) as Record<string, any>; // eslint-disable-line @typescript-eslint/no-explicit-any
  const cb = root["Body"]?.["stkCallback"] ?? root["stkCallback"] ?? root;
  const items: Array<{ Name?: string; Value?: unknown }> = cb?.["CallbackMetadata"]?.["Item"] ?? [];
  const pick = (name: string) => items.find((i) => i?.Name === name)?.Value ?? null;
  const receipt = pick("MpesaReceiptNumber");
  const txnDate = pick("TransactionDate");
  const amount = pick("Amount");
  const code = cb?.["ResultCode"];
  return {
    checkoutRequestId: cb?.["CheckoutRequestID"] ? String(cb["CheckoutRequestID"]) : null,
    merchantRequestId: cb?.["MerchantRequestID"] ? String(cb["MerchantRequestID"]) : null,
    resultCode: code === null || code === undefined ? null : Number(code),
    resultDesc: String(cb?.["ResultDesc"] ?? ""),
    receipt: receipt ? String(receipt) : null,
    transactionDate: txnDate ? String(txnDate) : null,
    amount: amount === null || amount === undefined ? null : Number(amount),
  };
}

async function baseUrl(fromSettings: string | null | undefined): Promise<string> {

  .middleware([requireSupabaseAuth])
  .handler(async ({ context }): Promise<MpesaSettings> => {

    const { data, error } = await client.rpc("neo_mpesa_settings_get");
    if (error) throw new Error(error.message);

    const base = await baseUrl(settings.callback_base_url);

      callback_temporary: temporary,
      // Four distinct things, never collapsed into one "connected" badge.
      status: !credentialsStored

          ? "configured"
          : settings.environment === "production" && settings.enabled && configuredBase && !temporary
            ? "production-ready"
            : "tested",

        tested &&
        settings.environment === "production" &&
        Boolean(settings.enabled) &&

export const saveMpesaSettings = createServerFn({ method: "POST" })
  .middleware([requireSupabaseAuth])
  .inputValidator((input: {
    environment: string;
    shortCode?: string | null;
    partyB?: string | null;
    accountReference?: string | null;
    transactionType?: string | null;
    consumerKey?: string | null;
    consumerSecret?: string | null;
    passkey?: string | null;

    partyB: input?.partyB ? String(input.partyB).trim() : null,
    accountReference: input?.accountReference ? String(input.accountReference).trim() : null,
    transactionType: input?.transactionType ? String(input.transactionType).trim() : "CustomerPayBillOnline",
    consumerKey: input?.consumerKey ? String(input.consumerKey).trim() : null,

    // show up later as confirmations that never arrive.

    // Live collections may not depend on a guessed or temporary address.
    let enabled = data.enabled;
    let enableBlocked: string | null = null;
    if (enabled && data.environment === "production") {
      if (!callback.value) {
        enableBlocked =
          "Enter the public callback address of your live site before switching production collections on.";
      } else if (isTemporaryHost(callback.value)) {
        enableBlocked =
          "That callback address belongs to a temporary preview site. Use your published address for live collections.";
      }
      if (enableBlocked) enabled = false;
    }

    const { data: result, error } = await client.rpc("neo_mpesa_settings_save", {
      _environment: data.environment,
      _short_code: data.shortCode,
      _party_b: data.partyB,
      _account_reference: data.accountReference,
      _transaction_type: data.transactionType,
      _consumer_key: data.consumerKey,
      _consumer_secret: data.consumerSecret,
      _passkey: data.passkey,
      _callback_base_url: callback.value,
      _enabled: enabled,
      _clear_secrets: data.clearSecrets,
      _rotate_token: data.rotateToken,
    });
    if (error) throw new Error(error.message);
    const saved = (result ?? { authorised: false }) as SaveResult;
    if (enableBlocked && saved.saved) {
      return { ...saved, enabled: false, enable_blocked: true, reason: enableBlocked };
    }
    return saved;
  });

/**
 * Sends an STK prompt for a restaurant bill or a guest folio. The pending
 * payment is created by the module's own routine, so balance and authority
 * rules are unchanged; only confirmation is delegated to the provider.
 */
export const chargeWithMpesa = createServerFn({ method: "POST" })
  .middleware([requireSupabaseAuth])
  .inputValidator((input: StkChargeInput) => ({
    sourceModule: input?.sourceModule === "folio" ? ("folio" as const) : ("pos_order" as const),
    sourceRecordId: String(input?.sourceRecordId ?? "").trim(),
    msisdn: String(input?.msisdn ?? "").trim(),
    amount: Number(input?.amount ?? 0),
    description: input?.description ? String(input.description).trim().slice(0, 60) : null,
  }))
  .handler(async ({ data, context }): Promise<StkChargeResult> => {
    const client = context.supabase as unknown as Rpc;
    if (!data.sourceRecordId) return { ok: false, status: "invalid", detail: "Nothing to charge." };
    if (!(data.amount > 0)) return { ok: false, status: "invalid", detail: "Enter an amount greater than zero." };

    const { loadMpesaCredentials, normaliseMsisdn, stkPush } = await import("@/lib/mpesa.server");
    const msisdn = normaliseMsisdn(data.msisdn);
    if (!msisdn) return { ok: false, status: "invalid", detail: "Enter a valid Kenyan mobile number." };

    const organizationId = await organisationOf(client, context.userId);
    if (!organizationId) return { ok: false, status: "no-organisation", detail: "No organisation is linked to this account." };

    const creds = await loadMpesaCredentials(organizationId);
    if (!creds || !creds.enabled) {
      return { ok: false, status: "not-configured", detail: "M-Pesa is not configured or is switched off for this workspace." };
    }
    const base = await baseUrl(creds.callbackBaseUrl);
    if (!base) {
      return { ok: false, status: "no-callback", detail: "Set the public callback base URL in M-Pesa settings first." };
    }

    // 1. Pending payment through the module's existing authority checks.
    const providerReference = `STK-${crypto.randomUUID().slice(0, 12).toUpperCase()}`;
    let paymentId: string | null = null;
    let paymentTable: "pos_payments" | "payments";
    let branchId: string | null = null;
    let accountReference = creds.accountReference ?? "NEO";

    if (data.sourceModule === "pos_order") {
      paymentTable = "pos_payments";
      const order = await client.from("pos_orders").select("id, branch_id, order_number").eq("id", data.sourceRecordId).maybeSingle();
      if (!order?.data) return { ok: false, status: "not-found", detail: "Bill not found." };
      branchId = (order.data.branch_id as string | null) ?? null;
      accountReference = String(order.data.order_number ?? accountReference);
      const { data: attempt, error } = await client.rpc("pos_record_payment_attempt", {
        _order_id: data.sourceRecordId,
        _amount: data.amount,
        _method: "mpesa",
        _tendered: null,
        _reference: providerReference,
        _shift_id: null,
        _idempotency_key: `mpesa:${providerReference}`,
        _manual_confirm: false,
        _note: null,
      });
      if (error) return { ok: false, status: "rejected", detail: error.message };
      paymentId = (attempt?.payment_id as string | undefined) ?? null;
    } else {
      paymentTable = "payments";
      const folio = await client.from("folios").select("id, branch_id, folio_number").eq("id", data.sourceRecordId).maybeSingle();
      if (!folio?.data) return { ok: false, status: "not-found", detail: "Folio not found." };
      branchId = (folio.data.branch_id as string | null) ?? null;
      accountReference = String(folio.data.folio_number ?? accountReference);
      const { data: payment, error } = await client.rpc("pms_record_payment", {
        _folio: data.sourceRecordId,
        _amount: data.amount,
        _method: "mpesa",
        _reference: providerReference,
        _notes: "M-Pesa STK prompt sent to guest",
        // The provider reference is the collection intent: a re-sent prompt
        // reuses the same receipt instead of opening a second one.
        _intent_key: `mpesa:${providerReference}`,
      });
      if (error) return { ok: false, status: "rejected", detail: error.message };
      paymentId = (payment as string | null) ?? null;
    }

    // 2. Ledger row for the prompt.
    const { data: transactionId, error: txnError } = await client.rpc("neo_mpesa_txn_start", {
      _source_module: data.sourceModule,
      _source_record: data.sourceRecordId,
      _payment_table: paymentTable,
      _payment_id: paymentId,
      _branch: branchId,
      _msisdn: msisdn,
      _amount: data.amount,
      _account_reference: accountReference,
      _description: data.description ?? "NEO payment",
    });
    if (txnError) return { ok: false, status: "rejected", detail: txnError.message };

    // 3. Push the prompt, then record what the provider said.
    const { supabaseAdmin: admin } = await import("@/integrations/supabase/client.server");
    const supabaseAdmin = admin as unknown as Rpc;
    const push = await stkPush(creds, {
      msisdn,
      amount: data.amount,
      accountReference,
      description: data.description ?? "NEO payment",
      callbackUrl: callbackUrl(base, creds.callbackToken),
    });

    if (!push.ok) {
      await supabaseAdmin.rpc("neo_mpesa_apply_result", {
        _transaction: transactionId,
        _result_code: -1,
        _result_desc: push.detail,
        _receipt: null,
        _transaction_date: null,
        _payload: { error: push.detail, status: push.status },
        _source: "push-error",
      });
      return { ok: false, status: push.status, detail: push.detail, transactionId: transactionId as string };
    }

    await supabaseAdmin
      .from("mpesa_transactions")
      .update({
        status: "pending",
        merchant_request_id: push.data.MerchantRequestID ?? null,
        checkout_request_id: push.data.CheckoutRequestID ?? null,
        raw_request: push.data.request,
        result_desc: push.data.CustomerMessage ?? push.data.ResponseDescription ?? null,
        updated_at: new Date().toISOString(),
      })
      .eq("id", transactionId);

    return {
      ok: true,
      status: "pending",
      detail: push.data.CustomerMessage ?? "Prompt sent. Ask the customer to enter their M-Pesa PIN.",
      transactionId: transactionId as string,
      checkoutRequestId: push.data.CheckoutRequestID ?? undefined,
      customerMessage: push.data.CustomerMessage ?? undefined,
    };
  });

/** Asks Daraja for the current state of a prompt and applies the outcome. */
export const reconcileMpesaTransaction = createServerFn({ method: "POST" })
  .middleware([requireSupabaseAuth])
  .inputValidator((input: { transactionId: string }) => ({ transactionId: String(input?.transactionId ?? "").trim() }))
  .handler(async ({ data, context }) => {
    const client = context.supabase as unknown as Rpc;
    if (!data.transactionId) return { ok: false, status: "invalid", detail: "No transaction supplied." };

    const existing = await client.from("mpesa_transactions").select("*").eq("id", data.transactionId).maybeSingle();
    const row = existing?.data as Record<string, any> | null;
    if (!row) return { ok: false, status: "not-found", detail: "Transaction not visible to you." };
    if (!(await client.rpc("neo_mpesa_can_collect", { _source_module: row["source_module"] })).data) {
      return { ok: false, status: "unauthorised", detail: "You cannot reconcile this payment." };
    }
    if (row["applied_at"]) {
      return { ok: true, status: row["status"] as string, detail: "Already reconciled.", duplicate: true };
    }
    if (!row["checkout_request_id"]) {
      return { ok: false, status: "no-reference", detail: "The prompt never reached Safaricom, so there is nothing to query." };
    }

    const { loadMpesaCredentials, stkQuery } = await import("@/lib/mpesa.server");
    const creds = await loadMpesaCredentials(row["organization_id"] as string);
    if (!creds) return { ok: false, status: "not-configured", detail: "M-Pesa credentials are incomplete." };

    const query = await stkQuery(creds, String(row["checkout_request_id"]));
    const { supabaseAdmin: admin } = await import("@/integrations/supabase/client.server");
    const supabaseAdmin = admin as unknown as Rpc;
    await supabaseAdmin
      .from("mpesa_transactions")
      .update({ last_query_at: new Date().toISOString() })
      .eq("id", data.transactionId);

    if (!query.ok) return { ok: false, status: query.status, detail: query.detail };

    const code = query.data.ResultCode === undefined ? null : Number(query.data.ResultCode);
    if (code === null) {
      return { ok: false, status: "unknown", detail: query.data.ResultDesc ?? "Safaricom returned no result code yet." };
    }
    // 1032/1037 and friends are terminal; anything else non-zero is a decline.
    const { data: applied, error } = await supabaseAdmin.rpc("neo_mpesa_apply_result", {
      _transaction: data.transactionId,
      _result_code: code,
      _result_desc: query.data.ResultDesc ?? "",
      _receipt: null,
      _transaction_date: null,
      _payload: query.data as unknown as Record<string, unknown>,
      _source: "status-query",
    });
    if (error) return { ok: false, status: "apply-failed", detail: error.message };
    return { ok: true, status: (applied?.status as string) ?? "unknown", detail: query.data.ResultDesc ?? "" };
  });

/** Reads the tenant's eTIMS credentials. Returns null when setup is incomplete. */
export async function loadEtimsCredentials(organizationId: string): Promise<EtimsCredentials | null> {
  const { supabaseAdmin } = await import("@/integrations/supabase/client.server");
  const { data, error } = await (supabaseAdmin as any)
    .from("etims_settings")
    .select("*")
    .eq("organization_id", organizationId)
    .maybeSingle();
  if (error || !data) return null;

  const baseUrl = String(data.base_url ?? "").trim().replace(/\/+$/, "");
  const kraPin = String(data.kra_pin ?? "").trim();
  const deviceSerial = String(data.device_serial ?? "").trim();
  const cmcKey = String(data.cmc_key ?? "").trim();
  if (!baseUrl || !kraPin || !deviceSerial || !cmcKey) return null;

  return {
    organizationId,
    environment: data.environment === "production" ? "production" : "sandbox",
    mode: data.mode === "vscu" ? "vscu" : "oscu",
    baseUrl,
    kraPin,
    branchCode: String(data.branch_code ?? "00").trim() || "00",
    deviceSerial,
    cmcKey,
    apiKey: data.api_key ? String(data.api_key) : null,
    enabled: Boolean(data.enabled),
    autoSubmit: Boolean(data.auto_submit),
  };
}

function headers(creds: EtimsCredentials): Record<string, string> {
  const base: Record<string, string> = {
    "Content-Type": "application/json",
    tin: creds.kraPin,
    bhfId: creds.branchCode,
    cmcKey: creds.cmcKey,
  };
  if (creds.apiKey) base["Authorization"] = `Bearer ${creds.apiKey}`;
  return base;
}

function stamp(date: Date): string {
  const p = (n: number) => String(n).padStart(2, "0");
  return `${date.getUTCFullYear()}${p(date.getUTCMonth() + 1)}${p(date.getUTCDate())}${p(date.getUTCHours())}${p(
    date.getUTCMinutes(),
  )}${p(date.getUTCSeconds())}`;
}

/** Maps a NEØ tax rate onto the KRA tax band letters. */
export function taxBand(rate: number): "A" | "B" | "C" | "E" {
  if (rate >= 15.5) return "B"; // standard rate (16%)
  if (rate > 0) return "E"; // reduced/other rated
  return "A"; // exempt or zero rated
}

/** Builds the KRA sales payload for a document. */
export function buildSalesPayload(creds: EtimsCredentials, doc: EtimsDocument): Record<string, unknown> {
  const now = new Date();

async function post(creds: EtimsCredentials, path: string, body: Record<string, unknown>) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), 25000);
  try {
    const response = await fetch(`${creds.baseUrl}${path}`, {
      method: "POST",
      headers: headers(creds),
      body: JSON.stringify(body),
      signal: controller.signal,
    });
    const text = await response.text();
    let parsed: any = null;
    try {
      parsed = text ? JSON.parse(text) : null;
    } catch {
      parsed = { raw: text.slice(0, 2000) };
    }
    return { httpStatus: response.status, ok: response.ok, body: parsed };
  } finally {
    clearTimeout(timer);
  }
}

/** Confirms the device is initialised and reachable, without filing anything. */
export async function verifyDevice(creds: EtimsCredentials): Promise<{ ok: boolean; detail: string }> {
  try {
    const result = await post(creds, "/selectInitOsdcInfo", {
      tin: creds.kraPin,
      bhfId: creds.branchCode,
      dvcSrlNo: creds.deviceSerial,
    });
    const code = String(result.body?.resultCd ?? "");
    if (!result.ok) {
      return { ok: false, detail: `KRA responded ${result.httpStatus}: ${JSON.stringify(result.body).slice(0, 400)}` };
    }
    // 000 = success, 902 = already initialised — both prove the device is known.
    if (code === "000" || code === "902") {
      return { ok: true, detail: `Device recognised by KRA (${creds.environment}, ${creds.mode.toUpperCase()}).` };
    }
    return { ok: false, detail: `KRA rejected the device: ${result.body?.resultMsg ?? code}` };
  } catch (error) {
    return { ok: false, detail: `Could not reach the eTIMS endpoint: ${(error as Error).message}` };
  }
}

/** Files a sales document and interprets the KRA outcome. */
export async function submitDocument(creds: EtimsCredentials, doc: EtimsDocument): Promise<EtimsResult> {
  const payload = buildSalesPayload(creds, doc);
  try {
    const result = await post(creds, "/insertTrnsSalesOsdc", payload);
    const code = String(result.body?.resultCd ?? "");
    const message = String(result.body?.resultMsg ?? "");
    const data = result.body?.data ?? {};

    if (!result.ok) {
      return {
        ok: false,
        status: result.httpStatus >= 500 ? "retry_required" : "failed",
        detail: `KRA responded ${result.httpStatus}: ${message || JSON.stringify(result.body).slice(0, 300)}`,
        request: payload,
        response: result.body ?? {},
      };
    }

    if (code === "000") {
      return {
        ok: true,
        status: "accepted",
        detail: message || "Accepted by KRA.",
        kraReference: data.rcptNo ? String(data.rcptNo) : (data.curRcptNo ? String(data.curRcptNo) : null),
        controlUnitReference: data.intrlData ? String(data.intrlData) : null,
        qrPayload: data.rcptSign ? String(data.rcptSign) : (data.qrCodeUrl ? String(data.qrCodeUrl) : null),
        request: payload,
        response: result.body ?? {},
      };
    }

    return {
      ok: false,
      status: "rejected",
      detail: message ? `${code}: ${message}` : `KRA rejected the document (${code}).`,
      request: payload,
      response: result.body ?? {},
    };
  } catch (error) {
    return {
      ok: false,
      status: "retry_required",
      detail: `Could not reach the eTIMS endpoint: ${(error as Error).message}`,
      request: payload,
      response: {},
    };
  }
}

/**
 * Queues a document through the guarded routine, then files it with KRA and
 * records the outcome. Safe to call twice: the queue routine is idempotent per
 * document number and an already accepted document is never re-filed.
 */
export const fileTaxDocument = createServerFn({ method: "POST" })
  .middleware([requireSupabaseAuth])
  .inputValidator((input: FileInput) => ({
    invoiceId: input?.invoiceId ? String(input.invoiceId) : null,
    creditNoteId: input?.creditNoteId ? String(input.creditNoteId) : null,
  }))
  .handler(async ({ data, context }) => {
    const client = context.supabase as unknown as Rpc;
    if (!data.invoiceId && !data.creditNoteId) {
      return { ok: false, status: "failed", detail: "Nothing to file." };
    }

    const { data: queued, error: queueError } = await client.rpc("neo_etims_queue_document", {
      _invoice: data.invoiceId,
      _credit_note: data.creditNoteId,
    });
    if (queueError) throw new Error(queueError.message);
    if (!queued?.queued) {
      return { ok: false, status: "failed", detail: queued?.reason ?? "The document could not be queued." };
    }

    const submissionId = String(queued.submission);

    // Already accepted by KRA — filing again would duplicate the fiscal receipt.
    const existing = await client
      .from("etims_submissions")
      .select("status, kra_reference")
      .eq("id", submissionId)
      .maybeSingle();
    if (existing?.data?.status === "accepted") {
      return {
        ok: true,
        status: "accepted",
        submissionId,
        detail: `Already filed with KRA${existing.data.kra_reference ? ` (${existing.data.kra_reference})` : ""}.`,
      };
    }

    const organizationId = await organisationOf(client, context.userId);
    if (!organizationId) {
      return { ok: false, status: "failed", submissionId, detail: "No organisation is linked to this account." };
    }

    const { loadEtimsCredentials, submitDocument } = await import("@/lib/etims.server");
    const creds = await loadEtimsCredentials(organizationId);
    const { supabaseAdmin: admin } = await import("@/integrations/supabase/client.server");
    const supabaseAdmin = admin as unknown as Rpc;

    if (!creds || !creds.enabled) {
      const detail = "eTIMS credentials are incomplete or filing is switched off.";
      await supabaseAdmin.rpc("neo_etims_apply_result", {
        _submission: submissionId,
        _status: "failed",
        _error: detail,
        _response: { reason: "not-configured" },
      });
      return { ok: false, status: "failed", submissionId, detail };
    }

    // Assemble the document from server-side records only.
    const doc = await buildDocument(supabaseAdmin, submissionId);
    if (!doc) {
      const detail = "The document could not be assembled for filing.";
      await supabaseAdmin.rpc("neo_etims_apply_result", {
        _submission: submissionId,
        _status: "failed",
        _error: detail,
      });
      return { ok: false, status: "failed", submissionId, detail };
    }

    const result = await submitDocument(creds, doc);
    await supabaseAdmin.rpc("neo_etims_apply_result", {
      _submission: submissionId,
      _status: result.status,
      _kra_reference: result.kraReference ?? null,
      _control_unit_reference: result.controlUnitReference ?? null,
      _qr_payload: result.qrPayload ?? null,
      _error: result.ok ? null : result.detail,
      _request: result.request ?? {},
      _response: result.response ?? {},
    });

    return { ok: result.ok, status: result.status, submissionId, detail: result.detail };
  });

/** Reads the invoice or credit note behind a submission and shapes it for KRA. */
async function buildDocument(admin: Rpc, submissionId: string) {

        taxRate: Number(l.tax_rate ?? 0),
        taxAmount: Number(l.tax_amount ?? 0),

    .from("credit_notes")
    .select("id, note_number, issue_date, currency, subtotal, tax_total, total, reason")
    .eq("id", submission.credit_note_id)

  return {

/**
 * Card terminal / QR gateway hand-off.
 *
 * NEØ speaks one contract to any processor, so a terminal supplier or QR app can
 * be plugged in with a URL and a key rather than a code change:
 *
 *   POST <endpoint_url>
 *   Authorization: Bearer <secret>            (only when a secret name is configured)
 *   {
 *     "intent_id":    "<uuid>",               our reference for this request
 *     "amount":       1500.00,
 *     "currency":     "KES",
 *     "reference":    "REQ-8A21C4F0",
 *     "method":       "card-terminal" | "qr",
 *     "terminal_id":  "<from configuration, optional>",
 *     "merchant_id":  "<from configuration, optional>",
 *     "callback_url": "https://<this site>/api/public/payments/callback/<token>",
 *     "environment":  "sandbox" | "production"
 *   }
 *
 * The processor may answer straight away:
 *   { "status": "approved" | "declined" | "cancelled" | "pending",
 *     "reference": "<their own reference>", "message": "..." }
 * or answer later by calling the callback_url with the same shape. Either way the
 * money is only ever settled by the processor's own answer — never by NEØ
 * assuming the request succeeded.
 *
 * Nothing here invents a processor. With no connected, enabled integration the
 * request is refused with the exact reason, and the checkout says so.
 */
import { createServerFn } from "@tanstack/react-start";
import { getRequest } from "@tanstack/react-start/server";
import { requireSupabaseAuth } from "@/integrations/supabase/auth-middleware";

export type GatewayDispatch = {
  ok: boolean;
  reason?: string;
  detail?: string;
  state?: string;
  providerReference?: string | null;
  /** What the guest scans, when a QR provider answered with a code to show. */
  qrPayload?: string | null;
  qrImageUrl?: string | null;
};

type AttemptRow = {
  id: string;
  organization_id: string;
  branch_id: string | null;
  method: string;
  provider_code: string | null;
  amount: number;
  currency: string | null;
  reference: string | null;
  state: string;
  environment: string | null;
  callback_token: string | null;
};

type ConnectionRow = {
  enabled: boolean;
  status: string;
  environment: string | null;
  configuration: Record<string, unknown> | null;
  secret_names: unknown;
};

const TERMINAL_OUTCOMES = new Set(["approved", "declined", "cancelled", "timeout", "reversed"]);

function callbackBase(): string | null {
  const configured = process.env["PUBLIC_SITE_URL"];
  if (configured) return configured.replace(/\/$/, "");
  const origin = getRequest()?.headers.get("origin") ?? getRequest()?.headers.get("host");
  if (!origin) return null;
  return origin.startsWith("http") ? origin.replace(/\/$/, "") : `https://${origin}`;
}

type RpcClient = { rpc: (fn: string, args?: Record<string, unknown>) => Promise<unknown> };

/** Records "sent to the processor" or "could not be sent", never an outcome. */
async function markSent(
  client: unknown,
  intentId: string,
  opts: { reference?: string | undefined; detail?: string | undefined; failed: boolean },
) {
  await (client as RpcClient).rpc("neo_payment_intent_mark_sent", {
    _intent: intentId,
    ...(opts.reference ? { _provider_reference: opts.reference } : {}),
    ...(opts.detail ? { _detail: opts.detail } : {}),
    _failed: opts.failed,
  });
}

export const dispatchPaymentIntent = createServerFn({ method: "POST" })
  .middleware([requireSupabaseAuth])
  .inputValidator((input: { intentId: string }) => {
    if (!input?.intentId) throw new Error("An intent is required");
    return input;
  })
  .handler(async ({ data, context }): Promise<GatewayDispatch> => {
    const { supabase } = context;

    const { data: attempt } = await supabase
      .from("payment_attempts")
      .select(
        "id, organization_id, branch_id, method, provider_code, amount, currency, reference, state, environment, callback_token",
      )
      .eq("payment_table", "payment_intents")
      .eq("payment_id", data.intentId)
      .maybeSingle();

    const row = attempt as AttemptRow | null;
    if (!row) return { ok: false, reason: "intent_not_found" };
    if (row.state !== "requested") {
      return { ok: true, state: row.state, detail: "This request has already been sent." };
    }
    if (!row.provider_code) return { ok: false, reason: "not_provider_backed" };

    const { data: conn } = await supabase
      .from("integration_connections")
      .select("enabled, status, environment, configuration, secret_names")
      .eq("provider_code", row.provider_code)
      .order("branch_id", { ascending: true, nullsFirst: false })
      .limit(1)
      .maybeSingle();

    const connection = conn as ConnectionRow | null;
    if (!connection || !connection.enabled || connection.status !== "connected") {
      return {
        ok: false,
        reason: "provider_not_connected",
        detail: "No processor is connected for this method, so nothing can be sent.",
      };
    }

    const config = (connection.configuration ?? {}) as Record<string, unknown>;
    const endpoint = String(config["endpoint_url"] ?? "").trim();
    if (!endpoint.startsWith("https://")) {
      return {
        ok: false,
        reason: "endpoint_missing",
        detail: "The processor has no secure request address configured yet.",
      };
    }

    const base = callbackBase();
    if (!base || !row.callback_token) {
      return { ok: false, reason: "callback_unavailable", detail: "This request has no answer address." };
    }

    const secretName = String(config["secret_name"] ?? "").trim();
    const { resolveConnectorSecret } = await import("@/lib/connector-secret.server");
    const resolvedSecret = resolveConnectorSecret(secretName);
    if (resolvedSecret.status === "invalid") {
      return {
        ok: false,
        reason: "secret_name_not_allowed",
        detail: "Connector secret names must start with NEO_CONNECTOR_.",
      };
    }
    const secret = resolvedSecret.value;
    if (resolvedSecret.status === "missing") {
      return {
        ok: false,
        reason: "secret_missing",
        detail: `The processor key ${secretName} has not been saved for this workspace.`,
      };
    }

    const payload = {
      intent_id: data.intentId,
      amount: Number(row.amount),
      currency: row.currency ?? "KES",
      reference: row.reference,
      method: row.method,
      terminal_id: config["terminal_id"] ?? null,
      merchant_id: config["merchant_id"] ?? null,
      callback_url: `${base}/api/public/payments/callback/${row.callback_token}`,
      environment: row.environment ?? connection.environment ?? "sandbox",
    };

    let answer: {
      status?: string;
      reference?: string;
      message?: string;
      qr_payload?: string;
      qr_string?: string;
      qr_image_url?: string;
    } = {};
    try {
      const res = await fetch(endpoint, {
        method: "POST",
        headers: {
          "content-type": "application/json",
          ...(secret ? { authorization: `Bearer ${secret}` } : {}),
        },
        body: JSON.stringify(payload),
        signal: AbortSignal.timeout(20_000),
      });
      const text = await res.text();
      try {
        answer = text ? (JSON.parse(text) as typeof answer) : {};
      } catch {
        answer = {};
      }
      if (!res.ok) {
        await markSent(supabase, data.intentId, {
          reference: answer.reference,
          detail: `The processor refused the request (${res.status}).`,
          failed: true,
        });
        return { ok: false, reason: "provider_rejected", detail: `Processor returned ${res.status}.`, state: "failed" };
      }
    } catch (e) {
      const detail = e instanceof Error ? e.message : "The processor could not be reached.";
      await markSent(supabase, data.intentId, { detail, failed: true });
      return { ok: false, reason: "provider_unreachable", detail, state: "failed" };
    }

    const status = String(answer.status ?? "pending").toLowerCase();

    // Some terminals answer on the spot. That answer is the processor's own, so
    // it settles the payment through the same path the webhook uses.
    if (TERMINAL_OUTCOMES.has(status)) {
      const { supabaseAdmin } = await import("@/integrations/supabase/client.server");
      const admin = supabaseAdmin as unknown as {
        rpc: (fn: string, args?: Record<string, unknown>) => Promise<{ data: unknown; error: unknown }>;
      };
      await admin.rpc("neo_payment_intent_apply", {
        _token: row.callback_token,
        _outcome: status,
        _provider_reference: answer.reference ?? undefined,
        _detail: answer.message ?? null,
        _payload: answer as Record<string, unknown>,
      });
      return { ok: true, state: status, providerReference: answer.reference ?? null };
    }

    await markSent(supabase, data.intentId, {
      reference: answer.reference,
      detail: answer.message,
      failed: false,
    });

    // A QR provider answers with the code the guest scans. It is kept on the
    // attempt so the till can show it again after a refresh, and it is never
    // treated as payment — only the webhook confirms money.
    const qrPayload = answer.qr_payload ?? answer.qr_string ?? null;
    const qrImageUrl = answer.qr_image_url ?? null;
    if (qrPayload || qrImageUrl) {
      await supabase.rpc("neo_payment_intent_set_provider_payload", {
        _intent: data.intentId,
        _payload: { qr_payload: qrPayload, qr_image_url: qrImageUrl, reference: answer.reference ?? null },
      });
    }

    return {
      ok: true,
      state: "pending",
      providerReference: answer.reference ?? null,
      qrPayload,
      qrImageUrl,
    };
  });

/**
 * Courier dispatch hand-off.
 *
 * NEØ speaks one contract to any courier service — the same shape the card
 * terminal and QR processors answer on — so a courier can be plugged in with a
 * URL and a key rather than a code change:
 *
 *   POST <endpoint_url>
 *   Authorization: Bearer <secret>            (only when a secret name is configured)
 *   {
 *     "job_id":       "<uuid>",               our reference for this request
 *     "order_number": "DL-000123",
 *     "contact":      { "name": "...", "phone": "..." },
 *     "address":      { ... },                the address snapshot on the order
 *     "amount":       1500.00,
 *     "currency":     "KES",
 *     "payment":      { "method": "cash", "status": "pending" },
 *     "callback_url": "https://<this site>/api/public/delivery/callback/<token>",
 *     "environment":  "sandbox" | "production"
 *   }
 *
 * The courier may answer straight away:
 *   { "status": "accepted" | "picked_up" | "delivered" | "failed" | "cancelled",
 *     "reference": "<their own reference>", "message": "..." }
 * or answer later by calling the callback_url with the same shape. Either way the
 * order only moves on the courier's own answer — never because NEØ sent a request.
 *
 * Nothing here invents a courier. With no connected, enabled integration the
 * request is refused with the exact reason, and the dispatch board says so.
 */
import { createServerFn } from "@tanstack/react-start";
import { getRequest } from "@tanstack/react-start/server";
import { requireSupabaseAuth } from "@/integrations/supabase/auth-middleware";

export type CourierDispatchResult = {
  ok: boolean;
  reason?: string;
  detail?: string;
  state?: string;
  providerReference?: string | null;
};

type JobRow = {
  id: string;
  branch_id: string | null;
  state: string;
  environment: string | null;
  callback_token: string | null;
  request_payload: Record<string, unknown> | null;
  provider_code: string;
};

type ConnectionRow = {
  enabled: boolean;
  status: string;
  environment: string | null;
  configuration: Record<string, unknown> | null;
};

const COURIER_OUTCOMES = new Set(["accepted", "assigned", "picked_up", "collected", "delivered", "failed", "declined", "rejected", "cancelled"]);

function callbackBase(): string | null {
  const configured = process.env["PUBLIC_SITE_URL"];
  if (configured) return configured.replace(/\/$/, "");
  const origin = getRequest()?.headers.get("origin") ?? getRequest()?.headers.get("host");
  if (!origin) return null;
  return origin.startsWith("http") ? origin.replace(/\/$/, "") : `https://${origin}`;
}

type RpcClient = { rpc: (fn: string, args?: Record<string, unknown>) => Promise<unknown> };

async function markSent(
  client: unknown,
  jobId: string,
  opts: { reference?: string | undefined; detail?: string | undefined; failed: boolean },
) {
  await (client as RpcClient).rpc("neo_delivery_dispatch_mark_sent", {
    _job: jobId,
    ...(opts.reference ? { _provider_reference: opts.reference } : {}),
    ...(opts.detail ? { _detail: opts.detail } : {}),
    _failed: opts.failed,
  });
}

export const dispatchCourierJob = createServerFn({ method: "POST" })
  .middleware([requireSupabaseAuth])
  .inputValidator((input: { jobId: string }) => {
    if (!input?.jobId) throw new Error("A dispatch job is required");
    return input;
  })
  .handler(async ({ data, context }): Promise<CourierDispatchResult> => {
    const { supabase } = context;

    const { data: found } = await supabase
      .from("delivery_dispatch_jobs")
      .select("id, branch_id, state, environment, callback_token, request_payload, provider_code")
      .eq("id", data.jobId)
      .maybeSingle();

    const job = found as JobRow | null;
    if (!job) return { ok: false, reason: "job_not_found" };
    if (job.state !== "requested") {
      return { ok: true, state: job.state, detail: "This dispatch has already been sent." };
    }

    const { data: conn } = await supabase
      .from("integration_connections")
      .select("enabled, status, environment, configuration")
      .eq("provider_code", job.provider_code)
      .order("branch_id", { ascending: true, nullsFirst: false })
      .limit(1)
      .maybeSingle();

    const connection = conn as ConnectionRow | null;
    if (!connection || !connection.enabled || connection.status !== "connected") {
      return {
        ok: false,
        reason: "provider_not_connected",
        detail: "No courier service is connected, so nothing can be sent.",
      };
    }

    const config = (connection.configuration ?? {}) as Record<string, unknown>;
    const endpoint = String(config["endpoint_url"] ?? "").trim();
    if (!endpoint.startsWith("https://")) {
      return {
        ok: false,
        reason: "endpoint_missing",
        detail: "The courier has no secure request address configured yet.",
      };
    }

    const base = callbackBase();
    if (!base || !job.callback_token) {
      return { ok: false, reason: "callback_unavailable", detail: "This dispatch has no answer address." };
    }

    const secretName = String(config["secret_name"] ?? "").trim();
    const { resolveConnectorSecret } = await import("@/lib/connector-secret.server");
    const resolvedSecret = resolveConnectorSecret(secretName);
    if (resolvedSecret.status === "invalid") {
      return {
        ok: false,
        reason: "secret_name_not_allowed",
        detail: "Connector secret names must start with NEO_CONNECTOR_.",
      };
    }
    const secret = resolvedSecret.value;
    if (resolvedSecret.status === "missing") {
      return {
        ok: false,
        reason: "secret_missing",
        detail: `The courier key ${secretName} has not been saved for this workspace.`,
      };
    }

    const req = (job.request_payload ?? {}) as Record<string, unknown>;
    const payload = {
      job_id: job.id,
      order_number: req["order_number"] ?? null,
      contact: { name: req["contact_name"] ?? null, phone: req["contact_phone"] ?? null },
      address: req["address"] ?? null,
      latitude: req["latitude"] ?? null,
      longitude: req["longitude"] ?? null,
      amount: Number(req["grand_total"] ?? 0),
      currency: req["currency"] ?? "KES",
      payment: { method: req["payment_method"] ?? null, status: req["payment_status"] ?? null },
      merchant_id: config["merchant_id"] ?? null,
      callback_url: `${base}/api/public/delivery/callback/${job.callback_token}`,
      environment: job.environment ?? connection.environment ?? "sandbox",
    };

    let answer: { status?: string; reference?: string; message?: string } = {};
    try {
      const res = await fetch(endpoint, {
        method: "POST",
        headers: {
          "content-type": "application/json",
          ...(secret ? { authorization: `Bearer ${secret}` } : {}),
        },
        body: JSON.stringify(payload),
        signal: AbortSignal.timeout(20_000),
      });
      const text = await res.text();
      try {
        answer = text ? (JSON.parse(text) as typeof answer) : {};
      } catch {
        answer = {};
      }
      if (!res.ok) {
        await markSent(supabase, job.id, {
          reference: answer.reference,
          detail: `The courier refused the request (${res.status}).`,
          failed: true,
        });
        return { ok: false, reason: "provider_rejected", detail: `Courier returned ${res.status}.`, state: "failed" };
      }
    } catch (e) {
      const detail = e instanceof Error ? e.message : "The courier could not be reached.";
      await markSent(supabase, job.id, { detail, failed: true });
      return { ok: false, reason: "provider_unreachable", detail, state: "failed" };
    }

    const status = String(answer.status ?? "pending").toLowerCase();

    // A courier that answers on the spot is still the courier answering, so that
    // answer travels the same path as the webhook rather than a shortcut.
    if (COURIER_OUTCOMES.has(status)) {
      const { supabaseAdmin } = await import("@/integrations/supabase/client.server");
      const admin = supabaseAdmin as unknown as {
        rpc: (fn: string, args?: Record<string, unknown>) => Promise<{ data: unknown; error: unknown }>;
      };
      await admin.rpc("neo_delivery_dispatch_apply", {
        _token: job.callback_token,
        _outcome: status,
        _provider_reference: answer.reference ?? undefined,
        _detail: answer.message ?? null,
        _payload: answer as Record<string, unknown>,
      });
      return { ok: true, state: status, providerReference: answer.reference ?? null };
    }

    await markSent(supabase, job.id, {
      reference: answer.reference,
      detail: answer.message,
      failed: false,
    });
    return { ok: true, state: "sent", providerReference: answer.reference ?? null };
  });

/**
 * Workspace-owned messaging credentials (server only).
 *
 * A manager connects their own Africa's Talking or WhatsApp Cloud account and
 * the values are written to `provider_credentials`, a table the Data API cannot
 * read. Nothing here is ever returned to the browser: callers get a masked
 * summary, never the stored value.
 */

type Admin = {
  from: (table: string) => {
    select: (columns: string) => {
      eq: (column: string, value: string) => Promise<{ data: unknown; error: { message: string } | null }>;
    };
  };
};

export type ProviderSecretMap = Record<string, Record<string, string>>;

/** Field names a workspace may store per provider. Anything else is ignored. */
export const CREDENTIAL_FIELDS: Record<string, string[]> = {
  africastalking: ["username", "api_key"],
  whatsapp: ["token", "phone_id"],
  resend: ["api_key"],
};

/** Shows that a value exists without revealing it. */
export function maskCredential(value: string): string {
  const trimmed = value.trim();
  if (trimmed.length <= 4) return "••••";
  return `••••${trimmed.slice(-4)}`;
}

/** Loads every stored credential set for an organisation, keyed by provider. */
export async function loadProviderSecrets(admin: unknown, organizationId: string): Promise<ProviderSecretMap> {
  const client = admin as Admin;
  const { data, error } = await client
    .from("provider_credentials")
    .select("provider, credentials")
    .eq("organization_id", organizationId);
  if (error) throw new Error(error.message);
  const rows = (data ?? []) as { provider: string; credentials: Record<string, unknown> | null }[];
  const map: ProviderSecretMap = {};
  for (const row of rows) {
    const entries: Record<string, string> = {};
    for (const [key, value] of Object.entries(row.credentials ?? {})) {
      if (typeof value === "string" && value.trim()) entries[key] = value.trim();
    }
    map[row.provider] = entries;
  }
  return map;
}

/** True when the workspace (or the platform) can actually reach the provider. */
export function providerHasCredentials(
  provider: string,
  secrets: ProviderSecretMap,
  envNames: string[],
): { ready: boolean; source: "workspace" | "platform" | "none"; missing: string[] } {
  const fields = CREDENTIAL_FIELDS[provider] ?? [];
  const held = secrets[provider] ?? {};
  const missingWorkspace = fields.filter((field) => !held[field]);
  if (fields.length && missingWorkspace.length === 0) {
    return { ready: true, source: "workspace", missing: [] };
  }
  const missingEnv = envNames.filter((name) => !process.env[name]);
  if (envNames.length && missingEnv.length === 0) {
    return { ready: true, source: "platform", missing: [] };
  }
  // Report the workspace gap: that is the one a manager can close themselves.
  return {
    ready: false,
    source: "none",
    missing: fields.length ? missingWorkspace : missingEnv,
  };
}

/**
 * Server-only resolver for generic connector secrets.
 *
 * A workspace can choose which connector-specific secret name a gateway uses,
 * but it must never be able to name arbitrary host environment variables such
 * as SUPABASE_SERVICE_ROLE_KEY. Generic connector secrets therefore live in a
 * dedicated NEO_CONNECTOR_* namespace.
 */

const CONNECTOR_SECRET_NAME = /^NEO_CONNECTOR_[A-Z0-9_]{1,80}$/;

export type ConnectorSecretResolution =
  | { status: "none"; value: undefined }
  | { status: "invalid"; value: undefined }
  | { status: "missing"; value: undefined }
  | { status: "ready"; value: string };

export function resolveConnectorSecret(name: string | null | undefined): ConnectorSecretResolution {
  const normalized = String(name ?? "").trim();
  if (!normalized) return { status: "none", value: undefined };
  if (!CONNECTOR_SECRET_NAME.test(normalized)) {
    return { status: "invalid", value: undefined };
  }
  const value = process.env[normalized]?.trim();
  if (!value) return { status: "missing", value: undefined };
  return { status: "ready", value };
}

export function isConnectorSecretName(name: string): boolean {
  const normalized = name.trim();
  return normalized.length === 0 || CONNECTOR_SECRET_NAME.test(normalized);
}

/**
 * Phase 6 — workflow engine (server only).
 *
 * The engine consumes durable `domain_events` rows written by database
 * triggers on real operational records, matches them against published
 * workflow versions, and performs the configured actions exactly once using
 * `automation_idempotency_keys`.
 *
 * It never invents operational data: messages are queued, and they only leave
 * the queue when a provider connection is genuinely configured.
 */
import type { SupabaseClient } from "@supabase/supabase-js";
import type { Database } from "@/integrations/supabase/types";
import {
  composeConfiguredMessage,
  findConfiguredTemplate,
  parseConfiguredTemplates,
  type ComposedMessage,
} from "@/lib/configured-communication-templates";

type Admin = SupabaseClient<Database>;

type EventRow = Database["public"]["Tables"]["domain_events"]["Row"];
type WorkflowRow = Database["public"]["Tables"]["workflow_definitions"]["Row"];

export type EngineMode = "live" | "test";

export type ActionOutcome = {
  index: number;
  type: string;
  status: "completed" | "skipped" | "failed" | "simulated";
  detail: string;
  result?: Record<string, string>;
};

type Condition = { field: string; operator: string; value: string };
type Action = { type: string; delay_minutes?: number; config: Record<string, unknown> };

function readField(event: EventRow, field: string): unknown {
  if (field.startsWith("payload.")) {
    const payload = (event.payload ?? {}) as Record<string, unknown>;
    return payload[field.slice("payload.".length)];
  }
  return (event as unknown as Record<string, unknown>)[field];
}

export function evaluateConditions(event: EventRow, conditions: Condition[]): boolean {
  return conditions.every((c) => {
    const actual = readField(event, c.field);
    const expected = c.value;
    switch (c.operator) {
      case "eq":
        return String(actual ?? "") === expected;
      case "neq":
        return String(actual ?? "") !== expected;
      case "gt":
        return Number(actual) > Number(expected);
      case "gte":
        return Number(actual) >= Number(expected);
      case "lt":
        return Number(actual) < Number(expected);
      case "lte":
        return Number(actual) <= Number(expected);
      case "in":
        return expected.split(",").map((v) => v.trim()).includes(String(actual ?? ""));
      case "contains":
        return String(actual ?? "").toLowerCase().includes(expected.toLowerCase());
      case "exists":
        return actual !== null && actual !== undefined && actual !== "";
      default:
        return false;
    }
  });
}

/** Returns true when the key was newly claimed (i.e. the action may proceed). */
async function claimIdempotency(admin: Admin, org: string, scope: string, key: string) {
  const { error } = await admin
    .from("automation_idempotency_keys")
    .insert({ organization_id: org, scope, idempotency_key: key });
  if (error) return false;
  return true;
}

async function resolveRecipients(
  admin: Admin,
  org: string,
  action: Action,
): Promise<{ userIds: string[]; note: string }> {
  const config = action.config ?? {};
  if (action.type === "notify_user" && typeof config["user_id"] === "string") {
    return { userIds: [config["user_id"] as string], note: "specific user" };
  }
  if (action.type === "notify_role" && typeof config["role_id"] === "string") {
    const { data } = await admin.from("user_roles").select("user_id").eq("role_id", config["role_id"] as string);
    return { userIds: (data ?? []).map((r) => r.user_id), note: "role members" };
  }
  if (action.type === "notify_department" && typeof config["department_id"] === "string") {
    const { data } = await admin
      .from("profiles")
      .select("id")
      .eq("organization_id", org)
      .eq("department_id", config["department_id"] as string);
    return { userIds: (data ?? []).map((r) => r.id), note: "department members" };
  }
  return { userIds: [], note: "no recipient rule matched" };
}

function interpolate(text: string, event: EventRow): string {
  const payload = (event.payload ?? {}) as Record<string, unknown>;
  return text.replace(/{{\s*([a-zA-Z0-9_.]+)\s*}}/g, (_, name: string) => {
    const key = name.startsWith("payload.") ? name.slice(8) : name;
    const value = payload[key];
    return value === undefined || value === null ? `[${key}]` : String(value);
  });
}

async function executeAction(
  admin: Admin,
  event: EventRow,
  workflow: WorkflowRow,
  runId: string,
  action: Action,
  index: number,
  mode: EngineMode,
): Promise<ActionOutcome> {
  const org = event.organization_id;
  const config = (action.config ?? {}) as Record<string, unknown>;
  const idemKey = `${workflow.id}:${event.id}:${index}`;
  const str = (k: string, fallback = "") => (typeof config[k] === "string" ? (config[k] as string) : fallback);

  if (mode === "test") {
    return {
      index,
      type: action.type,
      status: "simulated",
      detail: `Test mode — would run "${action.type}". No customer contacted, no record changed.`,
    };
  }

  if (action.delay_minutes && action.delay_minutes > 0) {
    const claimed = await claimIdempotency(admin, org, "schedule", idemKey);
    if (!claimed) return { index, type: action.type, status: "skipped", detail: "Already scheduled" };
    await admin.from("scheduled_actions").insert({
      organization_id: org,
      branch_id: event.branch_id,
      workflow_id: workflow.id,
      run_id: runId,
      event_id: event.id,
      action_type: action.type,
      action_config: config as never,
      context: { event_type: event.event_type, payload: event.payload } as never,
      scheduled_for: new Date(Date.now() + action.delay_minutes * 60_000).toISOString(),
      idempotency_key: idemKey,
    });
    return { index, type: action.type, status: "completed", detail: `Scheduled in ${action.delay_minutes} min` };
  }

  const claimed = await claimIdempotency(admin, org, action.type, idemKey);
  if (!claimed) return { index, type: action.type, status: "skipped", detail: "Duplicate suppressed" };

  switch (action.type) {
    case "notify_role":
    case "notify_user":
    case "notify_department": {
      const { userIds, note } = await resolveRecipients(admin, org, action);
      if (!userIds.length) {
        return { index, type: action.type, status: "failed", detail: `No recipient resolved (${note})` };
      }
      const rows = userIds.map((uid) => ({
        organization_id: org,
        user_id: uid,
        branch_id: event.branch_id,
        department_id: (config["department_id"] as string) ?? null,
        category: str("category", "automation"),
        title: interpolate(str("title", workflow.name), event),
        body: interpolate(str("body", `${event.event_type} occurred.`), event),
        link: str("link") || null,
        priority: str("priority", "normal"),
      }));
      const { error } = await admin.from("notifications").insert(rows);
      if (error) throw new Error(error.message);
      return { index, type: action.type, status: "completed", detail: `Notified ${rows.length} recipient(s)` };
    }

    case "create_task": {
      const dueMinutes = Number(config["due_in_minutes"] ?? 120);
      const { data, error } = await admin
        .from("tasks")
        .insert({
          organization_id: org,
          branch_id: event.branch_id,
          department_id: (config["department_id"] as string) ?? null,
          station_id: (config["station_id"] as string) ?? null,
          reference: `AUTO-${event.id.slice(0, 8)}-${index}`,
          title: interpolate(str("title", `Automated task: ${workflow.name}`), event),
          description: interpolate(str("description", `Created by automation from ${event.event_type}.`), event),
          assignee_id: (config["assignee_id"] as string) ?? null,
          due_at: new Date(Date.now() + dueMinutes * 60_000).toISOString(),
          priority: str("priority", "normal"),
          status: "open",
        })
        .select("id, assignee_id, department_id")
        .single();
      if (error) throw new Error(error.message);
      if (!data.assignee_id) {
        // No eligible assignee: leave it in the department queue and tell the supervisor.
        const { data: dept } = await admin
          .from("departments")
          .select("manager_id")
          .eq("id", data.department_id ?? "")
          .maybeSingle();
        if (dept?.manager_id) {
          await admin.from("notifications").insert({
            organization_id: org,
            user_id: dept.manager_id,
            branch_id: event.branch_id,
            category: "automation",
            title: "Unassigned automated task",
            body: `A task created by "${workflow.name}" has no eligible assignee and is waiting in the department queue.`,
            link: "/tasks",
            priority: "high",
          });
        }
      }
      return { index, type: action.type, status: "completed", detail: "Task created", result: { task_id: data.id } };
    }

    case "create_approval": {
      const requesterId = (config["requester_id"] as string) ?? event.created_by;
      if (!requesterId) {
        return { index, type: action.type, status: "failed", detail: "No requester could be resolved for the approval" };
      }
      const { data, error } = await admin
        .from("approval_requests")
        .insert({
          organization_id: org,
          branch_id: event.branch_id,
          department_id: (config["department_id"] as string) ?? null,
          reference: `AUTO-APR-${event.id.slice(0, 8)}`,
          approval_type: str("approval_type", "general"),
          requester_id: requesterId,
          reason: interpolate(str("reason", `Raised automatically by ${workflow.name}`), event),
          amount: config["amount"] ? Number(config["amount"]) : null,
          status: "pending_approval",
        })
        .select("id")
        .single();
      if (error) throw new Error(error.message);
      return { index, type: action.type, status: "completed", detail: "Approval requested", result: { approval_id: data.id } };
    }

    case "queue_message":
    case "feedback_invitation": {
      if (!event.customer_id) {
        return { index, type: action.type, status: "failed", detail: "No customer linked to this event" };
      }
      const { data: customer } = await admin
        .from("customers")
        .select("id, full_name, email, phone, marketing_consent, preferred_language")
        .eq("id", event.customer_id)
        .maybeSingle();
      if (!customer) return { index, type: action.type, status: "failed", detail: "Customer record not found" };

      const channel = str("channel", "email");
      const isMarketing = Boolean(config["is_marketing"]);
      const recipient = channel === "email" ? (customer.email ?? "") : (customer.phone ?? "");
      if (!recipient) {
        return { index, type: action.type, status: "failed", detail: `Customer has no ${channel} contact` };
      }
      if (isMarketing && !customer.marketing_consent) {
        await admin.from("communication_queue").insert({
          organization_id: org,
          branch_id: event.branch_id,
          customer_id: customer.id,
          channel,
          recipient,
          template_key: str("template_key"),
          is_marketing: true,
          consent_verified: false,
          status: "suppressed",
          suppression_reason: "unsubscribed",
          workflow_id: workflow.id,
          run_id: runId,
          source_module: event.source_module,
          idempotency_key: idemKey,
        });
        return { index, type: action.type, status: "skipped", detail: "Marketing consent not given — suppressed" };
      }

      const { data: template } = await admin
        .from("message_templates")
        .select("id, subject, body, status, is_marketing")
        .eq("organization_id", org)
        .eq("key", str("template_key"))
        .eq("channel", channel)
        .maybeSingle();

      // A hotel that branded this message in Administration → Communication
      // expects that wording to be what leaves the building. Without this the
      // queue silently fell back to the wording typed into the workflow action
      // and the customised template was never used by anything.
      let configured: ComposedMessage | null = null;
      if (!template) {
        const { data: settings } = await admin
          .from("organization_settings")
          .select("communication_templates")
          .eq("organization_id", org)
          .maybeSingle();
        const match = findConfiguredTemplate(
          parseConfiguredTemplates((settings as { communication_templates?: unknown } | null)?.communication_templates),
          str("template_key"),
          channel,
        );
        if (match) configured = composeConfiguredMessage(match, (text) => interpolate(text, event));
      }

      const { error } = await admin.from("communication_queue").insert({
        organization_id: org,
        branch_id: event.branch_id,
        customer_id: customer.id,
        template_id: template?.id ?? null,
        template_key: str("template_key"),
        channel,
        recipient,
        subject: template
          ? interpolate(template.subject, event)
          : (configured?.subject ?? str("subject")),
        rendered_body: template
          ? interpolate(template.body, event)
          : (configured?.body ?? interpolate(str("body"), event)),
        variables: (event.payload ?? {}) as never,
        is_marketing: isMarketing,
        consent_verified: !isMarketing || Boolean(customer.marketing_consent),
        scheduled_for: new Date().toISOString(),
        // A template the hotel enabled in Administration → Communication is an
        // approved template; it queues like any other rather than sitting in draft.
        status: (template && template.status === "active") || configured ? "queued" : "draft",
        workflow_id: workflow.id,
        run_id: runId,
        source_module: event.source_module,
        source_record_id: event.source_record_id,
        idempotency_key: idemKey,
      });
      if (error) throw new Error(error.message);

      if (action.type === "feedback_invitation") {
        await admin.from("feedback_invitations").insert({
          organization_id: org,
          branch_id: event.branch_id,
          customer_id: customer.id,
          source_module: event.source_module,
          source_record_id: event.source_record_id,
          channel,
          status: "scheduled",
          expires_at: new Date(Date.now() + 14 * 86_400_000).toISOString(),
          idempotency_key: idemKey,
        });
      }
      return {
        index,
        type: action.type,
        status: "completed",
        detail: template
          ? "Message queued from approved template"
          : configured
            ? "Message queued from the hotel's configured communication template"
            : "Queued as draft — no active template",
      };
    }

    case "crm_timeline": {
      if (!event.customer_id) return { index, type: action.type, status: "skipped", detail: "No customer on event" };
      const { error } = await admin.from("customer_timeline_events").insert({
        organization_id: org,
        customer_id: event.customer_id,
        branch_id: event.branch_id,
        event_type: event.event_type,
        description: interpolate(str("description", `${event.event_type} recorded automatically.`), event),
        source_module: event.source_module,
        source_record_id: event.source_record_id,
        customer_visible: Boolean(config["customer_visible"]),
      });
      if (error) throw new Error(error.message);
      return { index, type: action.type, status: "completed", detail: "Timeline event added" };
    }

    case "record_communication": {
      if (!event.customer_id) return { index, type: action.type, status: "skipped", detail: "No customer on event" };
      const { error } = await admin.from("customer_communications").insert({
        organization_id: org,
        branch_id: event.branch_id,
        customer_id: event.customer_id,
        channel: str("channel", "in-system"),
        direction: "outgoing",
        category: config["is_marketing"] ? "marketing" : "service",
        subject: interpolate(str("subject", workflow.name), event),
        summary: interpolate(str("summary", ""), event),
        delivery_status: "logged",
        is_marketing: Boolean(config["is_marketing"]),
        consent_verified: true,
        customer_visible: true,
        related_module: event.source_module,
      });
      if (error) throw new Error(error.message);
      return { index, type: action.type, status: "completed", detail: "Communication recorded" };
    }

    case "escalate": {
      const { error } = await admin.from("escalation_events").insert({
        organization_id: org,
        branch_id: event.branch_id,
        subject_type: event.source_module,
        subject_id: event.source_record_id ?? event.id,
        level: Number(config["level"] ?? 1),
        reason: interpolate(str("reason", `Escalated by ${workflow.name}`), event),
        notified_user_id: (config["user_id"] as string) ?? null,
        idempotency_key: idemKey,
      });
      if (error) throw new Error(error.message);
      return { index, type: action.type, status: "completed", detail: "Escalation raised" };
    }

    case "schedule_action":
    default:
      return { index, type: action.type, status: "skipped", detail: `Unsupported action "${action.type}"` };
  }
}

export type EngineReport = {
  eventsProcessed: number;
  runsCreated: number;
  actionsCompleted: number;
  failures: number;
  details: { event: string; workflow: string; outcomes: ActionOutcome[] }[];
};

export async function processDomainEvents(admin: Admin, org: string, limit = 25): Promise<EngineReport> {
  const report: EngineReport = { eventsProcessed: 0, runsCreated: 0, actionsCompleted: 0, failures: 0, details: [] };

  const { data: settings } = await admin
    .from("automation_settings")
    .select("automation_enabled")
    .eq("organization_id", org)
    .maybeSingle();
  if (settings && settings.automation_enabled === false) return report;

  const { data: events } = await admin
    .from("domain_events")
    .select("*")
    .eq("organization_id", org)
    .in("status", ["pending", "retrying"])
    .order("occurred_at", { ascending: true })
    .limit(limit);

  const { data: workflows } = await admin
    .from("workflow_definitions")
    .select("*")
    .eq("organization_id", org)
    .eq("status", "active");

  for (const event of events ?? []) {
    report.eventsProcessed += 1;
    const matching = (workflows ?? []).filter(
      (w) => w.trigger_event === event.event_type && (!w.branch_id || w.branch_id === event.branch_id),
    );

    let eventFailed = false;

    for (const workflow of matching) {
      const { data: version } = await admin
        .from("workflow_versions")
        .select("*")
        .eq("workflow_id", workflow.id)
        .eq("version", workflow.published_version ?? workflow.current_version)
        .maybeSingle();
      if (!version) continue;

      const conditions = (version.conditions ?? []) as unknown as Condition[];
      const actions = (version.actions ?? []) as unknown as Action[];
      const matched = evaluateConditions(event, conditions);

      const { data: run } = await admin
        .from("workflow_runs")
        .insert({
          organization_id: org,
          branch_id: event.branch_id,
          workflow_id: workflow.id,
          workflow_version: version.version,
          event_id: event.id,
          correlation_id: event.correlation_id,
          mode: "live",
          status: matched ? "running" : "skipped",
          matched,
          actions_total: matched ? actions.length : 0,
        })
        .select("id")
        .single();
      if (!run) continue;
      report.runsCreated += 1;
      if (!matched) {
        await admin
          .from("workflow_runs")
          .update({ status: "skipped", finished_at: new Date().toISOString() })
          .eq("id", run.id);
        continue;
      }

      const outcomes: ActionOutcome[] = [];
      let failed = false;

      for (let i = 0; i < actions.length; i += 1) {
        const action = actions[i] as Action;
        try {
          const outcome = await executeAction(admin, event, workflow, run.id, action, i, "live");
          outcomes.push(outcome);
          if (outcome.status === "completed") report.actionsCompleted += 1;
          if (outcome.status === "failed") failed = true;
          await admin.from("workflow_run_steps").insert({
            organization_id: org,
            run_id: run.id,
            step_index: i,
            action_type: action.type,
            status: outcome.status,
            detail: outcome.detail,
            result: (outcome.result ?? {}) as never,
            finished_at: new Date().toISOString(),
          });
        } catch (error) {
          failed = true;
          const message = error instanceof Error ? error.message : "Unknown error";
          outcomes.push({ index: i, type: action.type, status: "failed", detail: message });
          await admin.from("workflow_run_steps").insert({
            organization_id: org,
            run_id: run.id,
            step_index: i,
            action_type: action.type,
            status: "failed",
            detail: message,
            error_message: message,
            finished_at: new Date().toISOString(),
          });
          await admin.from("workflow_failures").insert({
            organization_id: org,
            branch_id: event.branch_id,
            workflow_id: workflow.id,
            run_id: run.id,
            event_id: event.id,
            step_index: i,
            error_category: "action_error",
            error_message: message,
            recommended_action: "Review the action configuration, then retry from Failed Workflows.",
          });
          break;
        }
      }

      await admin
        .from("workflow_runs")
        .update({
          status: failed ? "failed" : "completed",
          actions_completed: outcomes.filter((o) => o.status === "completed").length,
          finished_at: new Date().toISOString(),
        })
        .eq("id", run.id);

      await admin
        .from("workflow_definitions")
        .update({
          last_run_at: new Date().toISOString(),
          successful_runs: workflow.successful_runs + (failed ? 0 : 1),
          failed_runs: workflow.failed_runs + (failed ? 1 : 0),
        })
        .eq("id", workflow.id);

      if (failed) {
        report.failures += 1;
        eventFailed = true;
      }
      report.details.push({ event: event.event_type, workflow: workflow.name, outcomes });
    }

    await admin
      .from("domain_events")
      .update({
        status: eventFailed ? "failed" : "processed",
        processed_at: new Date().toISOString(),
        retry_count: event.retry_count + (eventFailed ? 1 : 0),
      })
      .eq("id", event.id);
  }

  return report;
}

/** Runs due scheduled actions by re-materialising them as live actions. */
export async function processScheduledActions(admin: Admin, org: string, limit = 25) {
  const { data: due } = await admin
    .from("scheduled_actions")
    .select("*")
    .eq("organization_id", org)
    .eq("status", "scheduled")
    .lte("scheduled_for", new Date().toISOString())
    .limit(limit);

  let processed = 0;
  for (const item of due ?? []) {
    if (!item.event_id || !item.workflow_id) {
      await admin.from("scheduled_actions").update({ status: "cancelled", last_error: "Missing source context" }).eq("id", item.id);
      continue;
    }
    const [{ data: event }, { data: workflow }] = await Promise.all([
      admin.from("domain_events").select("*").eq("id", item.event_id).maybeSingle(),
      admin.from("workflow_definitions").select("*").eq("id", item.workflow_id).maybeSingle(),
    ]);
    if (!event || !workflow) {
      await admin.from("scheduled_actions").update({ status: "cancelled", last_error: "Source no longer available" }).eq("id", item.id);
      continue;
    }
    try {
      const outcome = await executeAction(
        admin,
        event,
        workflow,
        item.run_id ?? "",
        { type: item.action_type, config: (item.action_config ?? {}) as Record<string, unknown> },
        999,
        "live",
      );
      await admin
        .from("scheduled_actions")
        .update({ status: outcome.status === "failed" ? "failed" : "completed", attempts: item.attempts + 1, last_error: outcome.status === "failed" ? outcome.detail : null })
        .eq("id", item.id);
      processed += 1;
    } catch (error) {
      await admin
        .from("scheduled_actions")
        .update({
          status: item.attempts >= 2 ? "failed" : "scheduled",
          attempts: item.attempts + 1,
          last_error: error instanceof Error ? error.message : "Unknown error",
        })
        .eq("id", item.id);
    }
  }
  return processed;
}

/**
 * Attempts delivery for queued messages.
 *
 * A message is only marked sent when the provider itself accepted it and
 * returned a reference; every attempt — accepted, refused or blocked — is
 * written to `communication_delivery_events` so the trail is auditable. A
 * refusal the provider will never accept (a malformed address, a rejected
 * sender) is failed permanently instead of being retried indefinitely.
 */
export async function processCommunicationQueue(admin: Admin, org: string, limit = 25) {
  // The live queue has a historical backlog. Releasing it when credentials
  // appear would send stale stock reports and delivery codes to real people.
  // Outbound delivery stays off until an operator sets a reviewed cutover time.
  const cutoff = process.env["NEO_OUTBOUND_CUTOFF"] ?? "";
  if (process.env["NEO_OUTBOUND_ENABLED"] !== "true" || !cutoff ||
      !Number.isFinite(Date.parse(cutoff))) {
    return { queued: 0, blocked: 0, sent: 0, failed: 0, disabled: true };
  }
  const { dispatchMessage } = await import("@/lib/communication-providers.server");
  const { loadProviderSecrets } = await import("@/lib/provider-credentials.server");

  const { data: providers } = await admin
    .from("provider_connections")
    .select("provider, channel, status, enabled, configuration")
    .eq("organization_id", org);

  // The workspace's own provider accounts, when a manager has connected them.
  const secrets = await loadProviderSecrets(admin, org);

  const { data: queued } = await admin
    .from("communication_queue")
    .select("*")
    .eq("organization_id", org)
    .eq("status", "queued")
    .gte("created_at", new Date(cutoff).toISOString())
    .lte("scheduled_for", new Date().toISOString())
    .limit(limit);

  let blocked = 0;
  let sent = 0;
  let failed = 0;

  const logEvent = async (
    messageId: string,
    provider: string,
    eventType: string,
    detail: string,
    reference?: string | null,
  ) => {
    await admin.from("communication_delivery_events").insert({
      organization_id: org,
      communication_queue_id: messageId,
      provider,
      event_type: eventType,
      provider_event_id: reference ?? null,
      detail,
      occurred_at: new Date().toISOString(),
    });
  };

  for (const message of queued ?? []) {
    const now = new Date().toISOString();
    const provider = (providers ?? []).find(
      (p) => p.channel === message.channel && p.enabled && p.status === "connected",
    );

    if (!provider) {
      blocked += 1;
      await admin
        .from("communication_queue")
        .update({
          failure_code: "provider_not_configured",
          failure_message: `No connected ${message.channel} provider. The message stays queued until a provider is connected.`,
          last_attempt_at: now,
        })
        .eq("id", message.id);
      continue;
    }

    if (!message.recipient || !message.rendered_body) {
      failed += 1;
      await admin
        .from("communication_queue")
        .update({
          status: "failed",
          failure_code: "message_incomplete",
          failure_message: "The message has no recipient or no content to send.",
          last_attempt_at: now,
          attempt_count: (message.attempt_count ?? 0) + 1,
        })
        .eq("id", message.id);
      await logEvent(message.id, provider.provider, "failed", "The message has no recipient or no content to send.");
      continue;
    }

    const attempt = (message.attempt_count ?? 0) + 1;
    const maxAttempts = message.max_attempts ?? 3;

    const outcome = await dispatchMessage(
      message.channel,
      {
        channel: message.channel,
        recipient: message.recipient,
        subject: message.subject,
        body: message.rendered_body,
      },
      (provider.configuration ?? {}) as Record<string, unknown>,
      secrets[provider.provider] ?? {},
    );

    if (outcome.ok) {
      sent += 1;
      await admin
        .from("communication_queue")
        .update({
          status: "sent",
          provider: provider.provider,
          provider_reference: outcome.reference,
          failure_code: null,
          failure_message: null,
          attempt_count: attempt,
          last_attempt_at: now,
        })
        .eq("id", message.id);
      await logEvent(message.id, provider.provider, "sent", outcome.detail, outcome.reference);
      continue;
    }

    const giveUp = outcome.permanent || attempt >= maxAttempts;
    // A missing key or sender is a configuration gap, not a message defect:
    // it stays queued so it goes out once the workspace is configured.
    const configurationGap =
      outcome.code === "provider_credentials_missing" ||
      outcome.code === "sender_not_configured" ||
      outcome.code === "channel_not_supported";

    if (configurationGap) blocked += 1;
    else if (giveUp) failed += 1;

    await admin
      .from("communication_queue")
      .update({
        status: configurationGap ? "queued" : giveUp ? "failed" : "queued",
        provider: provider.provider,
        failure_code: outcome.code,
        failure_message: outcome.detail,
        attempt_count: configurationGap ? (message.attempt_count ?? 0) : attempt,
        last_attempt_at: now,
      })
      .eq("id", message.id);

    await logEvent(
      message.id,
      provider.provider,
      configurationGap ? "blocked" : giveUp ? "failed" : "deferred",
      outcome.detail,
    );
  }

  return { queued: queued?.length ?? 0, blocked, sent, failed };
}

/** Creates the two-day supplier/requester reminders before queue delivery. */
export async function queueProcurementDeliveryReminders(admin: Admin, org: string) {
  const { data, error } = await admin.rpc("neo_procurement_queue_due_reminders", { _organization: org });
  if (error) throw new Error(`Procurement reminders could not be prepared: ${error.message}`);
  return data as { queued?: number; skipped?: number; cadence_days?: number } | null;
}

/** Raises escalations for genuinely overdue tasks and approvals. */
export async function processEscalations(admin: Admin, org: string) {
  const now = new Date().toISOString();
  const { data: policies } = await admin
    .from("escalation_policies")
    .select("*")
    .eq("organization_id", org)
    .eq("active", true);

  let raised = 0;

  for (const policy of policies ?? []) {
    if (policy.applies_to === "task") {
      const { data: overdue } = await admin
        .from("tasks")
        .select("id, title, department_id, branch_id, assignee_id, due_at, status")
        .eq("organization_id", org)
        .lt("due_at", now)
        .not("status", "in", "(completed,cancelled)")
        .limit(50);
      for (const task of overdue ?? []) {
        const key = `task-overdue:${task.id}:${policy.id}`;
        const claimed = await claimIdempotency(admin, org, "escalation", key);
        if (!claimed) continue;
        const { data: dept } = await admin
          .from("departments")
          .select("manager_id")
          .eq("id", task.department_id ?? "")
          .maybeSingle();
        await admin.from("escalation_events").insert({
          organization_id: org,
          branch_id: task.branch_id,
          department_id: task.department_id,
          policy_id: policy.id,
          subject_type: "task",
          subject_id: task.id,
          level: 1,
          reason: `Task "${task.title}" passed its due time.`,
          notified_user_id: dept?.manager_id ?? task.assignee_id,
          idempotency_key: key,
        });
        if (dept?.manager_id) {
          await admin.from("notifications").insert({
            organization_id: org,
            user_id: dept.manager_id,
            branch_id: task.branch_id,
            department_id: task.department_id,
            category: "escalation",
            title: "Overdue task escalated",
            body: `"${task.title}" is overdue and has been escalated by policy "${policy.name}".`,
            link: "/tasks",
            priority: "high",
          });
        }
        raised += 1;
      }
    }

    if (policy.applies_to === "approval") {
      const { data: pending } = await admin
        .from("approval_requests")
        .select("id, reference, branch_id, department_id, approver_id, created_at, status")
        .eq("organization_id", org)
        .eq("status", "pending_approval")
        .lt("created_at", new Date(Date.now() - 24 * 3_600_000).toISOString())
        .limit(50);
      for (const approval of pending ?? []) {
        const key = `approval-overdue:${approval.id}:${policy.id}`;
        const claimed = await claimIdempotency(admin, org, "escalation", key);
        if (!claimed) continue;
        await admin.from("escalation_events").insert({
          organization_id: org,
          branch_id: approval.branch_id,
          department_id: approval.department_id,
          policy_id: policy.id,
          subject_type: "approval",
          subject_id: approval.id,
          level: 1,
          reason: `Approval ${approval.reference} has been pending for more than 24 hours.`,
          notified_user_id: approval.approver_id,
          idempotency_key: key,
        });
        raised += 1;
      }
    }
  }
  return raised;
}

/** Simulated run used by Test Mode — nothing is written to operational tables. */
export async function simulateWorkflow(
  admin: Admin,
  org: string,
  workflowId: string,
  eventId: string | null,
): Promise<{ matched: boolean; outcomes: ActionOutcome[]; warnings: string[]; eventType: string | null }> {
  const warnings: string[] = [];
  const { data: workflow } = await admin
    .from("workflow_definitions")
    .select("*")
    .eq("id", workflowId)
    .eq("organization_id", org)
    .maybeSingle();
  if (!workflow) throw new Error("Workflow not found");

  const { data: version } = await admin
    .from("workflow_versions")
    .select("*")
    .eq("workflow_id", workflowId)
    .eq("version", workflow.current_version)
    .maybeSingle();
  if (!version) throw new Error("Workflow has no version to test");

  let event: EventRow | null = null;
  if (eventId) {
    const { data } = await admin.from("domain_events").select("*").eq("id", eventId).eq("organization_id", org).maybeSingle();
    event = data;
  } else {
    const { data } = await admin
      .from("domain_events")
      .select("*")
      .eq("organization_id", org)
      .eq("event_type", workflow.trigger_event ?? "")
      .order("occurred_at", { ascending: false })
      .limit(1)
      .maybeSingle();
    event = data;
  }
  if (!event) {
    warnings.push("No real event of this type has occurred yet, so conditions could not be evaluated against live data.");
  }

  const conditions = (version.conditions ?? []) as unknown as Condition[];
  const actions = (version.actions ?? []) as unknown as Action[];
  const matched = event ? evaluateConditions(event, conditions) : false;

  const { data: providers } = await admin
    .from("provider_connections")
    .select("channel, status, enabled")
    .eq("organization_id", org);

  const outcomes: ActionOutcome[] = [];
  for (let i = 0; i < actions.length; i += 1) {
    const action = actions[i] as Action;
    const config = action.config ?? {};
    if (action.type === "queue_message" || action.type === "feedback_invitation") {
      const channel = String(config["channel"] ?? "email");
      const provider = (providers ?? []).find((p) => p.channel === channel && p.enabled && p.status === "connected");
      if (!provider) warnings.push(`No connected ${channel} provider — messages would remain queued.`);
      const templateKey = String(config["template_key"] ?? "");
      if (!templateKey) warnings.push(`Action ${i + 1} has no message template selected.`);
    }
    if (["notify_role", "notify_user", "notify_department"].includes(action.type)) {
      const { userIds } = await resolveRecipients(admin, org, action);
      if (!userIds.length) warnings.push(`Action ${i + 1} resolves to no recipients.`);
    }
    outcomes.push({
      index: i,
      type: action.type,
      status: "simulated",
      detail: `Would run "${action.type}"${action.delay_minutes ? ` after ${action.delay_minutes} minutes` : ""}.`,
    });
  }

  await admin.from("workflow_runs").insert({
    organization_id: org,
    workflow_id: workflowId,
    workflow_version: version.version,
    event_id: event?.id ?? null,
    mode: "test",
    status: "completed",
    matched,
    actions_total: actions.length,
    actions_completed: 0,
    finished_at: new Date().toISOString(),
  });

  return { matched, outcomes, warnings, eventType: event?.event_type ?? null };
}
