  48:   END IF;
  49:   INSERT INTO public.print_jobs(organization_id, branch_id, printer_id, original_printer_id, document_type, title, payload, status, max_attempts, idempotency_key, created_by)
  50:   VALUES (p.organization_id, p.branch_id, p.id, p.id, _document_type, _title, coalesce(_payload,'{}'::jsonb), 'queued', greatest(1, coalesce(p.retry_limit,3)), _idempotency, auth.uid())
  51:   RETURNING id INTO j;
  52:   RETURN j;
  53: END $$;

  55: CREATE OR REPLACE FUNCTION public.neo_print_record_result(_job uuid, _success boolean, _error text DEFAULT NULL)
  56: RETURNS jsonb
  57: LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
  58: DECLARE jb public.print_jobs; fb uuid; new_status text; fell_over boolean := false; backoff integer;
  59: BEGIN
  60:   SELECT * INTO jb FROM public.print_jobs WHERE id = _job;
  61:   IF jb.id IS NULL THEN RAISE EXCEPTION 'Print job not found'; END IF;
  62:   IF jb.organization_id <> current_org_id() THEN RAISE EXCEPTION 'Print job belongs to another workspace'; END IF;
  63:   IF NOT (has_permission('pos','receipts.print') OR has_permission('pos','settings.manage') OR is_org_admin_for(jb.organization_id)) THEN
  64:     RAISE EXCEPTION 'permission_denied';
  65:   END IF;
  66: 
  67:   IF _success THEN
  68:     UPDATE public.print_jobs SET status='completed', attempts=attempts+1, completed_at=now(), last_error=NULL WHERE id=_job;
  69:     UPDATE public.printer_profiles SET health_status='online', last_success_at=now(), last_checked_at=now(), last_error=NULL WHERE id=jb.printer_id;
  70:     RETURN jsonb_build_object('status','completed','failed_over',false);
  71:   END IF;
  72: 
  73:   SELECT fallback_printer_id, coalesce(retry_backoff_seconds,30) INTO fb, backoff
  74:   FROM public.printer_profiles WHERE id = jb.printer_id;
  75: 
  76:   UPDATE public.printer_profiles
  77:      SET health_status='error', last_failure_at=now(), last_checked_at=now(), last_error=_error
  78:    WHERE id = jb.printer_id;
  79: 
  80:   IF jb.attempts + 1 < jb.max_attempts THEN
  81:     new_status := 'queued';
  82:     UPDATE public.print_jobs
  83:        SET status='queued', attempts=attempts+1, last_error=_error,
  84:            next_attempt_at = now() + make_interval(secs => backoff)
  85:      WHERE id=_job;
  86:   ELSIF fb IS NOT NULL AND NOT jb.failed_over THEN
  87:     new_status := 'queued'; fell_over := true;
  88:     UPDATE public.print_jobs
  89:        SET status='queued', attempts=0, printer_id=fb, failed_over=true, last_error=_error, next_attempt_at=now()
  90:      WHERE id=_job;
  91:   ELSE
  92:     new_status := 'failed';
  93:     UPDATE public.print_jobs SET status='failed', attempts=attempts+1, last_error=_error WHERE id=_job;
  94:   END IF;
  95: 
  96:   RETURN jsonb_build_object('status', new_status, 'failed_over', fell_over);
  97: END $$;

  99: CREATE OR REPLACE FUNCTION public.neo_print_retry(_job uuid)
 100: RETURNS uuid
 101: LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
 102: DECLARE jb public.print_jobs;
 103: BEGIN
 104:   SELECT * INTO jb FROM public.print_jobs WHERE id=_job;
 105:   IF jb.id IS NULL THEN RAISE EXCEPTION 'Print job not found'; END IF;
 106:   IF jb.organization_id <> current_org_id() THEN RAISE EXCEPTION 'Print job belongs to another workspace'; END IF;
 107:   IF NOT (has_permission('pos','receipts.print') OR has_permission('pos','settings.manage') OR is_org_admin_for(jb.organization_id)) THEN
 108:     RAISE EXCEPTION 'permission_denied';
 109:   END IF;
 110:   UPDATE public.print_jobs SET status='queued', attempts=0, next_attempt_at=now(), last_error=NULL WHERE id=_job;
 111:   RETURN _job;
 112: END $$;
