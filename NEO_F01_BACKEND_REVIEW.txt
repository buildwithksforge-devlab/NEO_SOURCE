  255: -- One PIN authorises one action. Expected status + row locks prevent two staff
  256: -- claiming the same work. Request IDs make uncertain-network retries idempotent.
  257: CREATE OR REPLACE FUNCTION public.neo_production_display_act(
  258:   _token text, _code text, _pin text, _ticket uuid, _to text,
  259:   _expected text, _request uuid, _reason text DEFAULT NULL
  260: )
  261: RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$
  262: DECLARE s public.neo_production_display_sessions; c public.neo_production_staff_codes;
  263:   t public.station_tickets; a public.neo_production_display_actions;
  264:   v_result jsonb; v_actor_name text; v_error text; v_bar boolean;
  265:   v_claims text:=current_setting('request.jwt.claims',true);
  266:   v_sub text:=current_setting('request.jwt.claim.sub',true);
  267:   v_role text:=current_setting('request.jwt.claim.role',true);
  268: BEGIN
  269:   s:=public.neo_production_require_session(_token);
  270:   SELECT * INTO s FROM public.neo_production_display_sessions WHERE id=s.id FOR UPDATE;
  271:   IF s.closed_at IS NOT NULL OR s.expires_at<=now() OR s.read_only THEN
  272:     RETURN jsonb_build_object('ok',false,'reason','display_unavailable','message','This display cannot perform production actions.');
  273:   END IF;
  274:   IF s.blocked_until>now() THEN
  275:     RETURN jsonb_build_object('ok',false,'reason','locked','message','Too many attempts. Ask a supervisor for help.','retry_at',s.blocked_until);
  276:   END IF;
  277:   IF _request IS NULL OR _expected IS NULL OR _ticket IS NULL
  278:      OR _to IS NULL OR _to NOT IN ('accepted','preparing','ready','delay')
  279:      OR length(coalesce(_reason,''))>500
  280:      OR (_to='delay' AND length(btrim(coalesce(_reason,'')))<3) THEN
  281:     RETURN jsonb_build_object('ok',false,'reason','invalid_action','message','Choose a valid production action and required reason.');
  282:   END IF;
  283:   PERFORM public.neo_production_context(s.opened_by);
  284:   PERFORM public.neo_production_assert_supervisor(s.id);
  285:   -- Restore before credential checking; no supervisor authority reaches staff work.
  286:   PERFORM set_config('request.jwt.claims',coalesce(v_claims,''),true);
  287:   PERFORM set_config('request.jwt.claim.sub',coalesce(v_sub,''),true);
  288:   PERFORM set_config('request.jwt.claim.role',coalesce(v_role,''),true);
  289:   PERFORM public.neo_access_cache_bump();
  290: 
  291:   SELECT * INTO c FROM public.neo_production_staff_codes
  292:     WHERE organization_id=s.organization_id AND staff_code=upper(btrim(left(coalesce(_code,''),24))) FOR UPDATE;
  293:   IF c.blocked_until>now() THEN
  294:     RETURN jsonb_build_object('ok',false,'reason','credentials','message','Staff code or PIN not accepted. Try later or reset your PIN in your staff account.');
  295:   END IF;
  296:   IF c.user_id IS NULL OR length(btrim(coalesce(_code,'')))>24 OR NOT coalesce(c.enabled,false) OR _pin IS NULL OR _pin !~ '^[0-9]{6}$'
  297:      OR extensions.crypt(left(coalesce(_pin,''),6),coalesce(c.pin_hash,extensions.gen_salt('bf',10))) IS DISTINCT FROM c.pin_hash THEN
  298:     -- Return, do not RAISE: rate-limit counters must survive the failed attempt.
  299:     IF s.attempt_window<now()-interval '15 minutes' THEN s.failed_attempts:=0; s.attempt_window:=now(); END IF;
  300:     UPDATE public.neo_production_display_sessions SET failed_attempts=s.failed_attempts+1, attempt_window=s.attempt_window,
  301:       blocked_until=CASE WHEN s.failed_attempts+1>=20 THEN now()+interval '15 minutes' ELSE NULL END WHERE id=s.id;
  302:     IF c.user_id IS NOT NULL THEN
  303:       UPDATE public.neo_production_staff_codes SET
  304:         failed_attempts=CASE WHEN blocked_until IS NOT NULL AND blocked_until<=now() THEN 1 ELSE failed_attempts+1 END,
  305:         blocked_until=CASE WHEN (CASE WHEN blocked_until IS NOT NULL AND blocked_until<=now() THEN 1 ELSE failed_attempts+1 END)>=5
  306:           THEN now()+interval '15 minutes' ELSE NULL END
  307:       WHERE organization_id=s.organization_id AND user_id=c.user_id;
  308:     END IF;
  309:     RETURN jsonb_build_object('ok',false,'reason','credentials','message','Staff code or PIN not accepted. Try later or reset your PIN in your staff account.');
  310:   END IF;
  311: 
  312:   PERFORM public.neo_production_context(c.user_id);
  313:   BEGIN
  314:     IF public.current_org_id() IS DISTINCT FROM s.organization_id OR NOT public.is_active_staff() THEN
  315:       RAISE EXCEPTION 'This staff account is not active in this organisation.' USING ERRCODE='42501';
  316:     END IF;
  317:     SELECT * INTO t FROM public.station_tickets WHERE id=_ticket AND organization_id=s.organization_id FOR UPDATE;
  318:     IF t.id IS NULL OR t.branch_id IS DISTINCT FROM s.branch_id OR t.station_id IS NULL
  319:        OR NOT (t.station_id=ANY(s.station_ids))
  320:        OR NOT (t.station_id=ANY(public.neo_kitchen_visible_station_ids())) THEN
  321:       RAISE EXCEPTION 'This ticket is outside your assigned display stations.' USING ERRCODE='42501';
  322:     END IF;
  323:     PERFORM public.neo_kitchen_assert_station_authority(t.id);
  324:     v_bar:=t.station_id IN (SELECT public.neo_bar_station_ids(t.branch_id));
  325:     IF NOT (public.has_permission('kitchen',CASE _to WHEN 'accepted' THEN 'ticket.accept' WHEN 'ready' THEN 'item.ready' ELSE 'item.prepare' END)
  326:       OR (v_bar AND public.has_permission('bar','queue','edit'))) THEN
  327:       RAISE EXCEPTION 'You do not have permission to perform this station action.' USING ERRCODE='42501';
  328:     END IF;
  329:     SELECT * INTO a FROM public.neo_production_display_actions WHERE session_id=s.id AND request_id=_request;
  330:     IF a.id IS NOT NULL THEN
  331:       IF a.actor_id IS DISTINCT FROM c.user_id OR a.ticket_id IS DISTINCT FROM _ticket
  332:         OR a.to_status IS DISTINCT FROM _to OR a.from_status IS DISTINCT FROM _expected
  333:         OR a.reason IS DISTINCT FROM nullif(btrim(_reason),'') THEN
  334:         RAISE EXCEPTION 'This request ID was already used for a different action.' USING ERRCODE='22023';
  335:       END IF;
  336:       v_result:=a.result || jsonb_build_object('idempotent',true);
  337:     ELSE
  338:       IF t.status IS DISTINCT FROM _expected THEN
  339:         RAISE EXCEPTION 'Ticket changed from % to %. Refresh and review it before acting.',_expected,t.status USING ERRCODE='40001';
  340:       END IF;
  341:       PERFORM id FROM public.station_ticket_items WHERE ticket_id=t.id ORDER BY id FOR UPDATE;
  342:       IF (t.assigned_chef_id IS NOT NULL AND t.assigned_chef_id<>c.user_id
  343:           OR EXISTS(SELECT 1 FROM public.station_ticket_items WHERE ticket_id=t.id
  344:             AND status NOT IN ('cancelled','completed') AND chef_id IS NOT NULL AND chef_id<>c.user_id))
  345:           AND NOT public.neo_can_assign_kitchen_work() THEN
  346:         RAISE EXCEPTION 'Another staff member owns this ticket. A supervisor must reassign it.' USING ERRCODE='42501';
  347:       END IF;
  348:       IF _to='delay' THEN
  349:         IF t.status IN ('ready','served','completed','cancelled') THEN RAISE EXCEPTION 'This ticket is no longer awaiting preparation.'; END IF;
  350:         UPDATE public.station_tickets SET delay_reason=btrim(_reason),updated_at=now() WHERE id=t.id;
  351:         INSERT INTO public.station_ticket_status_history(organization_id,ticket_id,from_status,to_status,reason,changed_by)
  352:           VALUES(s.organization_id,t.id,t.status,t.status,'delay: '||btrim(_reason),c.user_id);
  353:         v_result:=jsonb_build_object('ticket_status',t.status);
  354:       ELSE
  355:         v_result:=public.neo_kitchen_ticket_transition(t.id,_to,NULL);
  356:       END IF;
  357:       SELECT full_name INTO v_actor_name FROM public.profiles WHERE id=c.user_id;
  358:       v_result:=coalesce(v_result,'{}'::jsonb)||jsonb_build_object('ok',true,'actor_name',v_actor_name,'actor_id',c.user_id,'idempotent',false);
  359:       INSERT INTO public.neo_production_display_actions(session_id,request_id,organization_id,branch_id,station_id,ticket_id,
  360:         actor_id,supervisor_id,from_status,to_status,reason,result)
  361:         VALUES(s.id,_request,s.organization_id,s.branch_id,t.station_id,t.id,c.user_id,s.opened_by,
  362:           _expected,_to,nullif(btrim(_reason),''),v_result);
  363:     END IF;
  364:     UPDATE public.neo_production_staff_codes SET failed_attempts=0, blocked_until=NULL WHERE organization_id=s.organization_id AND user_id=c.user_id;
  365:   EXCEPTION WHEN OTHERS THEN

  371:   PERFORM set_config('request.jwt.claim.sub',coalesce(v_sub,''),true);
  372:   PERFORM set_config('request.jwt.claim.role',coalesce(v_role,''),true);
  373:   PERFORM public.neo_access_cache_bump();

  377:   PERFORM set_config('request.jwt.claim.sub',coalesce(v_sub,''),true);
  378:   PERFORM set_config('request.jwt.claim.role',coalesce(v_role,''),true);

    4:  * The department has two roles. The Storekeeper controls what is held in
    5:  * Stores: they receive department requisitions, check availability, issue in
    6:  * full or in part, return a request with a reason, run stock counts and route
    7:  * shortages to Procurement as a purchase request. The Receiving Clerk works
    8:  * only on stock coming INTO the organisation â€” supplier deliveries against an
    9:  * approved order â€” and never issues stock or approves an exception.
   10:  *
   11:  * Every write goes through an existing authoritative routine (req_issue,
   12:  * req_approve, neo_stores_requisition_reject, neo_stores_count_submit_variance)
   13:  * so stock balances and the audit trail stay owned by the database.
   14:  */

  353: /** Open a count for a store and snapshot the system quantities onto its lines. */
  354: export function useStartStockCount() {
  355:   const invalidate = useStoresInvalidate();
  356:   const { organizationId, branchId } = useStoresContext();
  357:   const { user } = useAuth();
  358:   return useMutation({
  359:     mutationFn: async (input: { locationId: string; countType: string; notes?: string }) => {
  360:       if (!organizationId) throw new Error("No organisation context");
  361:       const reference = `CNT-${new Date().toISOString().slice(0, 10).replace(/-/g, "")}-${Math.floor(Math.random() * 9000 + 1000)}`;
  362:       const { data, error } = await db
  363:         .from("stock_takes")
  364:         .insert({
  365:           organization_id: organizationId,
  366:           branch_id: branchId ?? null,
  367:           reference,
  368:           location_id: input.locationId,
  369:           count_type: input.countType,
  370:           status: "in-progress",
  371:           started_at: new Date().toISOString(),
  372:           counted_by: user?.id ?? null,
  373:           notes: input.notes ?? null,
  374:         })
  375:         .select("id")
  376:         .single();
  377:       if (error) throw new Error(error.message);
  378:       const takeId = String((data as StoresRow)["id"]);
  379: 
  380:       // The count only exists once its snapshot lines exist. If the snapshot
  381:       // fails the header is removed again, so no empty count is left behind.
  382:       try {
  383:         const { data: levels, error: lErr } = await db
  384:           .from("stock_levels")
  385:           .select("stock_item_id, quantity_on_hand, stock_items(base_unit, average_cost)")
  386:           .eq("location_id", input.locationId)
  387:           .limit(1000);
  388:         if (lErr) throw new Error(lErr.message);
  389: 
  390:         const lines = ((levels ?? []) as StoresRow[]).map((level) => ({
  391:           organization_id: organizationId,
  392:           stock_take_id: takeId,
  393:           stock_item_id: level["stock_item_id"],
  394:           system_quantity: num(level["quantity_on_hand"]),
  395:           unit: str((level["stock_items"] as StoresRow | null)?.["base_unit"], "unit"),
  396:           unit_cost: num((level["stock_items"] as StoresRow | null)?.["average_cost"]),
  397:         }));
  398:         if (lines.length) {
  399:           const { error: insErr } = await db.from("stock_take_lines").insert(lines);
  400:           if (insErr) throw new Error(insErr.message);
  401:         }
  402:       } catch (snapshotError) {
  403:         await db.from("stock_takes").delete().eq("id", takeId);
  404:         throw snapshotError;
  405:       }
  406: 
  407:       return takeId;
  408:     },
  409:     onSuccess: () => {
  410:       invalidate();
  411:       toast.success("Count opened with the system quantities snapshotted");
  412:     },
  413:     onError: (e: Error) => toast.error(e.message),
  414:   });
  415: }

  453: /** Reject counts that cannot physically exist. Returns an error message, or null. */
  454: export function validateCountedQuantity(counted: number, unit?: string | null): string | null {
  455:   if (!Number.isFinite(counted)) return "Enter a counted quantity";
  456:   if (counted < 0) return "A physical count cannot be negative â€” enter the quantity actually on the shelf";
  457:   const u = String(unit ?? "").trim().toLowerCase();
  458:   if (DISCRETE_UNITS.has(u) && !Number.isInteger(counted))
  459:     return `Count ${u} in whole numbers â€” a part ${u} cannot be counted`;
  460:   return null;
  461: }
